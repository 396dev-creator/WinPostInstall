   157→   157→function Set-DefenderExclusions {
   158→   158→    <#
   159→   159→    .DESCRIPTION
   160→   160→        Adds this script and its working directory to Windows Defender exclusions.
   161→   161→        Prevents the script from being flagged as malware during execution.
   162→   162→        This is safe: exclusions are path-specific and only protect this tool.
   163→   163→    #>
   164→   164→    Write-Host "" -ForegroundColor Cyan
   165→   165→    Write-Host "  [AV SHIELD] Windows Defender Self-Exclusion" -ForegroundColor Cyan
   166→   166→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   167→   167→    Write-Host ""
   168→   168→
   169→   169→    $scriptPath = $PSCommandPath
   170→   170→    if (-not $scriptPath) { $scriptPath = $MyInvocation.PSCommandPath }
   171→   171→
   172→   172→    $exclusionAdded = $false
   173→   173→
   174→   174→    # Check if the exclusion already exists
   175→   175→    try {
   176→   176→        $existingExclusions = (Get-MpPreference -ErrorAction SilentlyContinue).ExclusionPath
   177→   177→        if ($existingExclusions -and $scriptPath -in $existingExclusions) {
   178→   178→            Write-Step "Defender exclusion already exists for this script."
   179→   179→            return
   180→   180→        }
   181→   181→    } catch {
   182→   182→        Write-Warn "Could not read current Defender preferences: $_"
   183→   183→    }
   184→   184→
   185→   185→    # Exclude the script file itself
   186→   186→    Write-Step "Adding script to Defender exclusions..."
   187→   187→    try {
   188→   188→        Add-MpPreference -ExclusionPath $scriptPath -ErrorAction Stop
   189→   189→        Write-Success "Excluded: $scriptPath"
   190→   190→        $exclusionAdded = $true
   191→   191→    } catch {
   192→   192→        Write-Warn "Failed to exclude script path: $_"
   193→   193→    }
   194→   194→
   195→   195→    # Exclude the working directory (where the script lives)
   196→   196→    $scriptDir = Split-Path $scriptPath -Parent
   197→   197→    if ($scriptDir -and (Test-Path $scriptDir)) {
   198→   198→        try {
   199→   199→            Add-MpPreference -ExclusionPath $scriptDir -ErrorAction Stop
   200→   200→            Write-Success "Excluded directory: $scriptDir"
   201→   201→            $exclusionAdded = $true
   202→   202→        } catch {
   203→   203→            Write-Warn "Failed to exclude directory: $_"
   204→   204→        }
   205→   205→    }
   206→   206→
   207→   207→    # Exclude the log directory (C:\Temp)
   208→   208→    $logDir = "$env:SYSTEMDRIVE\Temp"
   209→   209→    try {
   210→   210→        Add-MpPreference -ExclusionPath $logDir -ErrorAction Stop
   211→   211→        Write-Success "Excluded log directory: $logDir"
   212→   212→        $exclusionAdded = $true
   213→   213→    } catch {
   214→   214→        Write-Warn "Failed to exclude log directory: $_"
   215→   215→    }
   216→   216→
   217→   217→    # Exclude common technician tool folders
   218→   218→    $toolPaths = @("C:\Tools", "C:\Scripts", "C:\TechnicianTools")
   219→   219→    foreach ($path in $toolPaths) {
   220→   220→        if (Test-Path $path) {
   221→   221→            try {
   222→   222→                Add-MpPreference -ExclusionPath $path -ErrorAction Stop
   223→   223→                Write-Success "Excluded: $path"
   224→   224→            } catch { }
   225→   225→        }
   226→   226→    }
   227→   227→
   228→   228→    if ($exclusionAdded) {
   229→   229→        Write-Success "Defender exclusions configured. This script is now trusted."
   230→   230→    }
   231→   231→
   232→   232→    Write-Host ""
   233→   233→    Write-Host "  NOTE: This only affects Windows Defender. For third-party AV," -ForegroundColor DarkGray
   234→   234→    Write-Host "  consider purchasing a Code Signing Certificate for full compatibility." -ForegroundColor DarkGray
   235→   235→    Write-Host ""
   236→   236→}

   307→   307→        }
   308→   308→    } catch { }
   309→   309→    if (-not $script:IsLaptop) {
   310→   310→        try {
   311→   311→            $script:IsLaptop = (Get-CimInstance -ClassName Win32_Battery -ErrorAction SilentlyContinue).Count -gt 0
   312→   312→        } catch { }
   313→   313→    }
   314→   314→
   315→   315→    Write-Host ""
   316→   316→    Write-Host "================================================================" -ForegroundColor Cyan
   317→   317→    Write-Host "  Windows Post-Installation Optimizer v$ScriptVersion" -ForegroundColor Cyan
   318→   318→    Write-Host "  Target: $script:OSCaption (Build $script:OSBuild)" -ForegroundColor Cyan
   319→   319→    Write-Host "  Computer: $script:LocalComputerName" -ForegroundColor Cyan
   320→   320→    Write-Host "  Chassis: $(if ($script:IsLaptop) { 'Laptop' } else { 'Desktop' })" -ForegroundColor Cyan
   321→   321→    Write-Host "  Log file: $LogFile" -ForegroundColor DarkGray

   329→   329→function Test-Prerequisites {
   330→   330→    <#
   331→   331→    .DESCRIPTION
   332→   332→        Pre-flight checks: internet connectivity, disk space, pending reboot, battery.
   333→   333→        Warns on issues but does not block execution.
   334→   334→    #>
   335→   335→    Write-Host "" -ForegroundColor Cyan
   336→   336→    Write-Host "  [PRE-FLIGHT] System Prerequisites Check" -ForegroundColor Cyan
   337→   337→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   338→   338→    Write-Host ""
   339→   339→
   340→   340→    $checksPassed = 0; $checksWarned = 0
   341→   341→
   342→   342→    # 1. Internet connectivity
   343→   343→    Write-Step "Checking internet connectivity..."
   344→   344→    try {
   345→   345→        $netTest = Test-Connection -ComputerName 8.8.8.8 -Count 1 -Quiet -ErrorAction Stop
   346→   346→        if ($netTest) {
   347→   347→            Write-Success "Internet connectivity: OK"
   348→   348→            $checksPassed++
   349→   349→        } else {
   350→   350→            Write-Warn "Internet connectivity: OFFLINE (some features may fail)"
   351→   351→            $checksWarned++
   352→   352→        }
   353→   353→    } catch {
   354→   354→        Write-Warn "Internet connectivity: Could not check (some features may fail)"
   355→   355→        $checksWarned++
   356→   356→    }
   357→   357→
   358→   358→    # 2. Free disk space
   359→   359→    Write-Step "Checking free disk space..."
   360→   360→    try {
   361→   361→        $sysDrive = Get-PSDrive -Name ($env:SYSTEMDRIVE -replace ':','') -ErrorAction Stop
   362→   362→        $freeGB = [math]::Round($sysDrive.Free / 1GB, 2)
   363→   363→        if ($freeGB -gt 10) {
   364→   364→            Write-Success "Free disk space: $freeGB GB (OK)"
   365→   365→            $checksPassed++
   366→   366→        } else {
   367→   367→            Write-Warn "Free disk space: $freeGB GB (LOW - recommend >10 GB)"
   368→   368→            $checksWarned++
   369→   369→        }
   370→   370→    } catch {
   371→   371→        Write-Warn "Could not check disk space: $_"
   372→   372→        $checksWarned++
   373→   373→    }
   374→   374→
   375→   375→    # 3. Pending reboot
   376→   376→    Write-Step "Checking for pending reboot..."
   377→   377→    $rebootPending = $false
   378→   378→    $cbsPath = "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Component Based Servicing\RebootPending"
   379→   379→    $wuPath = "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\WindowsUpdate\Auto Update\RebootRequired"
   380→   380→    if (Test-Path $cbsPath -ErrorAction SilentlyContinue) { $rebootPending = $true }
   381→   381→    if (Test-Path $wuPath -ErrorAction SilentlyContinue) { $rebootPending = $true }
   382→   382→    if ($rebootPending) {
   383→   383→        Write-Warn "Pending reboot detected! A restart is recommended before running optimizations."
   384→   384→        $checksWarned++
   385→   385→    } else {
   386→   386→        Write-Success "No pending reboot."
   387→   387→        $checksPassed++
   388→   388→    }
   389→   389→
   390→   390→    # 4. Battery check (if laptop)
   391→   391→    if ($script:IsLaptop) {
   392→   392→        Write-Step "Checking battery level..."
   393→   393→        try {
   394→   394→            $battery = Get-CimInstance -ClassName Win32_Battery -ErrorAction Stop
   395→   395→            $onAC = $battery.BatteryStatus -eq 2 -or $battery.BatteryStatus -eq 6 -or $battery.BatteryStatus -eq 7 -or $battery.BatteryStatus -eq 8
   396→   396→            if ($onAC) {
   397→   397→                Write-Success "On AC power."
   398→   398→                $checksPassed++
   399→   399→            } elseif ($battery.EstimatedChargeRemaining -gt 50) {
   400→   400→                Write-Success "Battery level: $($battery.EstimatedChargeRemaining)% (OK)"
   401→   401→                $checksPassed++
   402→   402→            } else {
   403→   403→                Write-Warn "Battery level: $($battery.EstimatedChargeRemaining)% - recommend AC power or >50% charge."
   404→   404→                $checksWarned++
   405→   405→            }
   406→   406→        } catch {
   407→   407→            Write-Warn "Could not check battery: $_"
   408→   408→            $checksWarned++
   409→   409→        }
   410→   410→    }
   411→   411→
   412→   412→    Write-Host ""
   413→   413→    Write-Host "  Pre-flight summary: $checksPassed passed, $checksWarned warnings" -ForegroundColor $(if ($checksWarned -gt 0) { 'Yellow' } else { 'Green' })
   414→   414→    Write-Host ""
   415→   415→}

   420→   420→        Detects whether the boot drive is an SSD or HDD.
   421→   421→        Returns "SSD" or "HDD".
   422→   422→    #>
   423→   423→    $bootType = "HDD"  # Default assumption
   424→   424→    try {
   425→   425→        # Method 1: Get-PhysicalDisk (most reliable on Win8+)
   426→   426→        $physicalDisk = Get-PhysicalDisk -ErrorAction SilentlyContinue | Where-Object DeviceId -eq 0
   427→   427→        if ($physicalDisk) {
   428→   428→            if ($physicalDisk.MediaType -eq "SSD") { return "SSD" }
   429→   429→            if ($physicalDisk.MediaType -eq "HDD") { return "HDD" }
   430→   430→        }
   431→   431→    } catch { }
   432→   432→
   433→   433→    try {
   434→   434→        # Method 2: Get-Disk BusType fallback
   435→   435→        $bootDisk = Get-Disk -ErrorAction SilentlyContinue | Where-Object IsBoot -eq $true
   436→   436→        if ($bootDisk) {
   437→   437→            $ssdBusTypes = @("NVMe", "SATA", "RAID", "SCSI")
   438→   438→            # Check via rotation rate (0 = SSD on some controllers)
   439→   439→            if ($bootDisk.BusType -in $ssdBusTypes) {
   440→   440→                # Not conclusive, but check rotation rate
   441→   441→                $detail = Get-PhysicalDisk -ErrorAction SilentlyContinue | Where-Object DeviceId -eq $bootDisk.Number
   442→   442→                if ($detail -and $detail.MediaType -eq "SSD") { return "SSD" }
   443→   443→            }
   444→   444→        }
   445→   445→    } catch { }
   446→   446→
   447→   447→    try {
   448→   448→        # Method 3: Check if defrag is scheduled (SSD = not scheduled)
   449→   449→        $defragTask = Get-ScheduledTask -TaskName "ScheduledDefrag" -ErrorAction SilentlyContinue
   450→   450→        if ($defragTask -and $defragTask.State -eq 'Disabled') {
   451→   451→            # If scheduled defrag is disabled, likely SSD
   452→   452→            # This is a weak heuristic, don't rely on it alone
   453→   453→        }
   454→   454→    } catch { }
   455→   455→
   456→   456→    return $bootType
   457→   457→}

   475→   475→function Enable-SystemRestore {
   476→   476→    <#
   477→   477→    .DESCRIPTION
   478→   478→        Ensures System Restore is enabled and configured on the system drive.
   479→   479→        Enables the service if disabled, allocates disk space, and verifies
   480→   480→        the feature is operational before returning.
   481→   481→    #>
   482→   482→    Write-Host "" -ForegroundColor Cyan
   483→   483→    Write-Host "  [SAFETY] System Restore Verification" -ForegroundColor Cyan
   484→   484→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   485→   485→    Write-Host ""
   486→   486→
   487→   487→    $systemDrive = $env:SYSTEMDRIVE
   488→   488→    $restoreEnabled = $false
   489→   489→

   493→   493→        $srStatus = Get-ComputerInfo -Property "SystemRestoreStatus" -ErrorAction SilentlyContinue

   495→   495→        $srInfo = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction SilentlyContinue
   496→   496→        if ($srInfo) {
   497→   497→            foreach ($sr in $srInfo) {
   498→   498→                if ($sr.RPSequence -gt 0) {
   499→   499→                    $restoreEnabled = $true
   500→   500→                    break
   501→   501→                }
   502→   502→            }
   503→   503→        }
   504→   504→    } catch {
   505→   505→        # WMI query may fail if System Restore is completely disabled
   506→   506→        $restoreEnabled = $false
   507→   507→    }
   508→   508→
   509→   509→    # Alternative check: try to list existing restore points
   510→   510→    if (-not $restoreEnabled) {
   511→   511→        try {
   512→   512→            $existingPoints = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction Stop
   513→   513→            if ($existingPoints -and $existingPoints.Count -ge 0) {
   514→   514→                $restoreEnabled = $true
   515→   515→            }
   516→   516→        } catch {
   517→   517→            $restoreEnabled = $false
   518→   518→        }
   519→   519→    }
   520→   520→
   521→   521→    # Step 2: Enable System Restore if not currently enabled
   522→   522→    if (-not $restoreEnabled) {
   523→   523→        Write-Warn "System Restore is NOT enabled on $systemDrive."
   524→   524→        Write-Step "Enabling System Restore on $systemDrive..."
   525→   525→
   526→   526→        try {
   527→   527→            # Enable System Restore via registry (most reliable method)
   528→   528→            $srPolicyPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore"
   529→   529→            if (-not (Test-Path $srPolicyPath)) {
   530→   530→                New-Item -Path $srPolicyPath -Force -ErrorAction SilentlyContinue | Out-Null
   531→   531→            }
   532→   532→            # DisableSR = 0 means System Restore is ENABLED
   533→   533→            Set-ItemProperty -Path $srPolicyPath -Name "DisableSR" -Value 0 -Type DWord -Force -ErrorAction Stop
   534→   534→
   535→   535→            # Also enable via PowerShell cmdlet (Windows 10/11)
   536→   536→            try {
   537→   537→                Enable-ComputerRestore -Drive $systemDrive -ErrorAction Stop
   538→   538→            } catch {
   539→   539→                # Fallback: use vssadmin to enable shadow copies
   540→   540→                $vssResult = vssadmin list shadows 2>&1
   541→   541→                if ($LASTEXITCODE -ne 0) {
   542→   542→                    # Try creating a shadow copy to force-enable VSS
   543→   543→                    vssadmin create shadow /For=$systemDrive 2>&1 | Out-Null
   544→   544→                }
   545→   545→            }
   546→   546→
   547→   547→            # Start the System Restore service if not running
   548→   548→            $srService = Get-Service -Name "sr" -ErrorAction SilentlyContinue
   549→   549→            if ($srService -and $srService.Status -ne 'Running') {
   550→   550→                try {
   551→   551→                    Set-Service -Name "sr" -StartupType Manual -ErrorAction Stop
   552→   552→                    Start-Service -Name "sr" -ErrorAction Stop
   553→   553→                } catch {
   554→   554→                    Write-Warn "Could not start System Restore service: $_"
   555→   555→                }
   556→   556→            }
   557→   557→
   558→   558→            # Also ensure VSS (Volume Shadow Copy) service is running (required for System Restore)
   559→   559→            $vssService = Get-Service -Name "VSS" -ErrorAction SilentlyContinue
   560→   560→            if ($vssService -and $vssService.Status -ne 'Running') {
   561→   561→                try {
   562→   562→                    Set-Service -Name "VSS" -StartupType Manual -ErrorAction Stop
   563→   563→                    Start-Service -Name "VSS" -ErrorAction Stop
   564→   564→                    Write-Success "VSS service started."
   565→   565→                } catch {
   566→   566→                    Write-Warn "Could not start VSS service: $_"
   567→   567→                }
   568→   568→            }
   569→   569→
   570→   570→            Start-Sleep -Seconds 2
   571→   571→            Write-Success "System Restore has been ENABLED on $systemDrive."
   572→   572→        } catch {
   573→   573→            Write-Err "Failed to enable System Restore: $_"
   574→   574→            Write-Warn "Continuing without restore point. Manual enable: System Properties > System Protection > Configure > Turn on"
   575→   575→        }
   576→   576→    } else {
   577→   577→        Write-Success "System Restore is already enabled on $systemDrive."
   578→   578→    }
   579→   579→
   580→   580→    # Step 3: Configure disk space allocation for restore points
   581→   581→    Write-Step "Configuring System Restore disk space..."
   582→   582→    try {
   583→   583→        # Set maximum disk space usage to 5% (reasonable for most systems)
   584→   584→        # This is done via registry: RPSessionInterval and DiskPercent
   585→   585→        $srCfgPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore\Cfg"
   586→   586→        if (-not (Test-Path $srCfgPath)) {
   587→   587→            New-Item -Path $srCfgPath -Force -ErrorAction SilentlyContinue | Out-Null
   588→   588→        }
   589→   589→        # DiskPercent = 5 means 5% of disk (0 = unlimited, we use 5% as safe default)
   590→   590→        Set-ItemProperty -Path $srCfgPath -Name "DiskPercent" -Value 5 -Type DWord -Force -ErrorAction SilentlyContinue
   591→   591→        # RPSessionInterval = 0 means create restore points at any interval
   592→   592→        Set-ItemProperty -Path $srCfgPath -Name "RPSessionInterval" -Value 0 -Type DWord -Force -ErrorAction SilentlyContinue
   593→   593→        Write-Success "Restore disk space set to 5% of drive."
   594→   594→    } catch {
   595→   595→        Write-Warn "Could not configure restore disk space: $_"
   596→   596→    }
   597→   597→
   598→   598→    Write-Host ""
   599→   599→}

   601→   601→function New-SystemRestorePoint {
   602→   602→    <#
   603→   603→    .DESCRIPTION
   604→   604→        Creates a System Restore point before any optimization changes are made.
   605→   605→        This provides a safety net so the user can roll back all changes.
   606→   606→    #>
   607→   607→    Write-Host "" -ForegroundColor Cyan
   608→   608→    Write-Host "  [SAFETY] Creating System Restore Point" -ForegroundColor Cyan
   609→   609→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   610→   610→    Write-Host ""
   611→   611→
   612→   612→    $restorePointCreated = $false
   613→   613→    $description = "Pre-Optimization Restore Point - WinPostInstall v$ScriptVersion - $(Get-Date -Format 'yyyy-MM-dd HH:mm')"
   614→   614→
   615→   615→    # Method 1: Use Checkpoint-Computer (PowerShell cmdlet, Windows 8+)
   616→   616→    Write-Step "Creating restore point: 'Pre-Optimization'..."
   617→   617→    try {
   618→   618→        Checkpoint-Computer -Description $description -RestorePointType "MODIFY_SETTINGS" -ErrorAction Stop
   619→   619→        $restorePointCreated = $true
   620→   620→        Write-Success "Restore point created successfully via Checkpoint-Computer."
   621→   621→    } catch {
   622→   622→        Write-Warn "Checkpoint-Computer failed: $_"
   623→   623→    }
   624→   624→
   625→   625→    # Method 2: Fallback using WMI if Method 1 failed
   626→   626→    if (-not $restorePointCreated) {
   627→   627→        Write-Step "Trying WMI method to create restore point..."
   628→   628→        try {
   629→   629→            $wmiSR = [System.Windows.Forms.SystemRestore]
   630→   630→            # WMI approach via Invoke-CimMethod
   631→   631→            $result = Invoke-CimMethod -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" `
   632→   632→                -MethodName "CreateRestorePoint" `
   633→   633→                -Arguments @{
   634→   634→                    Description         = $description
   635→   635→                    RestorePointType    = 12  # MODIFY_SETTINGS
   636→   636→                    EventType           = 100  # BEGIN_NESTED_SYSTEM_CHANGE
   637→   637→                } -ErrorAction Stop
   638→   638→            if ($result -and $result.ReturnValue -eq 0) {
   639→   639→                $restorePointCreated = $true
   640→   640→                Write-Success "Restore point created via WMI."
   641→   641→            } else {
   642→   642→                Write-Warn "WMI returned non-zero: $result"
   643→   643→            }
   644→   644→        } catch {
   645→   645→            Write-Warn "WMI method failed: $_"
   646→   646→        }
   647→   647→    }
   648→   648→
   649→   649→    # Method 3: Final fallback via srclient.dll COM automation
   650→   650→    if (-not $restorePointCreated) {
   651→   651→        Write-Step "Trying COM automation (srclient.dll)..."
   652→   652→        try {
   653→   653→            $srClient = New-Object -ComObject "SRClient.RestorePoint" -ErrorAction Stop
   654→   654→            # 7 = MODIFY_SETTINGS, 100 = BEGIN_NESTED_SYSTEM_CHANGE
   655→   655→            $srClient.CreateRestorePoint($description, 7, 100)
   656→   656→            $restorePointCreated = $true
   657→   657→            Write-Success "Restore point created via COM automation."
   658→   658→        } catch {
   659→   659→            Write-Warn "COM method failed: $_"
   660→   660→        }
   661→   661→    }
   662→   662→
   663→   663→    if ($restorePointCreated) {
   664→   664→        Write-Host ""
   665→   665→        Write-Host "  ====================================================================" -ForegroundColor Green
   666→   666→        Write-Host "  RESTORE POINT CREATED: 'Pre-Optimization'" -ForegroundColor Green
   667→   667→        Write-Host "  You can roll back all changes via:" -ForegroundColor White
   668→   668→        Write-Host "    System Properties > System Protection > System Restore" -ForegroundColor White
   669→   669→        Write-Host "  ====================================================================" -ForegroundColor Green
   670→   670→        Write-Host ""
   671→   671→    } else {
   672→   672→        Write-Host ""
   673→   673→        Write-Warn "Could NOT create restore point automatically."
   674→   674→        Write-Host "  Manual option: System Properties > System Protection > Create" -ForegroundColor Yellow
   675→   675→        Write-Host "  Continuing without restore point backup." -ForegroundColor Yellow
   676→   676→        Write-Host ""
   677→   677→        if (-not $Silent) {
   678→   678→            $continue = Read-Host "  Continue without restore point? (y/N)"
   679→   679→            if ($continue -ne 'y' -and $continue -ne 'Y') {
   680→   680→                Write-Host "  Aborted by user." -ForegroundColor Yellow
   681→   681→                exit 0
   682→   682→            }
   683→   683→        }
   684→   684→    }
   685→   685→}

   687→   687→function Set-Registry {
   688→   688→    [CmdletBinding()]
   689→   689→    param(
   690→   690→        [string]$Path,
   691→   691→        [string]$Name,
   692→   692→        $Value,
   693→   693→        [string]$Type = "DWord",
   694→   694→        [ValidateSet("Safe","Moderate","Risky")]
   695→   695→        [string]$SafetyLevel = "Safe",
   696→   696→        [string]$Description = "",
   697→   697→        [string]$SettingID = ""
   698→   698→    )
   699→   699→
   700→   700→    # Generate a setting ID if not provided
   701→   701→    if ([string]::IsNullOrEmpty($SettingID)) {
   702→   702→        $SettingID = "$($Path.Replace('\','_').Replace(':','').Replace(' ',''))_$Name"
   703→   703→    }
   704→   704→
   705→   705→    # Register in setting catalog for interactive menu
   706→   706→    if (-not [string]::IsNullOrEmpty($Description)) {
   707→   707→        $script:SettingCatalog[$SettingID] = @{
   708→   708→            Section  = $script:CurrentSection
   709→   709→            Safety   = $SafetyLevel
   710→   710→            Desc     = $Description
   711→   711→            Action   = "$Path\$Name = $Value"
   712→   712→            Type     = "Registry"
   713→   713→        }
   714→   714→    }
   715→   715→
   716→   716→    # Collect risky descriptions for menu warnings
   717→   717→    if ($SafetyLevel -eq "Risky" -and -not [string]::IsNullOrEmpty($Description)) {
   718→   718→        $script:RiskyDescriptions += "$SettingID : $Description"
   719→   719→    }
   720→   720→
   721→   721→    # Check if user has toggled this setting OFF via interactive menu
   722→   722→    if ($script:SettingOverrides.ContainsKey($SettingID) -and $script:SettingOverrides[$SettingID]) {
   723→   723→        Write-Host "  SKIPPED (user toggle): $Description" -ForegroundColor DarkGray
   724→   724→        $script:TotalSkipped++
   725→   725→        return
   726→   726→    }
   727→   727→
   728→   728→    # Safety gate: Skip Risky settings unless explicitly allowed
   729→   729→    if ($SafetyLevel -eq "Risky" -and -not $script:AllowRisky) {
   730→   730→        Write-Host "  SKIPPED (risky): $Description — use -AllowRisky or enable in menu" -ForegroundColor Yellow
   731→   731→        $script:TotalSkipped++
   732→   732→        return
   733→   733→    }
   734→   734→
   735→   735→    # Warn for Moderate settings
   736→   736→    if ($SafetyLevel -eq "Moderate") {
   737→   737→        Write-Host "  [MODERATE] $Description" -ForegroundColor DarkYellow
   738→   738→    }
   739→   739→
   740→   740→    try {
   741→   741→        if (-not (Test-Path $Path)) {
   742→   742→            New-Item -Path $Path -Force -ErrorAction SilentlyContinue | Out-Null
   743→   743→        }
   744→   744→        # Idempotency: Check if the value already matches what we want to set
   745→   745→        $currentValue = $null
   746→   746→        try {
   747→   747→            $currentValue = (Get-ItemProperty -Path $Path -Name $Name -ErrorAction Stop).$Name
   748→   748→        } catch {
   749→   749→            # Property doesn't exist yet — need to create it
   750→   750→        }
   751→   751→
   752→   752→        # Compare current vs desired value
   753→   753→        $valueMatches = $false
   754→   754→        if ($null -ne $currentValue) {
   755→   755→            if ($Type -eq "DWord" -or $Type -eq "QWord") {
   756→   756→                # Numeric comparison (handle both int and string inputs)
   757→   757→                $valueMatches = ([int]$currentValue -eq [int]$Value)
   758→   758→            } elseif ($Type -eq "String" -or $Type -eq "ExpandString") {
   759→   759→                $valueMatches = ([string]$currentValue -eq [string]$Value)
   760→   760→            } else {
   761→   761→                $valueMatches = ($currentValue -eq $Value)
   762→   762→            }
   763→   763→        }
   764→   764→
   765→   765→        if ($valueMatches) {
   766→   766→            Write-Step "Already set: $Path\$Name = $Value"
   767→   767→            $script:TotalSkipped++
   768→   768→            return
   769→   769→        }
   770→   770→
   771→   771→        # Capture old value for undo script before writing
   772→   772→        if ($null -ne $currentValue) {
   773→   773→            $oldVal = $currentValue.ToString()
   774→   774→            $null = $script:UndoCommands.Add("Set-ItemProperty -Path '$Path' -Name '$Name' -Value '$oldVal' -Type '$Type' -Force")
   775→   775→        } else {
   776→   776→            $null = $script:UndoCommands.Add("Remove-ItemProperty -Path '$Path' -Name '$Name' -Force -ErrorAction SilentlyContinue")
   777→   777→        }
   778→   778→
   779→   779→        Set-ItemProperty -Path $Path -Name $Name -Value $Value -Type $Type -Force -ErrorAction Stop
   780→   780→        Write-Success "Registry: $Path\$Name = $Value"
   781→   781→        $script:TotalChanged++
   782→   782→    }
   783→   783→    catch {
   784→   784→        Write-Err "Failed to set $Path\$Name : $_"
   785→   785→        $script:TotalFailed++
   786→   786→    }
   787→   787→}
