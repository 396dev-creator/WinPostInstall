   157→function Set-DefenderExclusions {
   158→    <#
   159→    .DESCRIPTION
   160→        Adds this script and its working directory to Windows Defender exclusions.
   161→        Prevents the script from being flagged as malware during execution.
   162→        This is safe: exclusions are path-specific and only protect this tool.
   163→    #>
   164→    Write-Host "" -ForegroundColor Cyan
   165→    Write-Host "  [AV SHIELD] Windows Defender Self-Exclusion" -ForegroundColor Cyan
   166→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   167→    Write-Host ""
   168→
   169→    $scriptPath = $PSCommandPath
   170→    if (-not $scriptPath) { $scriptPath = $MyInvocation.PSCommandPath }
   171→
   172→    $exclusionAdded = $false
   173→
   174→    # Check if the exclusion already exists
   175→    try {
   176→        $existingExclusions = (Get-MpPreference -ErrorAction SilentlyContinue).ExclusionPath
   177→        if ($existingExclusions -and $scriptPath -in $existingExclusions) {
   178→            Write-Step "Defender exclusion already exists for this script."
   179→            return
   180→        }
   181→    } catch {
   182→        Write-Warn "Could not read current Defender preferences: $_"
   183→    }
   184→
   185→    # Exclude the script file itself
   186→    Write-Step "Adding script to Defender exclusions..."
   187→    try {
   188→        Add-MpPreference -ExclusionPath $scriptPath -ErrorAction Stop
   189→        Write-Success "Excluded: $scriptPath"
   190→        $exclusionAdded = $true
   191→    } catch {
   192→        Write-Warn "Failed to exclude script path: $_"
   193→    }
   194→
   195→    # Exclude the working directory (where the script lives)
   196→    $scriptDir = Split-Path $scriptPath -Parent
   197→    if ($scriptDir -and (Test-Path $scriptDir)) {
   198→        try {
   199→            Add-MpPreference -ExclusionPath $scriptDir -ErrorAction Stop
   200→            Write-Success "Excluded directory: $scriptDir"
   201→            $exclusionAdded = $true
   202→        } catch {
   203→            Write-Warn "Failed to exclude directory: $_"
   204→        }
   205→    }
   206→
   207→    # Exclude the log directory (C:\Temp)
   208→    $logDir = "$env:SYSTEMDRIVE\Temp"
   209→    try {
   210→        Add-MpPreference -ExclusionPath $logDir -ErrorAction Stop
   211→        Write-Success "Excluded log directory: $logDir"
   212→        $exclusionAdded = $true
   213→    } catch {
   214→        Write-Warn "Failed to exclude log directory: $_"
   215→    }
   216→
   217→    # Exclude common technician tool folders
   218→    $toolPaths = @("C:\Tools", "C:\Scripts", "C:\TechnicianTools")
   219→    foreach ($path in $toolPaths) {
   220→        if (Test-Path $path) {
   221→            try {
   222→                Add-MpPreference -ExclusionPath $path -ErrorAction Stop
   223→                Write-Success "Excluded: $path"
   224→            } catch { }
   225→        }
   226→    }
   227→
   228→    if ($exclusionAdded) {
   229→        Write-Success "Defender exclusions configured. This script is now trusted."
   230→    }
   231→
   232→    Write-Host ""
   233→    Write-Host "  NOTE: This only affects Windows Defender. For third-party AV," -ForegroundColor DarkGray
   234→    Write-Host "  consider purchasing a Code Signing Certificate for full compatibility." -ForegroundColor DarkGray
   235→    Write-Host ""
   236→}

   238→function Initialize-Script {
   239→    if (-not ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
   240→        Write-Host ""
   241→        Write-Host "================================================================" -ForegroundColor Red
   242→        Write-Host "  [ERROR] This script must be run as Administrator." -ForegroundColor Red
   243→        Write-Host "================================================================" -ForegroundColor Red
   244→        Write-Host ""
   245→        Write-Host "  Right-click PowerShell -> 'Run as Administrator'" -ForegroundColor Yellow
   246→        Write-Host "  Then re-run this script." -ForegroundColor Yellow
   247→        Write-Host ""
   248→        Write-Host "  Press Enter to exit..." -ForegroundColor DarkGray
   249→        Read-Host "  "
   250→        exit 1
   251→    }
   252→
   253→    $logDir = "$env:SYSTEMDRIVE\Temp"
   254→    if (-not (Test-Path $logDir)) { New-Item -Path $logDir -ItemType Directory -Force | Out-Null }
   255→
   256→    try { Start-Transcript -Path $LogFile -Force -ErrorAction Stop | Out-Null } catch {
   257→        Write-Warn "Could not start transcript to $LogFile : $_"
   258→    }
   259→
   260→    # Set up Defender self-exclusion FIRST (before any optimization work)
   261→    Set-DefenderExclusions
   262→
   263→    # Registry backup - export critical hives before making any changes
   264→    Write-Host "" -ForegroundColor Cyan
   265→    Write-Host "  [SAFETY] Registry Backup" -ForegroundColor Cyan
   266→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   267→    Write-Host ""
   268→    try {
   269→        if (-not (Test-Path $script:BackupDir)) {
   270→            New-Item -Path $script:BackupDir -ItemType Directory -Force | Out-Null
   271→        }
   272→        Write-Step "Exporting registry hives to $script:BackupDir..."
   273→        $regExports = @(
   274→            @{ Hive = "HKLM\SOFTWARE\Policies"; File = "Policies.reg" },
   275→            @{ Hive = "HKLM\SOFTWARE\Microsoft\Windows\CurrentVersion\Policies"; File = "WinPolicies.reg" },
   276→            @{ Hive = "HKCU\SOFTWARE\Microsoft\Windows\CurrentVersion\Explorer"; File = "Explorer.reg" },
   277→            @{ Hive = "HKCU\SOFTWARE\Microsoft\Windows\CurrentVersion\GameDVR"; File = "GameDVR.reg" }
   278→        )
   279→        foreach ($regExport in $regExports) {
   280→            $exportPath = Join-Path $script:BackupDir $regExport.File
   281→            try {
   282→                reg export $regExport.Hive $exportPath /y 2>$null | Out-Null
   283→                Write-Success "Exported: $($regExport.Hive)"
   284→            } catch {
   285→                Write-Warn "Could not export $($regExport.Hive): $_"
   286→            }
   287→        }
   288→        Write-Success "Registry backup saved to: $script:BackupDir"
   289→    } catch {
   290→        Write-Warn "Registry backup failed: $_"
   291→    }
   292→    Write-Host ""
   293→
   294→    $osInfo = Get-CimInstance -ClassName Win32_OperatingSystem
   295→    $script:OSVersion = [version]$osInfo.Version
   296→    $script:OSBuild = $osInfo.BuildNumber
   297→    $script:IsWin11 = $script:OSBuild -ge 22000
   298→    $script:OSCaption = if ($script:IsWin11) { "Windows 11" } else { "Windows 10" }
   299→    $script:ComputerName = $env:COMPUTERNAME
   300→
   301→    # Laptop vs Desktop detection
   302→    $script:IsLaptop = $false
   303→    try {
   304→        $chassisTypes = Get-CimInstance -ClassName Win32_ChassisType -ErrorAction SilentlyContinue
   305→        if ($chassisTypes) {
   306→            $script:IsLaptop = ($chassisTypes | Where-Object { $_.ChassisType -in @(9, 10, 14) }).Count -gt 0
   307→        }
   308→    } catch { }
   309→    if (-not $script:IsLaptop) {
   310→        try {
   311→            $script:IsLaptop = (Get-CimInstance -ClassName Win32_Battery -ErrorAction SilentlyContinue).Count -gt 0
   312→        } catch { }
   313→    }
   314→
   315→    Write-Host ""
   316→    Write-Host "================================================================" -ForegroundColor Cyan
   317→    Write-Host "  Windows Post-Installation Optimizer v$ScriptVersion" -ForegroundColor Cyan
   318→    Write-Host "  Target: $script:OSCaption (Build $script:OSBuild)" -ForegroundColor Cyan
   319→    Write-Host "  Computer: $script:ComputerName" -ForegroundColor Cyan
   320→    Write-Host "  Chassis: $(if ($script:IsLaptop) { 'Laptop' } else { 'Desktop' })" -ForegroundColor Cyan
   321→    Write-Host "  Log file: $LogFile" -ForegroundColor DarkGray
   322→    Write-Host "================================================================" -ForegroundColor Cyan
   323→    Write-Host ""
   324→
   325→    # Run pre-flight checks
   326→    Test-Prerequisites
   327→}

   329→function Test-Prerequisites {
   330→    <#
   331→    .DESCRIPTION
   332→        Pre-flight checks: internet connectivity, disk space, pending reboot, battery.
   333→        Warns on issues but does not block execution.
   334→    #>
   335→    Write-Host "" -ForegroundColor Cyan
   336→    Write-Host "  [PRE-FLIGHT] System Prerequisites Check" -ForegroundColor Cyan
   337→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   338→    Write-Host ""
   339→
   340→    $checksPassed = 0; $checksWarned = 0
   341→
   342→    # 1. Internet connectivity
   343→    Write-Step "Checking internet connectivity..."
   344→    try {
   345→        $netTest = Test-Connection -ComputerName 8.8.8.8 -Count 1 -Quiet -ErrorAction Stop
   346→        if ($netTest) {
   347→            Write-Success "Internet connectivity: OK"
   348→            $checksPassed++
   349→        } else {
   350→            Write-Warn "Internet connectivity: OFFLINE (some features may fail)"
   351→            $checksWarned++
   352→        }
   353→    } catch {
   354→        Write-Warn "Internet connectivity: Could not check (some features may fail)"
   355→        $checksWarned++
   356→    }
   357→
   358→    # 2. Free disk space
   359→    Write-Step "Checking free disk space..."
   360→    try {
   361→        $sysDrive = Get-PSDrive -Name ($env:SYSTEMDRIVE -replace ':','') -ErrorAction Stop
   362→        $freeGB = [math]::Round($sysDrive.Free / 1GB, 2)
   363→        if ($freeGB -gt 10) {
   364→            Write-Success "Free disk space: $freeGB GB (OK)"
   365→            $checksPassed++
   366→        } else {
   367→            Write-Warn "Free disk space: $freeGB GB (LOW - recommend >10 GB)"
   368→            $checksWarned++
   369→        }
   370→    } catch {
   371→        Write-Warn "Could not check disk space: $_"
   372→        $checksWarned++
   373→    }
   374→
   375→    # 3. Pending reboot
   376→    Write-Step "Checking for pending reboot..."
   377→    $rebootPending = $false
   378→    $cbsPath = "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Component Based Servicing\RebootPending"
   379→    $wuPath = "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\WindowsUpdate\Auto Update\RebootRequired"
   380→    if (Test-Path $cbsPath -ErrorAction SilentlyContinue) { $rebootPending = $true }
   381→    if (Test-Path $wuPath -ErrorAction SilentlyContinue) { $rebootPending = $true }
   382→    if ($rebootPending) {
   383→        Write-Warn "Pending reboot detected! A restart is recommended before running optimizations."
   384→        $checksWarned++
   385→    } else {
   386→        Write-Success "No pending reboot."
   387→        $checksPassed++
   388→    }
   389→
   390→    # 4. Battery check (if laptop)
   391→    if ($script:IsLaptop) {
   392→        Write-Step "Checking battery level..."
   393→        try {
   394→            $battery = Get-CimInstance -ClassName Win32_Battery -ErrorAction Stop
   395→            $onAC = $battery.BatteryStatus -eq 2 -or $battery.BatteryStatus -eq 6 -or $battery.BatteryStatus -eq 7 -or $battery.BatteryStatus -eq 8
   396→            if ($onAC) {
   397→                Write-Success "On AC power."
   398→                $checksPassed++
   399→            } elseif ($battery.EstimatedChargeRemaining -gt 50) {
   400→                Write-Success "Battery level: $($battery.EstimatedChargeRemaining)% (OK)"
   401→                $checksPassed++
   402→            } else {
   403→                Write-Warn "Battery level: $($battery.EstimatedChargeRemaining)% - recommend AC power or >50% charge."
   404→                $checksWarned++
   405→            }
   406→        } catch {
   407→            Write-Warn "Could not check battery: $_"
   408→            $checksWarned++
   409→        }
   410→    }
   411→
   412→    Write-Host ""
   413→    Write-Host "  Pre-flight summary: $checksPassed passed, $checksWarned warnings" -ForegroundColor $(if ($checksWarned -gt 0) { 'Yellow' } else { 'Green' })
   414→    Write-Host ""
   415→}

   417→function Get-BootDriveType {
   418→    <#
   419→    .DESCRIPTION
   420→        Detects whether the boot drive is an SSD or HDD.
   421→        Returns "SSD" or "HDD".
   422→    #>
   423→    $bootType = "HDD"  # Default assumption
   424→    try {
   425→        # Method 1: Get-PhysicalDisk (most reliable on Win8+)
   426→        $physicalDisk = Get-PhysicalDisk -ErrorAction SilentlyContinue | Where-Object DeviceId -eq 0
   427→        if ($physicalDisk) {
   428→            if ($physicalDisk.MediaType -eq "SSD") { return "SSD" }
   429→            if ($physicalDisk.MediaType -eq "HDD") { return "HDD" }
   430→        }
   431→    } catch { }
   432→
   433→    try {
   434→        # Method 2: Get-Disk BusType fallback
   435→        $bootDisk = Get-Disk -ErrorAction SilentlyContinue | Where-Object IsBoot -eq $true
   436→        if ($bootDisk) {
   437→            $ssdBusTypes = @("NVMe", "SATA", "RAID", "SCSI")
   438→            # Check via rotation rate (0 = SSD on some controllers)
   439→            if ($bootDisk.BusType -in $ssdBusTypes) {
   440→                # Not conclusive, but check rotation rate
   441→                $detail = Get-PhysicalDisk -ErrorAction SilentlyContinue | Where-Object DeviceId -eq $bootDisk.Number
   442→                if ($detail -and $detail.MediaType -eq "SSD") { return "SSD" }
   443→            }
   444→        }
   445→    } catch { }
   446→
   447→    try {
   448→        # Method 3: Check if defrag is scheduled (SSD = not scheduled)
   449→        $defragTask = Get-ScheduledTask -TaskName "ScheduledDefrag" -ErrorAction SilentlyContinue
   450→        if ($defragTask -and $defragTask.State -eq 'Disabled') {
   451→            # If scheduled defrag is disabled, likely SSD
   452→            # This is a weak heuristic, don't rely on it alone
   453→        }
   454→    } catch { }
   455→
   456→    return $bootType
   457→}

   475→function Enable-SystemRestore {
   476→    <#
   477→    .DESCRIPTION
   478→        Ensures System Restore is enabled and configured on the system drive.
   479→        Enables the service if disabled, allocates disk space, and verifies
   480→        the feature is operational before returning.
   481→    #>
   482→    Write-Host "" -ForegroundColor Cyan
   483→    Write-Host "  [SAFETY] System Restore Verification" -ForegroundColor Cyan
   484→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   485→    Write-Host ""
   486→
   487→    $systemDrive = $env:SYSTEMDRIVE
   488→    $restoreEnabled = $false
   489→
   490→    # Step 1: Check if System Restore is currently enabled on the system drive
   491→    Write-Step "Checking System Restore status on $systemDrive..."
   492→    try {
   493→        $srStatus = Get-ComputerInfo -Property "SystemRestoreStatus" -ErrorAction SilentlyContinue
   494→        # Also check via WMI which is more reliable
   495→        $srInfo = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction SilentlyContinue
   496→        if ($srInfo) {
   497→            foreach ($sr in $srInfo) {
   498→                if ($sr.RPSequence -gt 0) {
   499→                    $restoreEnabled = $true
   500→                    break
   501→                }
   502→            }
   503→        }
   504→    } catch {
   505→        # WMI query may fail if System Restore is completely disabled
   506→        $restoreEnabled = $false
   507→    }
   508→
   509→    # Alternative check: try to list existing restore points
   510→    if (-not $restoreEnabled) {
   511→        try {
   512→            $existingPoints = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction Stop
   513→            if ($existingPoints -and $existingPoints.Count -ge 0) {
   514→                $restoreEnabled = $true
   515→            }
   516→        } catch {
   517→            $restoreEnabled = $false
   518→        }
   519→    }
   520→
   521→    # Step 2: Enable System Restore if not currently enabled
   522→    if (-not $restoreEnabled) {
   523→        Write-Warn "System Restore is NOT enabled on $systemDrive."
   524→        Write-Step "Enabling System Restore on $systemDrive..."
   525→
   526→        try {
   527→            # Enable System Restore via registry (most reliable method)
   528→            $srPolicyPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore"
   529→            if (-not (Test-Path $srPolicyPath)) {
   530→                New-Item -Path $srPolicyPath -Force -ErrorAction SilentlyContinue | Out-Null
   531→            }
   532→            # DisableSR = 0 means System Restore is ENABLED
   533→            Set-ItemProperty -Path $srPolicyPath -Name "DisableSR" -Value 0 -Type DWord -Force -ErrorAction Stop
   534→
   535→            # Also enable via PowerShell cmdlet (Windows 10/11)
   536→            try {
   537→                Enable-ComputerRestore -Drive $systemDrive -ErrorAction Stop
   538→            } catch {
   539→                # Fallback: use vssadmin to enable shadow copies
   540→                $vssResult = vssadmin list shadows 2>&1
   541→                if ($LASTEXITCODE -ne 0) {
   542→                    # Try creating a shadow copy to force-enable VSS
   543→                    vssadmin create shadow /For=$systemDrive 2>&1 | Out-Null
   544→                }
   545→            }
   546→
   547→            # Start the System Restore service if not running
   548→            $srService = Get-Service -Name "sr" -ErrorAction SilentlyContinue
   549→            if ($srService -and $srService.Status -ne 'Running') {
   550→                try {
   551→                    Set-Service -Name "sr" -StartupType Manual -ErrorAction Stop
   552→                    Start-Service -Name "sr" -ErrorAction Stop
   553→                } catch {
   554→                    Write-Warn "Could not start System Restore service: $_"
   555→                }
   556→            }
   557→
   558→            # Also ensure VSS (Volume Shadow Copy) service is running (required for System Restore)
   559→            $vssService = Get-Service -Name "VSS" -ErrorAction SilentlyContinue
   560→            if ($vssService -and $vssService.Status -ne 'Running') {
   561→                try {
   562→                    Set-Service -Name "VSS" -StartupType Manual -ErrorAction Stop
   563→                    Start-Service -Name "VSS" -ErrorAction Stop
   564→                    Write-Success "VSS service started."
   565→                } catch {
   566→                    Write-Warn "Could not start VSS service: $_"
   567→                }
   568→            }
   569→
   570→            Start-Sleep -Seconds 2
   571→            Write-Success "System Restore has been ENABLED on $systemDrive."
   572→        } catch {
   573→            Write-Err "Failed to enable System Restore: $_"
   574→            Write-Warn "Continuing without restore point. Manual enable: System Properties > System Protection > Configure > Turn on"
   575→        }
   576→    } else {
   577→        Write-Success "System Restore is already enabled on $systemDrive."
   578→    }
   579→
   580→    # Step 3: Configure disk space allocation for restore points
   581→    Write-Step "Configuring System Restore disk space..."
   582→    try {
   583→        # Set maximum disk space usage to 5% (reasonable for most systems)
   584→        # This is done via registry: RPSessionInterval and DiskPercent
   585→        $srCfgPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore\Cfg"
   586→        if (-not (Test-Path $srCfgPath)) {
   587→            New-Item -Path $srCfgPath -Force -ErrorAction SilentlyContinue | Out-Null
   588→        }
   589→        # DiskPercent = 5 means 5% of disk (0 = unlimited, we use 5% as safe default)
   590→        Set-ItemProperty -Path $srCfgPath -Name "DiskPercent" -Value 5 -Type DWord -Force -ErrorAction SilentlyContinue
   591→        # RPSessionInterval = 0 means create restore points at any interval
   592→        Set-ItemProperty -Path $srCfgPath -Name "RPSessionInterval" -Value 0 -Type DWord -Force -ErrorAction SilentlyContinue
   593→        Write-Success "Restore disk space set to 5% of drive."
   594→    } catch {
   595→        Write-Warn "Could not configure restore disk space: $_"
   596→    }
   597→
   598→    Write-Host ""
   599→}

   601→function New-SystemRestorePoint {
   602→    <#
   603→    .DESCRIPTION
   604→        Creates a System Restore point before any optimization changes are made.
   605→        This provides a safety net so the user can roll back all changes.
   606→    #>
   607→    Write-Host "" -ForegroundColor Cyan
   608→    Write-Host "  [SAFETY] Creating System Restore Point" -ForegroundColor Cyan
   609→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   610→    Write-Host ""
   611→
   612→    $restorePointCreated = $false
   613→    $description = "Pre-Optimization Restore Point - WinPostInstall v$ScriptVersion - $(Get-Date -Format 'yyyy-MM-dd HH:mm')"
   614→
   615→    # Method 1: Use Checkpoint-Computer (PowerShell cmdlet, Windows 8+)
   616→    Write-Step "Creating restore point: 'Pre-Optimization'..."
   617→    try {
   618→        Checkpoint-Computer -Description $description -RestorePointType "MODIFY_SETTINGS" -ErrorAction Stop
   619→        $restorePointCreated = $true
   620→        Write-Success "Restore point created successfully via Checkpoint-Computer."
   621→    } catch {
   622→        Write-Warn "Checkpoint-Computer failed: $_"
   623→    }
   624→
   625→    # Method 2: Fallback using WMI if Method 1 failed
   626→    if (-not $restorePointCreated) {
   627→        Write-Step "Trying WMI method to create restore point..."
   628→        try {
   629→            $wmiSR = [System.Windows.Forms.SystemRestore]
   630→            # WMI approach via Invoke-CimMethod
   631→            $result = Invoke-CimMethod -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" `
   632→                -MethodName "CreateRestorePoint" `
   633→                -Arguments @{
   634→                    Description         = $description
   635→                    RestorePointType    = 12  # MODIFY_SETTINGS
   636→                    EventType           = 100  # BEGIN_NESTED_SYSTEM_CHANGE
   637→                } -ErrorAction Stop
   638→            if ($result -and $result.ReturnValue -eq 0) {
   639→                $restorePointCreated = $true
   640→                Write-Success "Restore point created via WMI."
   641→            } else {
   642→                Write-Warn "WMI returned non-zero: $result"
   643→            }
   644→        } catch {
   645→            Write-Warn "WMI method failed: $_"
   646→        }
   647→    }
   648→
   649→    # Method 3: Final fallback via srclient.dll COM automation
   650→    if (-not $restorePointCreated) {
   651→        Write-Step "Trying COM automation (srclient.dll)..."
   652→        try {
   653→            $srClient = New-Object -ComObject "SRClient.RestorePoint" -ErrorAction Stop
   654→            # 7 = MODIFY_SETTINGS, 100 = BEGIN_NESTED_SYSTEM_CHANGE
   655→            $srClient.CreateRestorePoint($description, 7, 100)
   656→            $restorePointCreated = $true
   657→            Write-Success "Restore point created via COM automation."
   658→        } catch {
   659→            Write-Warn "COM method failed: $_"
   660→        }
   661→    }
   662→
   663→    if ($restorePointCreated) {
   664→        Write-Host ""
   665→        Write-Host "  ====================================================================" -ForegroundColor Green
   666→        Write-Host "  RESTORE POINT CREATED: 'Pre-Optimization'" -ForegroundColor Green
   667→        Write-Host "  You can roll back all changes via:" -ForegroundColor White
   668→        Write-Host "    System Properties > System Protection > System Restore" -ForegroundColor White
   669→        Write-Host "  ====================================================================" -ForegroundColor Green
   670→        Write-Host ""
   671→    } else {
   672→        Write-Host ""
   673→        Write-Warn "Could NOT create restore point automatically."
   674→        Write-Host "  Manual option: System Properties > System Protection > Create" -ForegroundColor Yellow
   675→        Write-Host "  Continuing without restore point backup." -ForegroundColor Yellow
   676→        Write-Host ""
   677→        if (-not $Silent) {
   678→            $continue = Read-Host "  Continue without restore point? (y/N)"
   679→            if ($continue -ne 'y' -and $continue -ne 'Y') {
   680→                Write-Host "  Aborted by user." -ForegroundColor Yellow
   681→                exit 0
   682→            }
   683→        }
   684→    }
   685→}

   687→function Set-Registry {
   688→    [CmdletBinding()]
   689→    param(
   690→        [string]$Path,
   691→        [string]$Name,
   692→        $Value,
   693→        [string]$Type = "DWord",
   694→        [ValidateSet("Safe","Moderate","Risky")]
   695→        [string]$SafetyLevel = "Safe",
   696→        [string]$Description = "",
   697→        [string]$SettingID = ""
   698→    )
   699→
   700→    # Generate a setting ID if not provided
   701→    if ([string]::IsNullOrEmpty($SettingID)) {
   702→        $SettingID = "$($Path.Replace('\','_').Replace(':','').Replace(' ',''))_$Name"
   703→    }
   704→
   705→    # Register in setting catalog for interactive menu
   706→    if (-not [string]::IsNullOrEmpty($Description)) {
   707→        $script:SettingCatalog[$SettingID] = @{
   708→            Section  = $script:CurrentSection
   709→            Safety   = $SafetyLevel
   710→            Desc     = $Description
   711→            Action   = "$Path\$Name = $Value"
   712→            Type     = "Registry"
   713→        }
   714→    }
   715→
   716→    # Collect risky descriptions for menu warnings
   717→    if ($SafetyLevel -eq "Risky" -and -not [string]::IsNullOrEmpty($Description)) {
   718→        $script:RiskyDescriptions += "$SettingID : $Description"
   719→    }
   720→
   721→    # Check if user has toggled this setting OFF via interactive menu
   722→    if ($script:SettingOverrides.ContainsKey($SettingID) -and $script:SettingOverrides[$SettingID]) {
   723→        Write-Host "  SKIPPED (user toggle): $Description" -ForegroundColor DarkGray
   724→        $script:TotalSkipped++
   725→        return
   726→    }
   727→
   728→    # Safety gate: Skip Risky settings unless explicitly allowed
   729→    if ($SafetyLevel -eq "Risky" -and -not $script:AllowRisky) {
   730→        Write-Host "  SKIPPED (risky): $Description — use -AllowRisky or enable in menu" -ForegroundColor Yellow
   731→        $script:TotalSkipped++
   732→        return
   733→    }
   734→
   735→    # Warn for Moderate settings
   736→    if ($SafetyLevel -eq "Moderate") {
   737→        Write-Host "  [MODERATE] $Description" -ForegroundColor DarkYellow
   738→    }
   739→
   740→    try {
   741→        if (-not (Test-Path $Path)) {
   742→            New-Item -Path $Path -Force -ErrorAction SilentlyContinue | Out-Null
   743→        }
   744→        # Idempotency: Check if the value already matches what we want to set
   745→        $currentValue = $null
   746→        try {
   747→            $currentValue = (Get-ItemProperty -Path $Path -Name $Name -ErrorAction Stop).$Name
   748→        } catch {
   749→            # Property doesn't exist yet — need to create it
   750→        }
   751→
   752→        # Compare current vs desired value
   753→        $valueMatches = $false
   754→        if ($null -ne $currentValue) {
   755→            if ($Type -eq "DWord" -or $Type -eq "QWord") {
   756→                # Numeric comparison (handle both int and string inputs)
   757→                $valueMatches = ([int]$currentValue -eq [int]$Value)
   758→            } elseif ($Type -eq "String" -or $Type -eq "ExpandString") {
   759→                $valueMatches = ([string]$currentValue -eq [string]$Value)
   760→            } else {
   761→                $valueMatches = ($currentValue -eq $Value)
   762→            }
   763→        }
   764→
   765→        if ($valueMatches) {
   766→            Write-Step "Already set: $Path\$Name = $Value"
   767→            $script:TotalSkipped++
   768→            return
   769→        }
   770→
   771→        # Capture old value for undo script before writing
   772→        if ($null -ne $currentValue) {
   773→            $oldVal = $currentValue.ToString()
   774→            $null = $script:UndoCommands.Add("Set-ItemProperty -Path '$Path' -Name '$Name' -Value '$oldVal' -Type '$Type' -Force")
   775→        } else {
   776→            $null = $script:UndoCommands.Add("Remove-ItemProperty -Path '$Path' -Name '$Name' -Force -ErrorAction SilentlyContinue")
   777→        }
   778→
   779→        Set-ItemProperty -Path $Path -Name $Name -Value $Value -Type $Type -Force -ErrorAction Stop
   780→        Write-Success "Registry: $Path\$Name = $Value"
   781→        $script:TotalChanged++
   782→    }
   783→    catch {
   784→        Write-Err "Failed to set $Path\$Name : $_"
   785→        $script:TotalFailed++
   786→    }
   787→}

   789→function Remove-Registry {
   790→    [CmdletBinding()]
   791→    param(
   792→        [string]$Path,
   793→        [string]$Name
   794→    )
   795→    try {
   796→        if (Test-Path $Path) {

   799→            try { $existing = (Get-ItemProperty -Path $Path -Name $Name -ErrorAction Stop).$Name } catch { }
   800→            if ($null -ne $existing) {
   801→                Remove-ItemProperty -Path $Path -Name $Name -Force -ErrorAction Stop 2>$null
   802→                Write-Success "Registry removed: $Path\$Name"

   804→                Write-Step "Already removed: $Path\$Name"
   805→            }
   806→        }
   807→    }

   809→        Write-Warn "Could not remove $Path\$Name : $_"
   810→    }

   813→function Disable-ScheduledTask {
   814→    [CmdletBinding()]
   815→    param(
   816→        [string]$TaskPath,
   817→        [string]$TaskName,
   818→        [ValidateSet("Safe","Moderate","Risky")]
   819→        [string]$SafetyLevel = "Safe",
   820→        [string]$Description = "",
   821→        [string]$SettingID = ""
   822→    )
   823→
   824→    # Generate a setting ID if not provided
   825→    if ([string]::IsNullOrEmpty($SettingID)) {
   826→        $SettingID = "Task_$($TaskPath.Replace('\','_').Replace(' ',''))_$TaskName"
   827→    }
   828→
   829→    # Register in setting catalog for interactive menu
   830→    if (-not [string]::IsNullOrEmpty($Description)) {
   831→        $script:SettingCatalog[$SettingID] = @{
   832→            Section  = $script:CurrentSection
   833→            Safety   = $SafetyLevel
   834→            Desc     = $Description
   835→            Action   = "Disable task: $TaskPath$TaskName"
   836→            Type     = "ScheduledTask"
   837→        }
   838→    }
   839→
   840→    # Collect risky descriptions for menu warnings
   841→    if ($SafetyLevel -eq "Risky" -and -not [string]::IsNullOrEmpty($Description)) {
   842→        $script:RiskyDescriptions += "$SettingID : $Description"
   843→    }
   844→
   845→    # Check if user has toggled this setting OFF via interactive menu
   846→    if ($script:SettingOverrides.ContainsKey($SettingID) -and $script:SettingOverrides[$SettingID]) {
   847→        Write-Host "  SKIPPED (user toggle): $Description" -ForegroundColor DarkGray
   848→        $script:TotalSkipped++
   849→        return
   850→    }
   851→
   852→    # Safety gate: Skip Risky settings unless explicitly allowed
   853→    if ($SafetyLevel -eq "Risky" -and -not $script:AllowRisky) {
   854→        Write-Host "  SKIPPED (risky): $Description — use -AllowRisky or enable in menu" -ForegroundColor Yellow
   855→        $script:TotalSkipped++
   856→        return
   857→    }
   858→
   859→    # Warn for Moderate settings
   860→    if ($SafetyLevel -eq "Moderate") {
   861→        Write-Host "  [MODERATE] $Description" -ForegroundColor DarkYellow
   862→    }
   863→
   864→    try {
   865→        $task = Get-ScheduledTask -TaskPath $TaskPath -TaskName $TaskName -ErrorAction SilentlyContinue
   866→        if ($task) {
   867→            if ($task.State -eq 'Disabled') {
   868→                Write-Step "Already disabled: $TaskPath$TaskName"
   869→                $script:TotalSkipped++
   870→            } else {
   871→                # Safety: Check for dependent services/tasks before disabling
   872→                $deps = $task.DependentOn
   873→                if ($deps) {
   874→                    Write-Host "  [DEPENDENCY] Task $TaskName has dependents: $($deps.TaskName -join ', ')" -ForegroundColor DarkYellow
   875→                }
   876→                Disable-ScheduledTask -TaskPath $TaskPath -TaskName $TaskName -ErrorAction Stop | Out-Null
   877→                Write-Success "Task disabled: $TaskPath$TaskName"
   878→                $script:TotalChanged++
   879→                $null = $script:UndoCommands.Add("Enable-ScheduledTask -TaskPath '$TaskPath' -TaskName '$TaskName' -ErrorAction SilentlyContinue")
   880→            }
   881→        }
   882→    }
   883→    catch {
   884→        Write-Warn "Could not disable task $TaskPath$TaskName : $_"
   885→    }
   886→}
