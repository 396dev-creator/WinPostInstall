   200→function Set-DefenderExclusions {
   201→    <#
   202→    .DESCRIPTION
   203→        Adds this script and its working directory to Windows Defender exclusions.
   204→        Prevents the script from being flagged as malware during execution.
   205→        This is safe: exclusions are path-specific and only protect this tool.
   206→    #>
   207→    Write-Host "" -ForegroundColor Cyan
   208→    Write-Host "  [AV SHIELD] Windows Defender Self-Exclusion" -ForegroundColor Cyan
   209→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   210→    Write-Host ""
   211→
   212→    $scriptPath = $PSCommandPath
   213→    if (-not $scriptPath) { $scriptPath = $MyInvocation.PSCommandPath }
   214→
   215→    $exclusionAdded = $false
   216→
   217→    # Check if the exclusion already exists
   218→    try {
   219→        $existingExclusions = (Get-MpPreference -ErrorAction SilentlyContinue).ExclusionPath
   220→        if ($existingExclusions -and $scriptPath -in $existingExclusions) {
   221→            Write-Step "Defender exclusion already exists for this script."
   222→            return
   223→        }
   224→    } catch {
   225→        Write-Warn "Could not read current Defender preferences: $_"
   226→    }
   227→
   228→    # Exclude the script file itself
   229→    Write-Step "Adding script to Defender exclusions..."
   230→    try {
   231→        Add-MpPreference -ExclusionPath $scriptPath -ErrorAction Stop
   232→        Write-Success "Excluded: $scriptPath"
   233→        $exclusionAdded = $true
   234→    } catch {
   235→        Write-Warn "Failed to exclude script path: $_"
   236→    }
   237→
   238→    # Exclude the working directory (where the script lives)
   239→    $scriptDir = Split-Path $scriptPath -Parent
   240→    if ($scriptDir -and (Test-Path $scriptDir)) {
   241→        try {
   242→            Add-MpPreference -ExclusionPath $scriptDir -ErrorAction Stop
   243→            Write-Success "Excluded directory: $scriptDir"
   244→            $exclusionAdded = $true
   245→        } catch {
   246→            Write-Warn "Failed to exclude directory: $_"
   247→        }
   248→    }
   249→
   250→    # Exclude the log directory (C:\Temp)
   251→    $logDir = "$env:SYSTEMDRIVE\Temp"
   252→    try {
   253→        Add-MpPreference -ExclusionPath $logDir -ErrorAction Stop
   254→        Write-Success "Excluded log directory: $logDir"
   255→        $exclusionAdded = $true
   256→    } catch {
   257→        Write-Warn "Failed to exclude log directory: $_"
   258→    }
   259→
   260→    # Exclude common technician tool folders
   261→    $toolPaths = @("C:\Tools", "C:\Scripts", "C:\TechnicianTools")
   262→    foreach ($path in $toolPaths) {
   263→        if (Test-Path $path) {
   264→            try {
   265→                Add-MpPreference -ExclusionPath $path -ErrorAction Stop
   266→                Write-Success "Excluded: $path"
   267→            } catch { }
   268→        }
   269→    }
   270→
   271→    if ($exclusionAdded) {
   272→        Write-Success "Defender exclusions configured. This script is now trusted."
   273→    }
   274→
   275→    Write-Host ""
   276→    Write-Host "  NOTE: This only affects Windows Defender. For third-party AV," -ForegroundColor DarkGray
   277→    Write-Host "  consider purchasing a Code Signing Certificate for full compatibility." -ForegroundColor DarkGray
   278→    Write-Host ""
   279→}

   372→function Test-Prerequisites {
   373→    <#
   374→    .DESCRIPTION
   375→        Pre-flight checks: internet connectivity, disk space, pending reboot, battery.
   376→        Warns on issues but does not block execution.
   377→    #>
   378→    Write-Host "" -ForegroundColor Cyan
   379→    Write-Host "  [PRE-FLIGHT] System Prerequisites Check" -ForegroundColor Cyan
   380→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   381→    Write-Host ""
   382→
   383→    $checksPassed = 0; $checksWarned = 0
   384→
   385→    # 1. Internet connectivity
   386→    Write-Step "Checking internet connectivity..."
   387→    try {
   388→        $netTest = Test-Connection -ComputerName 8.8.8.8 -Count 1 -Quiet -ErrorAction Stop
   389→        if ($netTest) {
   390→            Write-Success "Internet connectivity: OK"
   391→            $checksPassed++
   392→        } else {
   393→            Write-Warn "Internet connectivity: OFFLINE (some features may fail)"
   394→            $checksWarned++
   395→        }
   396→    } catch {
   397→        Write-Warn "Internet connectivity: Could not check (some features may fail)"
   398→        $checksWarned++
   399→    }
   400→
   401→    # 2. Free disk space
   402→    Write-Step "Checking free disk space..."
   403→    try {
   404→        $sysDrive = Get-PSDrive -Name ($env:SYSTEMDRIVE -replace ':','') -ErrorAction Stop
   405→        $freeGB = [math]::Round($sysDrive.Free / 1GB, 2)
   406→        if ($freeGB -gt 10) {
   407→            Write-Success "Free disk space: $freeGB GB (OK)"
   408→            $checksPassed++
   409→        } else {
   410→            Write-Warn "Free disk space: $freeGB GB (LOW - recommend >10 GB)"
   411→            $checksWarned++
   412→        }
   413→    } catch {
   414→        Write-Warn "Could not check disk space: $_"
   415→        $checksWarned++
   416→    }
   417→
   418→    # 3. Pending reboot
   419→    Write-Step "Checking for pending reboot..."
   420→    $rebootPending = $false
   421→    $cbsPath = "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Component Based Servicing\RebootPending"
   422→    $wuPath = "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\WindowsUpdate\Auto Update\RebootRequired"
   423→    if (Test-Path $cbsPath -ErrorAction SilentlyContinue) { $rebootPending = $true }
   424→    if (Test-Path $wuPath -ErrorAction SilentlyContinue) { $rebootPending = $true }
   425→    if ($rebootPending) {
   426→        Write-Warn "Pending reboot detected! A restart is recommended before running optimizations."
   427→        $checksWarned++
   428→    } else {
   429→        Write-Success "No pending reboot."
   430→        $checksPassed++
   431→    }
   432→
   433→    # 4. Battery check (if laptop)
   434→    if ($script:IsLaptop) {
   435→        Write-Step "Checking battery level..."
   436→        try {
   437→            $battery = Get-CimInstance -ClassName Win32_Battery -ErrorAction Stop
   438→            $onAC = $battery.BatteryStatus -eq 2 -or $battery.BatteryStatus -eq 6 -or $battery.BatteryStatus -eq 7 -or $battery.BatteryStatus -eq 8
   439→            if ($onAC) {
   440→                Write-Success "On AC power."
   441→                $checksPassed++
   442→            } elseif ($battery.EstimatedChargeRemaining -gt 50) {
   443→                Write-Success "Battery level: $($battery.EstimatedChargeRemaining)% (OK)"
   444→                $checksPassed++
   445→            } else {
   446→                Write-Warn "Battery level: $($battery.EstimatedChargeRemaining)% - recommend AC power or >50% charge."
   447→                $checksWarned++
   448→            }
   449→        } catch {
   450→            Write-Warn "Could not check battery: $_"
   451→            $checksWarned++
   452→        }
   453→    }
   454→
   455→    Write-Host ""
   456→    Write-Host "  Pre-flight summary: $checksPassed passed, $checksWarned warnings" -ForegroundColor $(if ($checksWarned -gt 0) { 'Yellow' } else { 'Green' })
   457→    Write-Host ""
   458→}

   460→function Get-BootDriveType {
   461→    <#
   462→    .DESCRIPTION
   463→        Detects whether the boot drive is an SSD or HDD.
   464→        Returns "SSD" or "HDD".
   465→    #>
   466→    $bootType = "HDD"  # Default assumption
   467→    try {
   468→        # Method 1: Get-PhysicalDisk (most reliable on Win8+)
   469→        $physicalDisk = Get-PhysicalDisk -ErrorAction SilentlyContinue | Where-Object DeviceId -eq 0
   470→        if ($physicalDisk) {
   471→            if ($physicalDisk.MediaType -eq "SSD") { return "SSD" }
   472→            if ($physicalDisk.MediaType -eq "HDD") { return "HDD" }
   473→        }
   474→    } catch { }
   475→
   476→    try {
   477→        # Method 2: Get-Disk BusType fallback
   478→        $bootDisk = Get-Disk -ErrorAction SilentlyContinue | Where-Object IsBoot -eq $true
   479→        if ($bootDisk) {
   480→            $ssdBusTypes = @("NVMe", "SATA", "RAID", "SCSI")
   481→            # Check via rotation rate (0 = SSD on some controllers)
   482→            if ($bootDisk.BusType -in $ssdBusTypes) {
   483→                # Not conclusive, but check rotation rate
   484→                $detail = Get-PhysicalDisk -ErrorAction SilentlyContinue | Where-Object DeviceId -eq $bootDisk.Number
   485→                if ($detail -and $detail.MediaType -eq "SSD") { return "SSD" }
   486→            }
   487→        }
   488→    } catch { }
   489→
   490→    try {
   491→        # Method 3: Check if defrag is scheduled (SSD = not scheduled)
   492→        $defragTask = Get-ScheduledTask -TaskName "ScheduledDefrag" -ErrorAction SilentlyContinue
   493→        if ($defragTask -and $defragTask.State -eq 'Disabled') {
   494→            # If scheduled defrag is disabled, likely SSD
   495→            # This is a weak heuristic, don't rely on it alone
   496→        }
   497→    } catch { }
   498→
   499→    return $bootType
   500→}

   518→function Enable-SystemRestore {
   519→    <#
   520→    .DESCRIPTION
   521→        Ensures System Restore is enabled and configured on the system drive.
   522→        Enables the service if disabled, allocates disk space, and verifies
   523→        the feature is operational before returning.
   524→    #>
   525→    Write-Host "" -ForegroundColor Cyan
   526→    Write-Host "  [SAFETY] System Restore Verification" -ForegroundColor Cyan
   527→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   528→    Write-Host ""
   529→
   530→    $systemDrive = $env:SYSTEMDRIVE
   531→    $restoreEnabled = $false
   532→
   533→    # Step 1: Check if System Restore is currently enabled on the system drive
   534→    Write-Step "Checking System Restore status on $systemDrive..."
   535→    try {
   536→        $srStatus = Get-ComputerInfo -Property "SystemRestoreStatus" -ErrorAction SilentlyContinue
   537→        # Also check via WMI which is more reliable
   538→        $srInfo = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction SilentlyContinue
   539→        if ($srInfo) {
   540→            foreach ($sr in $srInfo) {
   541→                if ($sr.RPSequence -gt 0) {
   542→                    $restoreEnabled = $true
   543→                    break
   544→                }
   545→            }
   546→        }
   547→    } catch {
   548→        # WMI query may fail if System Restore is completely disabled
   549→        $restoreEnabled = $false
   550→    }
   551→
   552→    # Alternative check: try to list existing restore points
   553→    if (-not $restoreEnabled) {
   554→        try {
   555→            $existingPoints = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction Stop
   556→            if ($existingPoints -and $existingPoints.Count -ge 0) {
   557→                $restoreEnabled = $true
   558→            }
   559→        } catch {
   560→            $restoreEnabled = $false
   561→        }
   562→    }
   563→
   564→    # Step 2: Enable System Restore if not currently enabled
   565→    if (-not $restoreEnabled) {
   566→        Write-Warn "System Restore is NOT enabled on $systemDrive."
   567→        Write-Step "Enabling System Restore on $systemDrive..."
   568→
   569→        try {
   570→            # Enable System Restore via registry (most reliable method)
   571→            $srPolicyPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore"
   572→            if (-not (Test-Path $srPolicyPath)) {
   573→                New-Item -Path $srPolicyPath -Force -ErrorAction SilentlyContinue | Out-Null
   574→            }
   575→            # DisableSR = 0 means System Restore is ENABLED
   576→            Set-ItemProperty -Path $srPolicyPath -Name "DisableSR" -Value 0 -Type DWord -Force -ErrorAction Stop
   577→
   578→            # Also enable via PowerShell cmdlet (Windows 10/11)
   579→            try {
   580→                Enable-ComputerRestore -Drive $systemDrive -ErrorAction Stop
   581→            } catch {
   582→                # Fallback: use vssadmin to enable shadow copies
   583→                $vssResult = vssadmin list shadows 2>&1
   584→                if ($LASTEXITCODE -ne 0) {
   585→                    # Try creating a shadow copy to force-enable VSS
   586→                    vssadmin create shadow /For=$systemDrive 2>&1 | Out-Null
   587→                }
   588→            }
   589→
   590→            # Start the System Restore service if not running
   591→            $srService = Get-Service -Name "sr" -ErrorAction SilentlyContinue
   592→            if ($srService -and $srService.Status -ne 'Running') {
   593→                try {
   594→                    Set-Service -Name "sr" -StartupType Manual -ErrorAction Stop
   595→                    Start-Service -Name "sr" -ErrorAction Stop
   596→                } catch {
   597→                    Write-Warn "Could not start System Restore service: $_"
   598→                }
   599→            }
   600→
   601→            # Also ensure VSS (Volume Shadow Copy) service is running (required for System Restore)
   602→            $vssService = Get-Service -Name "VSS" -ErrorAction SilentlyContinue
   603→            if ($vssService -and $vssService.Status -ne 'Running') {
   604→                try {
   605→                    Set-Service -Name "VSS" -StartupType Manual -ErrorAction Stop
   606→                    Start-Service -Name "VSS" -ErrorAction Stop
   607→                    Write-Success "VSS service started."
   608→                } catch {
   609→                    Write-Warn "Could not start VSS service: $_"
   610→                }
   611→            }
   612→
   613→            Start-Sleep -Seconds 2
   614→            Write-Success "System Restore has been ENABLED on $systemDrive."
   615→        } catch {
   616→            Write-Err "Failed to enable System Restore: $_"
   617→            Write-Warn "Continuing without restore point. Manual enable: System Properties > System Protection > Configure > Turn on"
   618→        }
   619→    } else {
   620→        Write-Success "System Restore is already enabled on $systemDrive."
   621→    }
   622→
   623→    # Step 3: Configure disk space allocation for restore points
   624→    Write-Step "Configuring System Restore disk space..."
   625→    try {
   626→        # Set maximum disk space usage to 5% (reasonable for most systems)
   627→        # This is done via registry: RPSessionInterval and DiskPercent
   628→        $srCfgPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore\Cfg"
   629→        if (-not (Test-Path $srCfgPath)) {
   630→            New-Item -Path $srCfgPath -Force -ErrorAction SilentlyContinue | Out-Null
   631→        }
   632→        # DiskPercent = 5 means 5% of disk (0 = unlimited, we use 5% as safe default)
   633→        Set-ItemProperty -Path $srCfgPath -Name "DiskPercent" -Value 5 -Type DWord -Force -ErrorAction SilentlyContinue
   634→        # RPSessionInterval = 0 means create restore points at any interval
   635→        Set-ItemProperty -Path $srCfgPath -Name "RPSessionInterval" -Value 0 -Type DWord -Force -ErrorAction SilentlyContinue
   636→        Write-Success "Restore disk space set to 5% of drive."
   637→    } catch {
   638→        Write-Warn "Could not configure restore disk space: $_"
   639→    }
   640→
   641→    Write-Host ""
   642→}

   644→function New-SystemRestorePoint {
   645→    <#
   646→    .DESCRIPTION
   647→        Creates a System Restore point before any optimization changes are made.
   648→        This provides a safety net so the user can roll back all changes.
   649→    #>
   650→    Write-Host "" -ForegroundColor Cyan
   651→    Write-Host "  [SAFETY] Creating System Restore Point" -ForegroundColor Cyan
   652→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   653→    Write-Host ""
   654→
   655→    $restorePointCreated = $false
   656→    $description = "Pre-Optimization Restore Point - WinPostInstall v$ScriptVersion - $(Get-Date -Format 'yyyy-MM-dd HH:mm')"
   657→
   658→    # Method 1: Use Checkpoint-Computer (PowerShell cmdlet, Windows 8+)
   659→    Write-Step "Creating restore point: 'Pre-Optimization'..."
   660→    try {
   661→        Checkpoint-Computer -Description $description -RestorePointType "MODIFY_SETTINGS" -ErrorAction Stop
   662→        $restorePointCreated = $true
   663→        Write-Success "Restore point created successfully via Checkpoint-Computer."
   664→    } catch {
   665→        Write-Warn "Checkpoint-Computer failed: $_"
   666→    }
   667→
   668→    # Method 2: Fallback using WMI if Method 1 failed
   669→    if (-not $restorePointCreated) {
   670→        Write-Step "Trying WMI method to create restore point..."
   671→        try {
   672→            $wmiSR = [System.Windows.Forms.SystemRestore]
   673→            # WMI approach via Invoke-CimMethod
   674→            $result = Invoke-CimMethod -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" `
   675→                -MethodName "CreateRestorePoint" `
   676→                -Arguments @{
   677→                    Description         = $description
   678→                    RestorePointType    = 12  # MODIFY_SETTINGS
   679→                    EventType           = 100  # BEGIN_NESTED_SYSTEM_CHANGE
   680→                } -ErrorAction Stop
   681→            if ($result -and $result.ReturnValue -eq 0) {
   682→                $restorePointCreated = $true
   683→                Write-Success "Restore point created via WMI."
   684→            } else {
   685→                Write-Warn "WMI returned non-zero: $result"
   686→            }
   687→        } catch {
   688→            Write-Warn "WMI method failed: $_"
   689→        }
   690→    }
   691→
   692→    # Method 3: Final fallback via srclient.dll COM automation
   693→    if (-not $restorePointCreated) {
   694→        Write-Step "Trying COM automation (srclient.dll)..."
   695→        try {
   696→            $srClient = New-Object -ComObject "SRClient.RestorePoint" -ErrorAction Stop
   697→            # 7 = MODIFY_SETTINGS, 100 = BEGIN_NESTED_SYSTEM_CHANGE
   698→            $srClient.CreateRestorePoint($description, 7, 100)
   699→            $restorePointCreated = $true
   700→            Write-Success "Restore point created via COM automation."
   701→        } catch {
   702→            Write-Warn "COM method failed: $_"
   703→        }
   704→    }
   705→
   706→    if ($restorePointCreated) {
   707→        Write-Host ""
   708→        Write-Host "  ====================================================================" -ForegroundColor Green
   709→        Write-Host "  RESTORE POINT CREATED: 'Pre-Optimization'" -ForegroundColor Green
   710→        Write-Host "  You can roll back all changes via:" -ForegroundColor White
   711→        Write-Host "    System Properties > System Protection > System Restore" -ForegroundColor White
   712→        Write-Host "  ====================================================================" -ForegroundColor Green
   713→        Write-Host ""
   714→    } else {
   715→        Write-Host ""
   716→        Write-Warn "Could NOT create restore point automatically."
   717→        Write-Host "  Manual option: System Properties > System Protection > Create" -ForegroundColor Yellow
   718→        Write-Host "  Continuing without restore point backup." -ForegroundColor Yellow
   719→        Write-Host ""
   720→        if (-not $Silent) {
   721→            $continue = Read-Host "  Continue without restore point? (y/N)"
   722→            if ($continue -ne 'y' -and $continue -ne 'Y') {
   723→                Write-Host "  Aborted by user." -ForegroundColor Yellow
   724→                exit 0
   725→            }
   726→        }
   727→    }
   728→}

   730→function Set-Registry {
   731→    [CmdletBinding()]
   732→    param(
   733→        [string]$Path,
   734→        [string]$Name,
   735→        $Value,
   736→        [string]$Type = "DWord",
   737→        [ValidateSet("Safe","Moderate","Risky")]
   738→        [string]$SafetyLevel = "Safe",
   739→        [string]$Description = "",
   740→        [string]$SettingID = ""
   741→    )
   742→
   743→    # Generate a setting ID if not provided
   744→    if ([string]::IsNullOrEmpty($SettingID)) {
   745→        $SettingID = "$($Path.Replace('\','_').Replace(':','').Replace(' ',''))_$Name"
   746→    }
   747→
   748→    # Register in setting catalog for interactive menu
   749→    if (-not [string]::IsNullOrEmpty($Description)) {
   750→        $script:SettingCatalog[$SettingID] = @{
   751→            Section  = $script:CurrentSection
   752→            Safety   = $SafetyLevel
   753→            Desc     = $Description
   754→            Action   = "$Path\$Name = $Value"
   755→            Type     = "Registry"
   756→        }
   757→    }
   758→
   759→    # Collect risky descriptions for menu warnings
   760→    if ($SafetyLevel -eq "Risky" -and -not [string]::IsNullOrEmpty($Description)) {
   761→        $script:RiskyDescriptions += "$SettingID : $Description"
   762→    }
   763→
   764→    # Check if user has toggled this setting OFF via interactive menu
   765→    if ($script:SettingOverrides.ContainsKey($SettingID) -and $script:SettingOverrides[$SettingID]) {
   766→        Write-Host "  SKIPPED (user toggle): $Description" -ForegroundColor DarkGray
   767→        $script:TotalSkipped++
   768→        return
   769→    }
   770→
   771→    # Safety gate: Skip Risky settings unless explicitly allowed
   772→    if ($SafetyLevel -eq "Risky" -and -not $script:AllowRisky) {
   773→        Write-Host "  SKIPPED (risky): $Description — use -AllowRisky or enable in menu" -ForegroundColor Yellow
   774→        $script:TotalSkipped++
   775→        return
   776→    }
   777→
   778→    # Warn for Moderate settings
   779→    if ($SafetyLevel -eq "Moderate") {
   780→        Write-Host "  [MODERATE] $Description" -ForegroundColor DarkYellow
   781→    }
   782→
   783→    try {
   784→        if (-not (Test-Path $Path)) {
   785→            New-Item -Path $Path -Force -ErrorAction SilentlyContinue | Out-Null
   786→        }
   787→        # Idempotency: Check if the value already matches what we want to set
   788→        $currentValue = $null
   789→        try {
   790→            $currentValue = (Get-ItemProperty -Path $Path -Name $Name -ErrorAction Stop).$Name
   791→        } catch {
   792→            # Property doesn't exist yet — need to create it
   793→        }
   794→
   795→        # Compare current vs desired value
   796→        $valueMatches = $false
   797→        if ($null -ne $currentValue) {
   798→            if ($Type -eq "DWord" -or $Type -eq "QWord") {
   799→                # Numeric comparison (handle both int and string inputs)
   800→                $valueMatches = ([int]$currentValue -eq [int]$Value)
   801→            } elseif ($Type -eq "String" -or $Type -eq "ExpandString") {
   802→                $valueMatches = ([string]$currentValue -eq [string]$Value)
   803→            } else {
   804→                $valueMatches = ($currentValue -eq $Value)
   805→            }
   806→        }
   807→
   808→        if ($valueMatches) {
   809→            Write-Step "Already set: $Path\$Name = $Value"
   810→            $script:TotalSkipped++
   811→            return
   812→        }
   813→
   814→        # Capture old value for undo script before writing
   815→        if ($null -ne $currentValue) {
   816→            $oldVal = $currentValue.ToString()
   817→            $null = $script:UndoCommands.Add("Set-ItemProperty -Path '$Path' -Name '$Name' -Value '$oldVal' -Type '$Type' -Force")
   818→            $null = $script:UndoDescriptions.Add("Restore registry: $Path\$Name = $oldVal")
   819→        } else {
   820→            $null = $script:UndoCommands.Add("Remove-ItemProperty -Path '$Path' -Name '$Name' -Force -ErrorAction SilentlyContinue")
   821→            $null = $script:UndoDescriptions.Add("Remove registry key that was added: $Path\$Name")
   822→        }
   823→
   824→        Set-ItemProperty -Path $Path -Name $Name -Value $Value -Type $Type -Force -ErrorAction Stop
   825→        Write-Success "Registry: $Path\$Name = $Value"
   826→        $script:TotalChanged++
   827→    }
   828→    catch {
   829→        Write-Err "Failed to set $Path\$Name : $_"
   830→        $script:TotalFailed++
   831→    }
   832→}

   834→function Remove-Registry {
   835→    [CmdletBinding()]
   836→    param(
   837→        [string]$Path,
   838→        [string]$Name
   839→    )
   840→    try {
   841→        if (Test-Path $Path) {
   842→            # Idempotency: Check if property exists before removing
   843→            $existing = $null
   844→            try { $existing = (Get-ItemProperty -Path $Path -Name $Name -ErrorAction Stop).$Name } catch { }
   845→            if ($null -ne $existing) {
   846→                Remove-ItemProperty -Path $Path -Name $Name -Force -ErrorAction Stop 2>$null
   847→                Write-Success "Registry removed: $Path\$Name"
   848→            } else {
   849→                Write-Step "Already removed: $Path\$Name"
   850→            }
   851→        }
   852→    }
   853→    catch {
   854→        Write-Warn "Could not remove $Path\$Name : $_"
   855→    }
   856→}

   858→function Disable-AppTask {
   859→    [CmdletBinding()]
   860→    param(
   861→        [string]$TaskPath,
   862→        [string]$TaskName,
   863→        [ValidateSet("Safe","Moderate","Risky")]
   864→        [string]$SafetyLevel = "Safe",
   865→        [string]$Description = "",
   866→        [string]$SettingID = ""
   867→    )
   868→
   869→    # Generate a setting ID if not provided
   870→    if ([string]::IsNullOrEmpty($SettingID)) {
   871→        $SettingID = "Task_$($TaskPath.Replace('\','_').Replace(' ',''))_$TaskName"
   872→    }
   873→
   874→    # Register in setting catalog for interactive menu
   875→    if (-not [string]::IsNullOrEmpty($Description)) {
   876→        $script:SettingCatalog[$SettingID] = @{
   877→            Section  = $script:CurrentSection
   878→            Safety   = $SafetyLevel
   879→            Desc     = $Description
   880→            Action   = "Disable task: $TaskPath$TaskName"
   881→            Type     = "ScheduledTask"
   882→        }
   883→    }
   884→
   885→    # Collect risky descriptions for menu warnings
   886→    if ($SafetyLevel -eq "Risky" -and -not [string]::IsNullOrEmpty($Description)) {
   887→        $script:RiskyDescriptions += "$SettingID : $Description"
   888→    }
   889→
   890→    # Check if user has toggled this setting OFF via interactive menu
   891→    if ($script:SettingOverrides.ContainsKey($SettingID) -and $script:SettingOverrides[$SettingID]) {
   892→        Write-Host "  SKIPPED (user toggle): $Description" -ForegroundColor DarkGray
   893→        $script:TotalSkipped++
   894→        return
   895→    }
   896→
   897→    # Safety gate: Skip Risky settings unless explicitly allowed
   898→    if ($SafetyLevel -eq "Risky" -and -not $script:AllowRisky) {
   899→        Write-Host "  SKIPPED (risky): $Description — use -AllowRisky or enable in menu" -ForegroundColor Yellow
   900→        $script:TotalSkipped++
   901→        return
   902→    }
   903→
   904→    # Warn for Moderate settings
   905→    if ($SafetyLevel -eq "Moderate") {
   906→        Write-Host "  [MODERATE] $Description" -ForegroundColor DarkYellow
   907→    }
   908→
   909→    try {
   910→        $task = Get-ScheduledTask -TaskPath $TaskPath -TaskName $TaskName -ErrorAction SilentlyContinue
   911→        if ($task) {
   912→            if ($task.State -eq 'Disabled') {
   913→                Write-Step "Already disabled: $TaskPath$TaskName"
   914→                $script:TotalSkipped++
   915→            } else {
   916→                # Safety: Check for dependent services/tasks before disabling
   917→                $deps = $task.DependentOn
   918→                if ($deps) {
   919→                    Write-Host "  [DEPENDENCY] Task $TaskName has dependents: $($deps.TaskName -join ', ')" -ForegroundColor DarkYellow
   920→                }
   921→                $task | Disable-ScheduledTask -ErrorAction Stop
   922→                Write-Success "Task disabled: $TaskPath$TaskName"
   923→                $script:TotalChanged++
   924→                $null = $script:UndoCommands.Add("Enable-ScheduledTask -TaskPath '$TaskPath' -TaskName '$TaskName' -ErrorAction SilentlyContinue")
   925→                $null = $script:UndoDescriptions.Add("Re-enable scheduled task: $TaskPath$TaskName")
   926→            }
   927→        }
   928→    }
   929→    catch {
   930→        Write-Warn "Could not disable task $TaskPath$TaskName : $_"
   931→    }
   932→}

  1747→    }
  1748→
  1749→    # === HARDWARE-AWARE: RAM detection ===
  1750→    $totalRAM = 0
  1751→    try {
  1752→        $totalRAM = [math]::Round((Get-CimInstance Win32_ComputerSystem).TotalPhysicalMemory / 1GB, 2)

  1763→    } catch { Write-Warn "Hibernation disable failed: $_" }

  1769→        -SafetyLevel "Safe" -Description "Disables Fast Startup — prevents partial shutdown/driver issues" -SettingID "Perf_NoFastStartup"

  1774→        -Name "ModernSleep" -Value 0 -Type DWord `
  1775→        -SafetyLevel "Safe" -Description "Disables Modern Standby network — prevents battery drain during sleep" -SettingID "Perf_NoModernSleep"
  1776→

  1778→    if ($script:IsLaptop) {

  1781→        Write-Step "Keeping Balanced power plan (laptop)."
  1782→    } else {

  1786→            $highPerf = powercfg.exe /list | Select-String -Pattern "High performance|高性能"

  1794→                $highPerf = powercfg.exe /list | Select-String -Pattern "High performance|高性能"

  1796→                    $guid = ($highPerf.ToString() -split '\s+')[3]
  1797→                    powercfg.exe /setactive $guid 2>$null
  1798→                    Write-Success "High Performance plan created and activated ($guid)"
  1799→                } else { Write-Warn "Could not find or create High Performance plan." }
  1800→            }
  1801→        } catch { Write-Warn "Power plan change failed: $_" }

  1805→        try {
  1806→            powercfg.exe /setacvalueindex SCHEME_CURRENT SUB_BUTTONS LIDACTION 0 2>$null
  1807→            powercfg.exe /setdcvalueindex SCHEME_CURRENT SUB_BUTTONS LIDACTION 0 2>$null
  1808→            powercfg.exe /setactive SCHEME_CURRENT 2>$null

  1934→
  1935→    # 11. Disable NTFS last access time updates (improves SSD performance)

  1942→    }
  1943→
  1944→    Write-Host ""
  1945→    Write-Success "Performance and Power optimization complete."
  1946→}
