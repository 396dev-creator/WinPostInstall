   158→function Set-DefenderExclusions {
   159→    <#
   160→    .DESCRIPTION
   161→        Adds this script and its working directory to Windows Defender exclusions.
   162→        Prevents the script from being flagged as malware during execution.
   163→        This is safe: exclusions are path-specific and only protect this tool.
   164→    #>
   165→    Write-Host "" -ForegroundColor Cyan
   166→    Write-Host "  [AV SHIELD] Windows Defender Self-Exclusion" -ForegroundColor Cyan
   167→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   168→    Write-Host ""
   169→
   170→    $scriptPath = $PSCommandPath
   171→    if (-not $scriptPath) { $scriptPath = $MyInvocation.PSCommandPath }
   172→
   173→    $exclusionAdded = $false
   174→
   175→    # Check if the exclusion already exists
   176→    try {
   177→        $existingExclusions = (Get-MpPreference -ErrorAction SilentlyContinue).ExclusionPath
   178→        if ($existingExclusions -and $scriptPath -in $existingExclusions) {
   179→            Write-Step "Defender exclusion already exists for this script."
   180→            return
   181→        }
   182→    } catch {
   183→        Write-Warn "Could not read current Defender preferences: $_"
   184→    }
   185→
   186→    # Exclude the script file itself
   187→    Write-Step "Adding script to Defender exclusions..."
   188→    try {
   189→        Add-MpPreference -ExclusionPath $scriptPath -ErrorAction Stop
   190→        Write-Success "Excluded: $scriptPath"
   191→        $exclusionAdded = $true
   192→    } catch {
   193→        Write-Warn "Failed to exclude script path: $_"
   194→    }
   195→
   196→    # Exclude the working directory (where the script lives)
   197→    $scriptDir = Split-Path $scriptPath -Parent
   198→    if ($scriptDir -and (Test-Path $scriptDir)) {
   199→        try {
   200→            Add-MpPreference -ExclusionPath $scriptDir -ErrorAction Stop
   201→            Write-Success "Excluded directory: $scriptDir"
   202→            $exclusionAdded = $true
   203→        } catch {
   204→            Write-Warn "Failed to exclude directory: $_"
   205→        }
   206→    }
   207→
   208→    # Exclude the log directory (C:\Temp)
   209→    $logDir = "$env:SYSTEMDRIVE\Temp"
   210→    try {
   211→        Add-MpPreference -ExclusionPath $logDir -ErrorAction Stop
   212→        Write-Success "Excluded log directory: $logDir"
   213→        $exclusionAdded = $true
   214→    } catch {
   215→        Write-Warn "Failed to exclude log directory: $_"
   216→    }
   217→
   218→    # Exclude common technician tool folders
   219→    $toolPaths = @("C:\Tools", "C:\Scripts", "C:\TechnicianTools")
   220→    foreach ($path in $toolPaths) {
   221→        if (Test-Path $path) {
   222→            try {
   223→                Add-MpPreference -ExclusionPath $path -ErrorAction Stop
   224→                Write-Success "Excluded: $path"
   225→            } catch { }
   226→        }
   227→    }
   228→
   229→    if ($exclusionAdded) {
   230→        Write-Success "Defender exclusions configured. This script is now trusted."
   231→    }
   232→
   233→    Write-Host ""
   234→    Write-Host "  NOTE: This only affects Windows Defender. For third-party AV," -ForegroundColor DarkGray
   235→    Write-Host "  consider purchasing a Code Signing Certificate for full compatibility." -ForegroundColor DarkGray
   236→    Write-Host ""
   237→}

   239→function Initialize-Script {
   240→    if (-not ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
   241→        Write-Host ""
   242→        Write-Host "================================================================" -ForegroundColor Red
   243→        Write-Host "  [ERROR] This script must be run as Administrator." -ForegroundColor Red
   244→        Write-Host "================================================================" -ForegroundColor Red
   245→        Write-Host ""
   246→        Write-Host "  Right-click PowerShell -> 'Run as Administrator'" -ForegroundColor Yellow
   247→        Write-Host "  Then re-run this script." -ForegroundColor Yellow
   248→        Write-Host ""
   249→        Write-Host "  Press Enter to exit..." -ForegroundColor DarkGray
   250→        Read-Host "  "
   251→        exit 1
   252→    }
   253→
   254→    $logDir = "$env:SYSTEMDRIVE\Temp"
   255→    if (-not (Test-Path $logDir)) { New-Item -Path $logDir -ItemType Directory -Force | Out-Null }
   256→
   257→    try { Start-Transcript -Path $LogFile -Force -ErrorAction Stop | Out-Null } catch {
   258→        Write-Warn "Could not start transcript to $LogFile : $_"
   259→    }
   260→
   261→    # Set up Defender self-exclusion FIRST (before any optimization work)
   262→    Set-DefenderExclusions
   263→
   264→    # Registry backup - export critical hives before making any changes
   265→    Write-Host "" -ForegroundColor Cyan
   266→    Write-Host "  [SAFETY] Registry Backup" -ForegroundColor Cyan
   267→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   268→    Write-Host ""
   269→    try {
   270→        if (-not (Test-Path $script:BackupDir)) {
   271→            New-Item -Path $script:BackupDir -ItemType Directory -Force | Out-Null
   272→        }
   273→        Write-Step "Exporting registry hives to $script:BackupDir..."
   274→        $regExports = @(
   275→            @{ Hive = "HKLM\SOFTWARE\Policies"; File = "Policies.reg" },
   276→            @{ Hive = "HKLM\SOFTWARE\Microsoft\Windows\CurrentVersion\Policies"; File = "WinPolicies.reg" },
   277→            @{ Hive = "HKCU\SOFTWARE\Microsoft\Windows\CurrentVersion\Explorer"; File = "Explorer.reg" },
   278→            @{ Hive = "HKCU\SOFTWARE\Microsoft\Windows\CurrentVersion\GameDVR"; File = "GameDVR.reg" }
   279→        )
   280→        foreach ($regExport in $regExports) {
   281→            $exportPath = Join-Path $script:BackupDir $regExport.File
   282→            try {
   283→                reg export $regExport.Hive $exportPath /y 2>$null | Out-Null
   284→                Write-Success "Exported: $($regExport.Hive)"
   285→            } catch {
   286→                Write-Warn "Could not export $($regExport.Hive): $_"
   287→            }
   288→        }
   289→        Write-Success "Registry backup saved to: $script:BackupDir"
   290→    } catch {
   291→        Write-Warn "Registry backup failed: $_"
   292→    }
   293→    Write-Host ""
   294→
   295→    $osInfo = Get-CimInstance -ClassName Win32_OperatingSystem
   296→    $script:OSVersion = [version]$osInfo.Version
   297→    $script:OSBuild = $osInfo.BuildNumber
   298→    $script:IsWin11 = $script:OSBuild -ge 22000
   299→    $script:OSCaption = if ($script:IsWin11) { "Windows 11" } else { "Windows 10" }
   300→    $script:LocalComputerName = $env:COMPUTERNAME
   301→
   302→    # Laptop vs Desktop detection
   303→    $script:IsLaptop = $false
   304→    try {
   305→        $chassisTypes = Get-CimInstance -ClassName Win32_ChassisType -ErrorAction SilentlyContinue
   306→        if ($chassisTypes) {
   307→            $script:IsLaptop = ($chassisTypes | Where-Object { $_.ChassisType -in @(9, 10, 14) }).Count -gt 0
   308→        }
   309→    } catch { }
   310→    if (-not $script:IsLaptop) {
   311→        try {
   312→            $script:IsLaptop = (Get-CimInstance -ClassName Win32_Battery -ErrorAction SilentlyContinue).Count -gt 0
   313→        } catch { }
   314→    }
   315→
   316→    Write-Host ""
   317→    Write-Host "================================================================" -ForegroundColor Cyan
   318→    Write-Host "  Windows Post-Installation Optimizer v$ScriptVersion" -ForegroundColor Cyan
   319→    Write-Host "  Target: $script:OSCaption (Build $script:OSBuild)" -ForegroundColor Cyan
   320→    Write-Host "  Computer: $script:LocalComputerName" -ForegroundColor Cyan
   321→    Write-Host "  Chassis: $(if ($script:IsLaptop) { 'Laptop' } else { 'Desktop' })" -ForegroundColor Cyan
   322→    Write-Host "  Log file: $LogFile" -ForegroundColor DarkGray
   323→    Write-Host "================================================================" -ForegroundColor Cyan
   324→    Write-Host ""
   325→
   326→    # Run pre-flight checks
   327→    Test-Prerequisites
   328→}

   330→function Test-Prerequisites {
   331→    <#
   332→    .DESCRIPTION
   333→        Pre-flight checks: internet connectivity, disk space, pending reboot, battery.
   334→        Warns on issues but does not block execution.
   335→    #>
   336→    Write-Host "" -ForegroundColor Cyan
   337→    Write-Host "  [PRE-FLIGHT] System Prerequisites Check" -ForegroundColor Cyan
   338→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   339→    Write-Host ""
   340→
   341→    $checksPassed = 0; $checksWarned = 0
   342→
   343→    # 1. Internet connectivity
   344→    Write-Step "Checking internet connectivity..."
   345→    try {
   346→        $netTest = Test-Connection -ComputerName 8.8.8.8 -Count 1 -Quiet -ErrorAction Stop
   347→        if ($netTest) {
   348→            Write-Success "Internet connectivity: OK"
   349→            $checksPassed++
   350→        } else {
   351→            Write-Warn "Internet connectivity: OFFLINE (some features may fail)"
   352→            $checksWarned++
   353→        }
   354→    } catch {
   355→        Write-Warn "Internet connectivity: Could not check (some features may fail)"
   356→        $checksWarned++
   357→    }
   358→
   359→    # 2. Free disk space
   360→    Write-Step "Checking free disk space..."
   361→    try {
   362→        $sysDrive = Get-PSDrive -Name ($env:SYSTEMDRIVE -replace ':','') -ErrorAction Stop
   363→        $freeGB = [math]::Round($sysDrive.Free / 1GB, 2)
   364→        if ($freeGB -gt 10) {
   365→            Write-Success "Free disk space: $freeGB GB (OK)"
   366→            $checksPassed++
   367→        } else {
   368→            Write-Warn "Free disk space: $freeGB GB (LOW - recommend >10 GB)"
   369→            $checksWarned++
   370→        }
   371→    } catch {
   372→        Write-Warn "Could not check disk space: $_"
   373→        $checksWarned++
   374→    }
   375→
   376→    # 3. Pending reboot
   377→    Write-Step "Checking for pending reboot..."
   378→    $rebootPending = $false
   379→    $cbsPath = "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Component Based Servicing\RebootPending"
   380→    $wuPath = "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\WindowsUpdate\Auto Update\RebootRequired"
   381→    if (Test-Path $cbsPath -ErrorAction SilentlyContinue) { $rebootPending = $true }
   382→    if (Test-Path $wuPath -ErrorAction SilentlyContinue) { $rebootPending = $true }
   383→    if ($rebootPending) {
   384→        Write-Warn "Pending reboot detected! A restart is recommended before running optimizations."
   385→        $checksWarned++
   386→    } else {
   387→        Write-Success "No pending reboot."
   388→        $checksPassed++
   389→    }
   390→
   391→    # 4. Battery check (if laptop)
   392→    if ($script:IsLaptop) {
   393→        Write-Step "Checking battery level..."
   394→        try {
   395→            $battery = Get-CimInstance -ClassName Win32_Battery -ErrorAction Stop
   396→            $onAC = $battery.BatteryStatus -eq 2 -or $battery.BatteryStatus -eq 6 -or $battery.BatteryStatus -eq 7 -or $battery.BatteryStatus -eq 8
   397→            if ($onAC) {
   398→                Write-Success "On AC power."
   399→                $checksPassed++
   400→            } elseif ($battery.EstimatedChargeRemaining -gt 50) {
   401→                Write-Success "Battery level: $($battery.EstimatedChargeRemaining)% (OK)"
   402→                $checksPassed++
   403→            } else {
   404→                Write-Warn "Battery level: $($battery.EstimatedChargeRemaining)% - recommend AC power or >50% charge."
   405→                $checksWarned++
   406→            }
   407→        } catch {
   408→            Write-Warn "Could not check battery: $_"
   409→            $checksWarned++
   410→        }
   411→    }
   412→
   413→    Write-Host ""
   414→    Write-Host "  Pre-flight summary: $checksPassed passed, $checksWarned warnings" -ForegroundColor $(if ($checksWarned -gt 0) { 'Yellow' } else { 'Green' })
   415→    Write-Host ""
   416→}

   418→function Get-BootDriveType {
   419→    <#
   420→    .DESCRIPTION
   421→        Detects whether the boot drive is an SSD or HDD.
   422→        Returns "SSD" or "HDD".
   423→    #>
   424→    $bootType = "HDD"  # Default assumption
   425→    try {
   426→        # Method 1: Get-PhysicalDisk (most reliable on Win8+)
   427→        $physicalDisk = Get-PhysicalDisk -ErrorAction SilentlyContinue | Where-Object DeviceId -eq 0
   428→        if ($physicalDisk) {
   429→            if ($physicalDisk.MediaType -eq "SSD") { return "SSD" }
   430→            if ($physicalDisk.MediaType -eq "HDD") { return "HDD" }
   431→        }
   432→    } catch { }
   433→
   434→    try {
   435→        # Method 2: Get-Disk BusType fallback
   436→        $bootDisk = Get-Disk -ErrorAction SilentlyContinue | Where-Object IsBoot -eq $true
   437→        if ($bootDisk) {
   438→            $ssdBusTypes = @("NVMe", "SATA", "RAID", "SCSI")
   439→            # Check via rotation rate (0 = SSD on some controllers)
   440→            if ($bootDisk.BusType -in $ssdBusTypes) {
   441→                # Not conclusive, but check rotation rate
   442→                $detail = Get-PhysicalDisk -ErrorAction SilentlyContinue | Where-Object DeviceId -eq $bootDisk.Number
   443→                if ($detail -and $detail.MediaType -eq "SSD") { return "SSD" }
   444→            }
   445→        }
   446→    } catch { }
   447→
   448→    try {
   449→        # Method 3: Check if defrag is scheduled (SSD = not scheduled)
   450→        $defragTask = Get-ScheduledTask -TaskName "ScheduledDefrag" -ErrorAction SilentlyContinue
   451→        if ($defragTask -and $defragTask.State -eq 'Disabled') {
   452→            # If scheduled defrag is disabled, likely SSD
   453→            # This is a weak heuristic, don't rely on it alone
   454→        }
   455→    } catch { }
   456→
   457→    return $bootType
   458→}

   476→function Enable-SystemRestore {
   477→    <#
   478→    .DESCRIPTION
   479→        Ensures System Restore is enabled and configured on the system drive.
   480→        Enables the service if disabled, allocates disk space, and verifies
   481→        the feature is operational before returning.
   482→    #>
   483→    Write-Host "" -ForegroundColor Cyan
   484→    Write-Host "  [SAFETY] System Restore Verification" -ForegroundColor Cyan
   485→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   486→    Write-Host ""
   487→
   488→    $systemDrive = $env:SYSTEMDRIVE
   489→    $restoreEnabled = $false
   490→
   491→    # Step 1: Check if System Restore is currently enabled on the system drive
   492→    Write-Step "Checking System Restore status on $systemDrive..."
   493→    try {
   494→        $srStatus = Get-ComputerInfo -Property "SystemRestoreStatus" -ErrorAction SilentlyContinue
   495→        # Also check via WMI which is more reliable
   496→        $srInfo = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction SilentlyContinue
   497→        if ($srInfo) {
   498→            foreach ($sr in $srInfo) {
   499→                if ($sr.RPSequence -gt 0) {
   500→                    $restoreEnabled = $true
   501→                    break
   502→                }
   503→            }
   504→        }
   505→    } catch {
   506→        # WMI query may fail if System Restore is completely disabled
   507→        $restoreEnabled = $false
   508→    }
   509→
   510→    # Alternative check: try to list existing restore points
   511→    if (-not $restoreEnabled) {
   512→        try {
   513→            $existingPoints = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction Stop
   514→            if ($existingPoints -and $existingPoints.Count -ge 0) {
   515→                $restoreEnabled = $true
   516→            }
   517→        } catch {
   518→            $restoreEnabled = $false
   519→        }
   520→    }
   521→
   522→    # Step 2: Enable System Restore if not currently enabled
   523→    if (-not $restoreEnabled) {
   524→        Write-Warn "System Restore is NOT enabled on $systemDrive."
   525→        Write-Step "Enabling System Restore on $systemDrive..."
   526→
   527→        try {
   528→            # Enable System Restore via registry (most reliable method)
   529→            $srPolicyPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore"
   530→            if (-not (Test-Path $srPolicyPath)) {
   531→                New-Item -Path $srPolicyPath -Force -ErrorAction SilentlyContinue | Out-Null
   532→            }
   533→            # DisableSR = 0 means System Restore is ENABLED
   534→            Set-ItemProperty -Path $srPolicyPath -Name "DisableSR" -Value 0 -Type DWord -Force -ErrorAction Stop
   535→
   536→            # Also enable via PowerShell cmdlet (Windows 10/11)
   537→            try {
   538→                Enable-ComputerRestore -Drive $systemDrive -ErrorAction Stop
   539→            } catch {
   540→                # Fallback: use vssadmin to enable shadow copies
   541→                $vssResult = vssadmin list shadows 2>&1
   542→                if ($LASTEXITCODE -ne 0) {
   543→                    # Try creating a shadow copy to force-enable VSS
   544→                    vssadmin create shadow /For=$systemDrive 2>&1 | Out-Null
   545→                }
   546→            }
   547→
   548→            # Start the System Restore service if not running
   549→            $srService = Get-Service -Name "sr" -ErrorAction SilentlyContinue
   550→            if ($srService -and $srService.Status -ne 'Running') {
   551→                try {
   552→                    Set-Service -Name "sr" -StartupType Manual -ErrorAction Stop
   553→                    Start-Service -Name "sr" -ErrorAction Stop
   554→                } catch {
   555→                    Write-Warn "Could not start System Restore service: $_"
   556→                }
   557→            }
   558→
   559→            # Also ensure VSS (Volume Shadow Copy) service is running (required for System Restore)
   560→            $vssService = Get-Service -Name "VSS" -ErrorAction SilentlyContinue
   561→            if ($vssService -and $vssService.Status -ne 'Running') {
   562→                try {
   563→                    Set-Service -Name "VSS" -StartupType Manual -ErrorAction Stop
   564→                    Start-Service -Name "VSS" -ErrorAction Stop
   565→                    Write-Success "VSS service started."
   566→                } catch {
   567→                    Write-Warn "Could not start VSS service: $_"
   568→                }
   569→            }
   570→
   571→            Start-Sleep -Seconds 2
   572→            Write-Success "System Restore has been ENABLED on $systemDrive."
   573→        } catch {
   574→            Write-Err "Failed to enable System Restore: $_"
   575→            Write-Warn "Continuing without restore point. Manual enable: System Properties > System Protection > Configure > Turn on"
   576→        }
   577→    } else {
   578→        Write-Success "System Restore is already enabled on $systemDrive."
   579→    }
   580→
   581→    # Step 3: Configure disk space allocation for restore points
   582→    Write-Step "Configuring System Restore disk space..."
   583→    try {
   584→        # Set maximum disk space usage to 5% (reasonable for most systems)
   585→        # This is done via registry: RPSessionInterval and DiskPercent
   586→        $srCfgPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore\Cfg"
   587→        if (-not (Test-Path $srCfgPath)) {
   588→            New-Item -Path $srCfgPath -Force -ErrorAction SilentlyContinue | Out-Null
   589→        }
   590→        # DiskPercent = 5 means 5% of disk (0 = unlimited, we use 5% as safe default)
   591→        Set-ItemProperty -Path $srCfgPath -Name "DiskPercent" -Value 5 -Type DWord -Force -ErrorAction SilentlyContinue
   592→        # RPSessionInterval = 0 means create restore points at any interval
   593→        Set-ItemProperty -Path $srCfgPath -Name "RPSessionInterval" -Value 0 -Type DWord -Force -ErrorAction SilentlyContinue
   594→        Write-Success "Restore disk space set to 5% of drive."
   595→    } catch {
   596→        Write-Warn "Could not configure restore disk space: $_"
   597→    }
   598→
   599→    Write-Host ""
   600→}

   602→function New-SystemRestorePoint {
   603→    <#
   604→    .DESCRIPTION
   605→        Creates a System Restore point before any optimization changes are made.
   606→        This provides a safety net so the user can roll back all changes.
   607→    #>
   608→    Write-Host "" -ForegroundColor Cyan
   609→    Write-Host "  [SAFETY] Creating System Restore Point" -ForegroundColor Cyan
   610→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   611→    Write-Host ""
   612→
   613→    $restorePointCreated = $false
   614→    $description = "Pre-Optimization Restore Point - WinPostInstall v$ScriptVersion - $(Get-Date -Format 'yyyy-MM-dd HH:mm')"
   615→
   616→    # Method 1: Use Checkpoint-Computer (PowerShell cmdlet, Windows 8+)
   617→    Write-Step "Creating restore point: 'Pre-Optimization'..."
   618→    try {
   619→        Checkpoint-Computer -Description $description -RestorePointType "MODIFY_SETTINGS" -ErrorAction Stop
   620→        $restorePointCreated = $true
   621→        Write-Success "Restore point created successfully via Checkpoint-Computer."
   622→    } catch {
   623→        Write-Warn "Checkpoint-Computer failed: $_"
   624→    }
   625→
   626→    # Method 2: Fallback using WMI if Method 1 failed
   627→    if (-not $restorePointCreated) {
   628→        Write-Step "Trying WMI method to create restore point..."
   629→        try {
   630→            $wmiSR = [System.Windows.Forms.SystemRestore]
   631→            # WMI approach via Invoke-CimMethod
   632→            $result = Invoke-CimMethod -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" `
   633→                -MethodName "CreateRestorePoint" `
   634→                -Arguments @{
   635→                    Description         = $description
   636→                    RestorePointType    = 12  # MODIFY_SETTINGS
   637→                    EventType           = 100  # BEGIN_NESTED_SYSTEM_CHANGE
   638→                } -ErrorAction Stop
   639→            if ($result -and $result.ReturnValue -eq 0) {
   640→                $restorePointCreated = $true
   641→                Write-Success "Restore point created via WMI."
   642→            } else {
   643→                Write-Warn "WMI returned non-zero: $result"
   644→            }
   645→        } catch {
   646→            Write-Warn "WMI method failed: $_"
   647→        }
   648→    }
   649→
   650→    # Method 3: Final fallback via srclient.dll COM automation
   651→    if (-not $restorePointCreated) {
   652→        Write-Step "Trying COM automation (srclient.dll)..."
   653→        try {
   654→            $srClient = New-Object -ComObject "SRClient.RestorePoint" -ErrorAction Stop
   655→            # 7 = MODIFY_SETTINGS, 100 = BEGIN_NESTED_SYSTEM_CHANGE
   656→            $srClient.CreateRestorePoint($description, 7, 100)
   657→            $restorePointCreated = $true
   658→            Write-Success "Restore point created via COM automation."
   659→        } catch {
   660→            Write-Warn "COM method failed: $_"
   661→        }
   662→    }
   663→
   664→    if ($restorePointCreated) {
   665→        Write-Host ""
   666→        Write-Host "  ====================================================================" -ForegroundColor Green
   667→        Write-Host "  RESTORE POINT CREATED: 'Pre-Optimization'" -ForegroundColor Green
   668→        Write-Host "  You can roll back all changes via:" -ForegroundColor White
   669→        Write-Host "    System Properties > System Protection > System Restore" -ForegroundColor White
   670→        Write-Host "  ====================================================================" -ForegroundColor Green
   671→        Write-Host ""
   672→    } else {
   673→        Write-Host ""
   674→        Write-Warn "Could NOT create restore point automatically."
   675→        Write-Host "  Manual option: System Properties > System Protection > Create" -ForegroundColor Yellow
   676→        Write-Host "  Continuing without restore point backup." -ForegroundColor Yellow
   677→        Write-Host ""
   678→        if (-not $Silent) {
   679→            $continue = Read-Host "  Continue without restore point? (y/N)"
   680→            if ($continue -ne 'y' -and $continue -ne 'Y') {
   681→                Write-Host "  Aborted by user." -ForegroundColor Yellow
   682→                exit 0
   683→            }
   684→        }
   685→    }
   686→}

   688→function Set-Registry {
   689→    [CmdletBinding()]
   690→    param(
   691→        [string]$Path,
   692→        [string]$Name,
   693→        $Value,
   694→        [string]$Type = "DWord",
   695→        [ValidateSet("Safe","Moderate","Risky")]
   696→        [string]$SafetyLevel = "Safe",
   697→        [string]$Description = "",
   698→        [string]$SettingID = ""
   699→    )
   700→
   701→    # Generate a setting ID if not provided
   702→    if ([string]::IsNullOrEmpty($SettingID)) {
   703→        $SettingID = "$($Path.Replace('\','_').Replace(':','').Replace(' ',''))_$Name"
   704→    }
   705→
   706→    # Register in setting catalog for interactive menu
   707→    if (-not [string]::IsNullOrEmpty($Description)) {
   708→        $script:SettingCatalog[$SettingID] = @{
   709→            Section  = $script:CurrentSection
   710→            Safety   = $SafetyLevel
   711→            Desc     = $Description
   712→            Action   = "$Path\$Name = $Value"
   713→            Type     = "Registry"
   714→        }
   715→    }
   716→
   717→    # Collect risky descriptions for menu warnings
   718→    if ($SafetyLevel -eq "Risky" -and -not [string]::IsNullOrEmpty($Description)) {
   719→        $script:RiskyDescriptions += "$SettingID : $Description"
   720→    }
   721→
   722→    # Check if user has toggled this setting OFF via interactive menu
   723→    if ($script:SettingOverrides.ContainsKey($SettingID) -and $script:SettingOverrides[$SettingID]) {
   724→        Write-Host "  SKIPPED (user toggle): $Description" -ForegroundColor DarkGray
   725→        $script:TotalSkipped++
   726→        return
   727→    }
   728→
   729→    # Safety gate: Skip Risky settings unless explicitly allowed
   730→    if ($SafetyLevel -eq "Risky" -and -not $script:AllowRisky) {
   731→        Write-Host "  SKIPPED (risky): $Description — use -AllowRisky or enable in menu" -ForegroundColor Yellow
   732→        $script:TotalSkipped++
   733→        return
   734→    }
   735→
   736→    # Warn for Moderate settings
   737→    if ($SafetyLevel -eq "Moderate") {
   738→        Write-Host "  [MODERATE] $Description" -ForegroundColor DarkYellow
   739→    }
   740→
   741→    try {
   742→        if (-not (Test-Path $Path)) {
   743→            New-Item -Path $Path -Force -ErrorAction SilentlyContinue | Out-Null
   744→        }
   745→        # Idempotency: Check if the value already matches what we want to set
   746→        $currentValue = $null
   747→        try {
   748→            $currentValue = (Get-ItemProperty -Path $Path -Name $Name -ErrorAction Stop).$Name
   749→        } catch {
   750→            # Property doesn't exist yet — need to create it
   751→        }
   752→
   753→        # Compare current vs desired value
   754→        $valueMatches = $false
   755→        if ($null -ne $currentValue) {
   756→            if ($Type -eq "DWord" -or $Type -eq "QWord") {
   757→                # Numeric comparison (handle both int and string inputs)
   758→                $valueMatches = ([int]$currentValue -eq [int]$Value)
   759→            } elseif ($Type -eq "String" -or $Type -eq "ExpandString") {
   760→                $valueMatches = ([string]$currentValue -eq [string]$Value)
   761→            } else {
   762→                $valueMatches = ($currentValue -eq $Value)
   763→            }
   764→        }
   765→
   766→        if ($valueMatches) {
   767→            Write-Step "Already set: $Path\$Name = $Value"
   768→            $script:TotalSkipped++
   769→            return
   770→        }
   771→
   772→        # Capture old value for undo script before writing
   773→        if ($null -ne $currentValue) {
   774→            $oldVal = $currentValue.ToString()
   775→            $null = $script:UndoCommands.Add("Set-ItemProperty -Path '$Path' -Name '$Name' -Value '$oldVal' -Type '$Type' -Force")
   776→            $null = $script:UndoDescriptions.Add("Restore registry: $Path\$Name = $oldVal")
   777→        } else {
   778→            $null = $script:UndoCommands.Add("Remove-ItemProperty -Path '$Path' -Name '$Name' -Force -ErrorAction SilentlyContinue")
   779→            $null = $script:UndoDescriptions.Add("Remove registry key that was added: $Path\$Name")
   780→        }
   781→
   782→        Set-ItemProperty -Path $Path -Name $Name -Value $Value -Type $Type -Force -ErrorAction Stop
   783→        Write-Success "Registry: $Path\$Name = $Value"
   784→        $script:TotalChanged++
   785→    }
   786→    catch {
   787→        Write-Err "Failed to set $Path\$Name : $_"
   788→        $script:TotalFailed++
   789→    }
   790→}

   792→function Remove-Registry {
   793→    [CmdletBinding()]
   794→    param(
   795→        [string]$Path,
   796→        [string]$Name
   797→    )
   798→    try {
   799→        if (Test-Path $Path) {
   800→            # Idempotency: Check if property exists before removing
   801→            $existing = $null
   802→            try { $existing = (Get-ItemProperty -Path $Path -Name $Name -ErrorAction Stop).$Name } catch { }
   803→            if ($null -ne $existing) {
   804→                Remove-ItemProperty -Path $Path -Name $Name -Force -ErrorAction Stop 2>$null
   805→                Write-Success "Registry removed: $Path\$Name"
   806→            } else {
   807→                Write-Step "Already removed: $Path\$Name"
   808→            }
   809→        }
   810→    }
   811→    catch {
   812→        Write-Warn "Could not remove $Path\$Name : $_"
   813→    }

   826→

   829→        $SettingID = "Task_$($TaskPath.Replace('\','_').Replace(' ',''))_$TaskName"
   830→    }
   831→
   832→    # Register in setting catalog for interactive menu
   833→    if (-not [string]::IsNullOrEmpty($Description)) {
   834→        $script:SettingCatalog[$SettingID] = @{
   835→            Section  = $script:CurrentSection
   836→            Safety   = $SafetyLevel
   837→            Desc     = $Description
   838→            Action   = "Disable task: $TaskPath$TaskName"
   839→            Type     = "ScheduledTask"
   840→        }
   841→    }
   842→
   843→    # Collect risky descriptions for menu warnings
   844→    if ($SafetyLevel -eq "Risky" -and -not [string]::IsNullOrEmpty($Description)) {
   845→        $script:RiskyDescriptions += "$SettingID : $Description"
   846→    }
   847→

   850→        Write-Host "  SKIPPED (user toggle): $Description" -ForegroundColor DarkGray
   851→        $script:TotalSkipped++
   852→        return
   853→    }
   854→
   855→    # Safety gate: Skip Risky settings unless explicitly allowed
   856→    if ($SafetyLevel -eq "Risky" -and -not $script:AllowRisky) {
   857→        Write-Host "  SKIPPED (risky): $Description — use -AllowRisky or enable in menu" -ForegroundColor Yellow
   858→        $script:TotalSkipped++
   859→        return
   860→    }
   861→
   862→    # Warn for Moderate settings
   863→    if ($SafetyLevel -eq "Moderate") {
   864→        Write-Host "  [MODERATE] $Description" -ForegroundColor DarkYellow
   865→    }
   866→
   867→    try {
   868→        $task = Get-ScheduledTask -TaskPath $TaskPath -TaskName $TaskName -ErrorAction SilentlyContinue
   869→        if ($task) {
   870→            if ($task.State -eq 'Disabled') {
   871→                Write-Step "Already disabled: $TaskPath$TaskName"
   872→                $script:TotalSkipped++
   873→            } else {
   874→                # Safety: Check for dependent services/tasks before disabling
   875→                $deps = $task.DependentOn
   876→                if ($deps) {
   877→                    Write-Host "  [DEPENDENCY] Task $TaskName has dependents: $($deps.TaskName -join ', ')" -ForegroundColor DarkYellow
   878→                }
   879→                $task | Disable-ScheduledTask -ErrorAction Stop
   880→                Write-Success "Task disabled: $TaskPath$TaskName"
   881→                $script:TotalChanged++
   882→                $null = $script:UndoCommands.Add("Enable-ScheduledTask -TaskPath '$TaskPath' -TaskName '$TaskName' -ErrorAction SilentlyContinue")
   883→                $null = $script:UndoDescriptions.Add("Re-enable scheduled task: $TaskPath$TaskName")
   884→            }
   885→        }
   886→    }
   887→    catch {
   888→        Write-Warn "Could not disable task $TaskPath$TaskName : $_"
   889→    }
   890→}
