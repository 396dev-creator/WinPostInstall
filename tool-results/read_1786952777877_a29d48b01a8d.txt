    39→function Initialize-Script {
    40→    # Require Administrator
    41→    if (-not ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
    42→        Write-Host "[ERROR] This script must be run as Administrator." -ForegroundColor Red
    43→        Write-Host "Right-click PowerShell and select 'Run as Administrator', then re-run this script." -ForegroundColor Yellow
    44→        if (-not $Silent) { Read-Host "Press Enter to exit" }
    45→        exit 1
    46→    }
    47→
    48→    # Create log directory
    49→    $logDir = "$env:SYSTEMDRIVE\Temp"
    50→    if (-not (Test-Path $logDir)) { New-Item -Path $logDir -ItemType Directory -Force | Out-Null }
    51→
    52→    # Start transcript logging
    53→    Start-Transcript -Path $LogFile -Force | Out-Null
    54→
    55→    # Detect OS
    56→    $osInfo = Get-CimInstance -ClassName Win32_OperatingSystem
    57→    $script:OSVersion = [version]$osInfo.Version
    58→    $script:OSBuild = $osInfo.BuildNumber
    59→    $script:IsWin11 = $script:OSBuild -ge 22000
    60→    $script:OSCaption = if ($script:IsWin11) { "Windows 11" } else { "Windows 10" }
    61→    $script:ComputerName = $env:COMPUTERNAME
    62→
    63→    Write-Host ""
    64→    Write-Host "================================================================" -ForegroundColor Cyan
    65→    Write-Host "  Windows Post-Installation Optimizer v$ScriptVersion" -ForegroundColor Cyan
    66→    Write-Host "  Target: $script:OSCaption (Build $script:OSBuild)" -ForegroundColor Cyan
    67→    Write-Host "  Computer: $script:ComputerName" -ForegroundColor Cyan
    68→    Write-Host "  Log file: $LogFile" -ForegroundColor DarkGray
    69→    Write-Host "================================================================" -ForegroundColor Cyan
    70→    Write-Host ""
    71→}

    89→function Set-Registry {
    90→    <#
    91→    .SYNOPSIS
    92→        Safely set a registry value, creating the path if needed.
    93→    #>
    94→    [CmdletBinding()]
    95→    param(
    96→        [string]$Path,
    97→        [string]$Name,
    98→        $Value,
    99→        [string]$Type = "DWord"
   100→    )
   101→    try {
   102→        if (-not (Test-Path $Path)) {
   103→            New-Item -Path $Path -Force | Out-Null
   104→        }
   105→        Set-ItemProperty -Path $Path -Name $Name -Value $Value -Type $Type -Force -ErrorAction Stop
   106→        Write-Success "Registry: $Path\$Name = $Value"
   107→    }
   108→    catch {
   109→        Write-Err "Failed to set $Path\$Name : $_"
   110→    }
   111→}

   113→function Remove-Registry {
   114→    [CmdletBinding()]
   115→    param(
   116→        [string]$Path,
   117→        [string]$Name
   118→    )
   119→    try {
   120→        if (Test-Path $Path) {
   121→            Remove-ItemProperty -Path $Path -Name $Name -Force -ErrorAction Stop 2>$null
   122→            Write-Success "Registry removed: $Path\$Name"
   123→        }
   124→    }
   125→    catch {
   126→        Write-Warn "Could not remove $Path\$Name : $_"
   127→    }
   128→}

   136→function Invoke-PrivacyTweaks {
   137→    Write-Host ""
   138→    Write-Host "--- PRIVACY & TELEMETRY (Aggressive) ---" -ForegroundColor Magenta
   139→    Write-Host ""
   140→
   141→    # 1. Disable Telemetry via Registry
   142→    Write-Step "Disabling Windows telemetry..."
   143→    Set-Registry -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows\DataCollection" `
   144→        -Name "AllowTelemetry" -Value 0 -Type DWord
   145→
   146→    # Disable Diagnostic Data Viewer
   147→    Set-Registry -Path "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Diagnostics\DiagTrack" `
   148→        -Name "ShowedToastAtLevel" -Value 0 -Type DWord
   149→
   150→    # Disable Customer Experience Improvement Program (CEIP)
   151→    $ceipPaths = @(
   152→        "HKLM:\SOFTWARE\Policies\Microsoft\SQMClient\Windows",
   153→        "HKCU:\SOFTWARE\Microsoft\SQMClient\Windows"
   154→    )
   155→    foreach ($p in $ceipPaths) {
   156→        Set-Registry -Path $p -Name "CEIPEnable" -Value 0 -Type DWord
   157→    }
   158→
   159→    # 2. Disable Diagnostics Tracking Service
   160→    Write-Step "Stopping and disabling diagnostic tracking services..."
   161→    $diagServices = @(
   162→        "DiagTrack",           # Connected User Experiences and Telemetry
   163→        "dmwappushservice",    # WAP Push Message Routing Service
   164→        "WerSvc"               # Windows Error Reporting Service
   165→    )
   166→    foreach ($svc in $diagServices) {
   167→        try {
   168→            $service = Get-Service -Name $svc -ErrorAction SilentlyContinue
   169→            if ($service) {
   170→                Stop-Service -Name $svc -Force -ErrorAction SilentlyContinue
   171→                Set-Service -Name $svc -StartupType Disabled -ErrorAction SilentlyContinue
   172→                Write-Success "Service disabled: $svc"
   173→            } else {
   174→                Write-Warn "Service not found: $svc"
   175→            }
   176→        } catch {
   177→            Write-Warn "Could not disable service $svc : $_"
   178→        }
   179→    }
   180→
   181→    # 3. Disable Activity History
   182→    Write-Step "Disabling activity history..."
   183→    Set-Registry -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows\System" `
   184→        -Name "EnableActivityFeed" -Value 0 -Type DWord
   185→    Set-Registry -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows\System" `
   186→        -Name "PublishUserActivities" -Value 0 -Type DWord
   187→    Set-Registry -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows\System" `
   188→        -Name "UploadUserActivities" -Value 0 -Type DWord
   189→
   190→    # 4. Disable Location Tracking
   191→    Write-Step "Disabling location tracking..."
   192→    Set-Registry -Path "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\CapabilityAccessManager\ConsentStore\location" `
   193→        -Name "Value" -Value "Deny" -Type String
   194→
   195→    # 5. Disable Advertising ID
   196→    Write-Step "Disabling advertising ID..."
   197→    Set-Registry -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows\AdvertisingInfo" `
   198→        -Name "DisabledByGroupPolicy" -Value 1 -Type DWord
   199→    Set-Registry -Path "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\AdvertisingInfo" `
   200→        -Name "Enabled" -Value 0 -Type DWord
   201→
   202→    # 6. Disable Cortana
   203→    Write-Step "Disabling Cortana..."
   204→    Set-Registry -Path "HKLM:\SOFTWARE\Policies\Microsoft\WindowsStore" `
   205→        -Name "DisableWindowsStoreRealTimeConfig" -Value 1 -Type DWord
   206→
   207→    if ($script:IsWin11) {
   208→        # Win11 specific Cortana disable
   209→        $cortanaPath = "HKLM:\SOFTWARE\Policies\Microsoft\Windows\Windows Search"
   210→        Set-Registry -Path $cortanaPath -Name "AllowCortana" -Value 0 -Type DWord
   211→        Set-Registry -Path $cortanaPath -Name "DisableWebSearch" -Value 1 -Type DWord
   212→        Set-Registry -Path $cortanaPath -Name "AllowSearchToUseLocation" -Value 0 -Type DWord
   213→
   214→        # Disable Copilot (Win11)
   215→        Write-Step "Disabling Windows Copilot..."
   216→        Set-Registry -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows\WindowsCopilot" `
   217→            -Name "TurnOffWindowsCopilot" -Value 1 -Type DWord
   218→    } else {
   219→        # Win10 specific Cortana disable
   220→        Set-Registry -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows\Windows Search" `
   221→            -Name "AllowCortana" -Value 0 -Type DWord
   222→        Set-Registry -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows\Windows Search" `
   223→            -Name "DisableWebSearch" -Value 1 -Type DWord
   224→    }
   225→
   226→    # 7. Disable Timeline / Clipboard History (Win10)
   227→    if (-not $script:IsWin11) {
   228→        Write-Step "Disabling Timeline..."
   229→        Set-Registry -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows\System" `
   230→            -Name "EnableActivityFeed" -Value 0 -Type DWord
   231→    }
   232→
   233→    # 8. Disable Clipboard cloud sync
   234→    Write-Step "Disabling Clipboard cloud sync..."
   235→    Set-Registry -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows\System" `
   236→        -Name "AllowClipboardHistory" -Value 0 -Type DWord
   237→
   238→    # 9. Disable Feedback notifications
   239→    Write-Step "Disabling Feedback notifications..."
   240→    Set-Registry -Path "HKCU:\SOFTWARE\Microsoft\Siuf\Rules" `
   241→        -Name "NumberOfSIUFInPeriod" -Value 0 -Type DWord
   242→    Set-Registry -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows\DataCollection" `
   243→        -Name "DoNotShowFeedbackNotifications" -Value 1 -Type DWord
   244→
   245→    # 10. Disable Tailored Experiences (content-based suggestions)
   246→    Write-Step "Disabling Tailored Experiences..."
   247→    Set-Registry -Path "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\ContentDeliveryManager" `
   248→        -Name "SubscribedContent-338389Enabled" -Value 0 -Type DWord
   249→
   250→    # 11. Disable targeted tips
   251→    Write-Step "Disabling targeted tips & suggestions..."
   252→    Set-Registry -Path "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Explorer\Advanced" `
   253→        -Name "Start_TrackProgs" -Value 0 -Type DWord
   254→    Set-Registry -Path "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Explorer\Advanced" `
   255→        -Name "Start_TrackDocs" -Value 0 -Type DWord
   256→    Set-Registry -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows\CloudContent" `
   257→        -Name "DisableSoftLanding" -Value 1 -Type DWord
   258→    Set-Registry -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows\CloudContent" `
   259→        -Name "DisableWindowsConsumerFeatures" -Value 1 -Type DWord
   260→
   261→    # 12. Disable Text Input Telemetry
   262→    Write-Step "Disabling text input telemetry..."
   263→    $tipPath = "HKCU:\SOFTWARE\Microsoft\Input\Settings"
   264→    if (Test-Path $tipPath) {
   265→        Set-Registry -Path $tipPath -Name "InsightsDisabled" -Value 1 -Type DWord
   266→    }
   267→
   268→    # 13. Disable Wi-Fi Sense (auto-connect to open networks, share networks)
   269→    Write-Step "Disabling Wi-Fi Sense..."
   270→    Set-Registry -Path "HKLM:\SOFTWARE\Microsoft\PolicyManager\default\WiFi\AllowWiFiHotSpotReporting" `
   271→        -Name "Value" -Value 0 -Type DWord
   272→    Set-Registry -Path "HKLM:\SOFTWARE\Microsoft\PolicyManager\default\WiFi\AllowAutoConnectToWiFiSenseHotspots" `
   273→        -Name "Value" -Value 0 -Type DWord
   274→
   275→    Write-Host ""
   276→    Write-Success "Privacy & Telemetry section complete."
   277→}

   498→function Invoke-ServiceStartupOptimization {
   499→    Write-Host ""
   500→    Write-Host "--- SERVICE & STARTUP OPTIMIZATION ---" -ForegroundColor Magenta
   501→    Write-Host ""
   502→
   503→    # Services to disable for performance
   504→    $disableServices = @(
   505→        @{ Name = "WSearch";               Desc = "Windows Search" },
   506→        @{ Name = "WMPNetworkSvc";         Desc = "Windows Media Player Network Sharing" },
   507→        @{ Name = "WpcSvc";                Desc = "Parental Controls" },
   508→        @{ Name = "WerSvc";                Desc = "Windows Error Reporting" },
   509→        @{ Name = "TabletInputService";    Desc = "Tablet PC Input" },
   510→        @{ Name = "RetailDemo";            Desc = "Retail Demo Service" },
   511→        @{ Name = "PimIndexMaintenance";   Desc = "People Indexing" },
   512→        @{ Name = "MessagingService";      Desc = "Messaging Service" },
   513→        @{ Name = "MapsBroker";            Desc = "Downloaded Maps Manager" },
   514→        @{ Name = "lfsvc";                 Desc = "Geolocation Service" },
   515→        @{ Name = "HvHost";                Desc = "Hyper-V Host Compute Service" },
   516→        @{ Name = "wisvc";                 Desc = "Windows Insider Service" },
   517→        @{ Name = "WbioSrvc";              Desc = "Windows Biometric Service" }
   518→    )
   519→
   520→    foreach ($svc in $disableServices) {
   521→        try {
   522→            $service = Get-Service -Name $svc.Name -ErrorAction SilentlyContinue

  1028→    $input = Read-Host "  Enter section numbers (comma-separated, e.g. '1,3,5') or 'A' for all"
  1029→
  1030→    if ($input -eq '0') {
  1031→        Write-Host "  Exiting without changes." -ForegroundColor Yellow
  1032→        Stop-Transcript | Out-Null
  1033→        exit 0
  1034→    }
  1035→
  1036→    $selected = @()
  1037→
  1038→    if ($input.Trim().ToUpper() -eq 'A') {
  1039→        $selected = @("Privacy", "Debloat", "ExplorerUI", "ServiceStartup",
  1040→                       "Performance", "Security", "Network", "WindowsUpdate",
  1041→                       "DiskCleanup", "Winget", "FinalTouches")
  1042→    } else {
  1043→        $numbers = $input -split ',' | ForEach-Object { $_.Trim() }
  1044→        $allSections = @{
  1045→            '1' = 'Privacy';        '2' = 'Debloat';        '3' = 'ExplorerUI'
  1046→            '4' = 'ServiceStartup'; '5' = 'Performance';  '6' = 'Security'
  1047→            '7' = 'Network';        '8' = 'WindowsUpdate';  '9' = 'DiskCleanup'
  1048→            '10' = 'Winget';        '11' = 'FinalTouches'
  1049→        }
  1050→        foreach ($num in $numbers) {
  1051→            if ($allSections.ContainsKey($num)) {
  1052→                $selected += $allSections[$num]
  1053→            } else {
  1054→                Write-Warn "Invalid selection: $num (skipped)"
  1055→            }
  1056→        }
  1057→    }
  1058→
  1059→    return $selected
  1060→}
