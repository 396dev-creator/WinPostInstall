     3→export interface SettingDependency {
     4→  /** Setting IDs that this setting requires to be enabled */
     5→  requires?: string[];
     6→  /** Setting IDs that conflict with this setting (cannot both be enabled) */
     7→  conflicts?: string[];
     8→  /** Human-readable conflict message */
     9→  conflictMessage?: string;
    10→}

    12→export interface Setting {
    13→  id: string;
    14→  name: string;
    15→  safety: SafetyLevel;
    16→  desc: string;
    17→  /** Dependencies and conflicts with other settings */
    18→  deps?: SettingDependency;
    19→}

   543→/**
   544→ * Check for conflicts and unsatisfied dependencies given current settings state
   545→ */
   546→export function checkSettingConflicts(
   547→  settingId: string,
   548→  settingsState: Record<string, boolean>
   549→): { type: "conflict" | "unmet_require"; settingId: string; settingName: string; message: string }[] {
   550→  const warnings: { type: "conflict" | "unmet_require"; settingId: string; settingName: string; message: string }[] = [];
   551→  const deps = settingDependencies[settingId];
   552→  if (!deps) return warnings;
   553→
   554→  // Check conflicts
   555→  if (deps.conflicts) {
   556→    for (const conflictId of deps.conflicts) {
   557→      if (settingsState[conflictId]) {
   558→        // Find the name of the conflicting setting
   559→        let conflictName = conflictId;
   560→        for (const section of sections) {
   561→          const found = section.settings.find((s) => s.id === conflictId);
   562→          if (found) { conflictName = found.name; break; }
   563→        }
   564→        warnings.push({
   565→          type: "conflict",
   566→          settingId: conflictId,
   567→          settingName: conflictName,
   568→          message: deps.conflictMessage || `${settingId} conflicts with ${conflictName}`,
   569→        });
   570→      }
   571→    }
   572→  }
   573→
   574→  // Check unmet requirements (only warn when this setting is being enabled)
   575→  if (deps.requires && settingsState[settingId]) {
   576→    for (const reqId of deps.requires) {
   577→      if (!settingsState[reqId]) {
   578→        let reqName = reqId;
   579→        for (const section of sections) {
   580→          const found = section.settings.find((s) => s.id === reqId);
   581→          if (found) { reqName = found.name; break; }
   582→        }
   583→        warnings.push({
   584→          type: "unmet_require",
   585→          settingId: reqId,
   586→          settingName: reqName,
   587→          message: deps.conflictMessage || `${settingId} requires ${reqName} to be enabled`,
   588→        });
   589→      }
   590→    }
   591→  }
   592→
   593→  return warnings;
   594→}

   596→/**
   597→ * Get all conflicts for all currently enabled settings
   598→ */
   599→export function getAllConflicts(
   600→  settingsState: Record<string, boolean>
   601→): { settingId: string; settingName: string; type: "conflict" | "unmet_require"; targetId: string; targetName: string; message: string }[] {
   602→  const allWarnings: { settingId: string; settingName: string; type: "conflict" | "unmet_require"; targetId: string; targetName: string; message: string }[] = [];
   603→  const seen = new Set<string>();
   604→
   605→  for (const section of sections) {
   606→    for (const setting of section.settings) {
   607→      if (!settingsState[setting.id]) continue;
   608→      const deps = settingDependencies[setting.id];
   609→      if (!deps) continue;
   610→
   611→      if (deps.conflicts) {
   612→        for (const conflictId of deps.conflicts) {
   613→          if (settingsState[conflictId]) {
   614→            const key = [setting.id, conflictId].sort().join("|");
   615→            if (seen.has(key)) continue;
   616→            seen.add(key);
   617→            let conflictName = conflictId;
   618→            for (const s2 of sections) {
   619→              const found = s2.settings.find((s) => s.id === conflictId);
   620→              if (found) { conflictName = found.name; break; }
   621→            }
   622→            allWarnings.push({
   623→              settingId: setting.id,
   624→              settingName: setting.name,
   625→              type: "conflict",
   626→              targetId: conflictId,
   627→              targetName: conflictName,
   628→              message: deps.conflictMessage || `${setting.name} conflicts with ${conflictName}`,
   629→            });
   630→          }
   631→        }
   632→      }
   633→
   634→      if (deps.requires) {
   635→        for (const reqId of deps.requires) {
   636→          if (!settingsState[reqId]) {
   637→            const key = `${setting.id}|req|${reqId}`;
   638→            if (seen.has(key)) continue;
   639→            seen.add(key);
   640→            let reqName = reqId;
   641→            for (const s2 of sections) {
   642→              const found = s2.settings.find((s) => s.id === reqId);
   643→              if (found) { reqName = found.name; break; }
   644→            }
   645→            allWarnings.push({
   646→              settingId: setting.id,
   647→              settingName: setting.name,
   648→              type: "unmet_require",
   649→              targetId: reqId,
   650→              targetName: reqName,
   651→              message: deps.conflictMessage || `${setting.name} requires ${reqName}`,
   652→            });
   653→          }
   654→        }
   655→      }
   656→    }
   657→  }
   658→
   659→  return allWarnings;
   660→}
