    17→# We'll process the file line by line, joining continuation lines (backtick at end)
    18→# and then splitting them back after annotation.
    19→
    20→# First, let's join continuation lines into logical lines
    21→# Each logical line may span multiple physical lines connected by backtick
    22→
    23→class LogicalLine:
    24→    def __init__(self, start_line, end_line, content):
    25→        self.start_line = start_line  # 0-based
    26→        self.end_line = end_line      # 0-based, inclusive
    27→        self.content = content        # joined content (backticks removed)
    28→        self.physical_lines = []      # original physical lines
    29→
    30→# Join continuation lines
    31→logical_lines = []
    32→i = 0
    33→while i < len(lines):
    34→    line = lines[i].rstrip('\n')
    35→    start = i
    36→    content_parts = []
    37→    physical = [lines[i]]
    38→    
    39→    while True:
    40→        stripped = line.rstrip()
    41→        if stripped.endswith('`'):
    42→            # Continuation line - remove backtick and join
    43→            content_parts.append(stripped[:-1])
    44→            i += 1
    45→            if i >= len(lines):
    46→                break
    47→            line = lines[i].rstrip('\n')
    48→            physical.append(lines[i])
    49→        else:
    50→            content_parts.append(stripped)
    51→            break
    52→    
    53→    content = ' '.join(content_parts)
    54→    ll = LogicalLine(start, i, content)
    55→    ll.physical_lines = physical
    56→    logical_lines.append(ll)
    57→    i += 1
    58→
    59→# Now we need to identify Set-Registry and Disable-ScheduledTask calls
    60→# and annotate the ones missing -SafetyLevel
    61→
    62→# Track which section we're in
    63→current_section = ""
    64→
    65→# For each logical line containing Set-Registry or Disable-ScheduledTask without -SafetyLevel,
    66→# we need to generate the annotation and modify the line
    67→
    68→annotated_count = 0
    69→skip_count = 0
    70→
    71→# We'll rebuild the output as physical lines
    72→output_lines = list(lines)  # start with a copy
    73→
    74→# Process in reverse order so line numbers don't shift
    75→# First, collect all the edits we need to make
    76→edits = []  # list of (start_line, end_line, new_content)
    77→
    78→for ll in logical_lines:
    79→    content = ll.content
    80→    
    81→    # Track section
    82→    section_match = re.search(r'\$script:CurrentSection\s*=\s*"(\w+)"', content)
    83→    if section_match:
    84→        current_section = section_match.group(1)
    85→    
    86→    # Check for Set-Registry without -SafetyLevel
    87→    if 'Set-Registry' in content and '-SafetyLevel' not in content:
    88→        # Extract parameters
    89→        path_match = re.search(r'-Path\s+"([^"]+)"', content)
    90→        name_match = re.search(r'-Name\s+"([^"]+)"', content)
    91→        value_match = re.search(r'-Value\s+(\S+)', content)
    92→        
    93→        if not path_match or not name_match:
    94→            continue
    95→        
    96→        path = path_match.group(1)
    97→        name = name_match.group(1)
    98→        value = value_match.group(1) if value_match else ""
    99→        
   100→        # Determine safety level, description, setting ID
   101→        safety, desc, setting_id = classify_set_registry(current_section, path, name, value)
   102→        
   103→        if safety is None:
   104→            skip_count += 1
   105→            continue
   106→        
   107→        # Build annotation string
   108→        annotation = f' -SafetyLevel "{safety}" -Description "{desc}" -SettingID "{setting_id}"'
   109→        
   110→        # Find where to insert the annotation - after the last parameter of the Set-Registry call
   111→        # We need to insert it on the last physical line of the logical line
   112→        # The annotation should go after the last -Type parameter or last parameter
   113→        
   114→        edits.append((ll.start_line, ll.end_line, ll.physical_lines, annotation, 'Set-Registry'))
   115→        annotated_count += 1
   116→    
   117→    # Check for Disable-ScheduledTask without -SafetyLevel
   118→    elif 'Disable-ScheduledTask' in content and '-SafetyLevel' not in content:
   119→        # Skip if this is the built-in Disable-ScheduledTask function definition
   120→        if 'function Disable-ScheduledTask' in content:
   121→            continue
   122→        # Skip if it's the internal call inside the Disable-ScheduledTask function itself
   123→        if ll.start_line >= 813 and ll.start_line <= 886:
   124→            continue
   125→        
   126→        taskpath_match = re.search(r'-TaskPath\s+"([^"]+)"', content)
   127→        taskname_match = re.search(r'-TaskName\s+"([^"]+)"', content)
   128→        
   129→        taskpath = taskpath_match.group(1) if taskpath_match else ""
   130→        taskname = taskname_match.group(1) if taskname_match else ""
   131→        
   132→        safety, desc, setting_id = classify_disable_task(current_section, taskpath, taskname)
   133→        
   134→        if safety is None:
   135→            skip_count += 1
   136→            continue
   137→        
   138→        annotation = f' -SafetyLevel "{safety}" -Description "{desc}" -SettingID "{setting_id}"'
   139→        edits.append((ll.start_line, ll.end_line, ll.physical_lines, annotation, 'Disable-ScheduledTask'))
   140→        annotated_count += 1
   141→
   142→print(f"Found {annotated_count} calls to annotate, {skip_count} skipped")
   143→print(f"Total edits: {len(edits)}")
