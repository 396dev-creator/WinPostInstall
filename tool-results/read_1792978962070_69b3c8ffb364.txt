   155→# Self-elevate: If not running as Administrator, re-launch in an elevated window
   156→# This allows the user to simply double-click the script
   157→#
   158→# CRITICAL: When invoked via `powershell -Command "& 'script.ps1' ..."`, $PSCommandPath

   178→            # $PSCommandPath is empty  -  script is running from memory (irm | iex or -Command).

   185→            Write-Host "  Elevated window opened (re-downloaded from GitHub). This window can be closed." -ForegroundColor Green

   203→function Set-DefenderExclusions {
   204→    <#
   205→    .DESCRIPTION
   206→        Adds this script and its working directory to Windows Defender exclusions.
   207→        Prevents the script from being flagged as malware during execution.
   208→        This is safe: exclusions are path-specific and only protect this tool.
   209→    #>
   210→    Write-Host "" -ForegroundColor Cyan
   211→    Write-Host "  [AV SHIELD] Windows Defender Self-Exclusion" -ForegroundColor Cyan
   212→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   213→    Write-Host ""
   214→
   215→    $scriptPath = $PSCommandPath
   216→    if (-not $scriptPath) { $scriptPath = $MyInvocation.PSCommandPath }
   217→
   218→    $exclusionAdded = $false
   219→
   220→    # Check if the exclusion already exists
   221→    try {
   222→        $existingExclusions = (Get-MpPreference -ErrorAction SilentlyContinue).ExclusionPath
   223→        if ($existingExclusions -and $scriptPath -in $existingExclusions) {
   224→            Write-Step "Defender exclusion already exists for this script."
   225→            return
   226→        }
   227→    } catch {
   228→        Write-Warn "Could not read current Defender preferences: $_"
   229→    }
   230→
   231→    # Exclude the script file itself
   232→    Write-Step "Adding script to Defender exclusions..."
   233→    try {
   234→        Add-MpPreference -ExclusionPath $scriptPath -ErrorAction Stop
   235→        Write-Success "Excluded: $scriptPath"
   236→        $exclusionAdded = $true
   237→    } catch {
   238→        Write-Warn "Failed to exclude script path: $_"
   239→    }
   240→
   241→    # Exclude the working directory (where the script lives)
   242→    $scriptDir = Split-Path $scriptPath -Parent
   243→    if ($scriptDir -and (Test-Path $scriptDir)) {
   244→        try {
   245→            Add-MpPreference -ExclusionPath $scriptDir -ErrorAction Stop
   246→            Write-Success "Excluded directory: $scriptDir"
   247→            $exclusionAdded = $true
   248→        } catch {
   249→            Write-Warn "Failed to exclude directory: $_"
   250→        }
   251→    }
   252→
   253→    # Exclude the log directory (C:\Temp)
   254→    $logDir = "$env:SYSTEMDRIVE\Temp"
   255→    try {
   256→        Add-MpPreference -ExclusionPath $logDir -ErrorAction Stop
   257→        Write-Success "Excluded log directory: $logDir"
   258→        $exclusionAdded = $true
   259→    } catch {
   260→        Write-Warn "Failed to exclude log directory: $_"
   261→    }
   262→
   263→    # Exclude common technician tool folders
   264→    $toolPaths = @("C:\Tools", "C:\Scripts", "C:\TechnicianTools")
   265→    foreach ($path in $toolPaths) {
   266→        if (Test-Path $path) {
   267→            try {
   268→                Add-MpPreference -ExclusionPath $path -ErrorAction Stop
   269→                Write-Success "Excluded: $path"
   270→            } catch { }
   271→        }
   272→    }
   273→
   274→    if ($exclusionAdded) {
   275→        Write-Success "Defender exclusions configured. This script is now trusted."
   276→    }
   277→
   278→    Write-Host ""
   279→    Write-Host "  NOTE: This only affects Windows Defender. For third-party AV," -ForegroundColor DarkGray
   280→    Write-Host "  consider purchasing a Code Signing Certificate for full compatibility." -ForegroundColor DarkGray
   281→    Write-Host ""
   282→}

   292→        Write-Host "  Then re-run this script." -ForegroundColor Yellow

   298→

   312→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray

   314→    try {
   315→        if (-not (Test-Path $script:BackupDir)) {
   316→            New-Item -Path $script:BackupDir -ItemType Directory -Force | Out-Null
   317→        }

   319→        $regExports = @(
   320→            @{ Hive = "HKLM\SOFTWARE\Policies"; File = "Policies.reg" },
   321→            @{ Hive = "HKLM\SOFTWARE\Microsoft\Windows\CurrentVersion\Policies"; File = "WinPolicies.reg" },
   322→            @{ Hive = "HKCU\SOFTWARE\Microsoft\Windows\CurrentVersion\Explorer"; File = "Explorer.reg" },
   323→            @{ Hive = "HKCU\SOFTWARE\Microsoft\Windows\CurrentVersion\GameDVR"; File = "GameDVR.reg" }

   325→        foreach ($regExport in $regExports) {
   326→            $exportPath = Join-Path $script:BackupDir $regExport.File
   327→            try {
   328→                reg export $regExport.Hive $exportPath /y 2>$null | Out-Null
   329→                Write-Success "Exported: $($regExport.Hive)"
   330→            } catch {
   331→                Write-Warn "Could not export $($regExport.Hive): $_"
   332→            }
   333→        }
   334→        Write-Success "Registry backup saved to: $script:BackupDir"
   335→    } catch {
   336→        Write-Warn "Registry backup failed: $_"
   337→    }
   338→    Write-Host ""

   353→        }
   354→    } catch { }
   355→    if (-not $script:IsLaptop) {
   356→        try {
   357→            $script:IsLaptop = (Get-CimInstance -ClassName Win32_Battery -ErrorAction SilentlyContinue).Count -gt 0
   358→        } catch { }
   359→    }
   360→
   361→    Write-Host ""
   362→    Write-Host "================================================================" -ForegroundColor Cyan
   363→    Write-Host "  Windows Post-Installation Optimizer v$ScriptVersion" -ForegroundColor Cyan
   364→    Write-Host "  Target: $script:OSCaption (Build $script:OSBuild)" -ForegroundColor Cyan
   365→    Write-Host "  Computer: $script:LocalComputerName" -ForegroundColor Cyan
   366→    Write-Host "  Chassis: $(if ($script:IsLaptop) { 'Laptop' } else { 'Desktop' })" -ForegroundColor Cyan
   367→    Write-Host "  Log file: $LogFile" -ForegroundColor DarkGray
   368→    Write-Host "================================================================" -ForegroundColor Cyan
   369→    Write-Host ""
   370→
   371→    # Run pre-flight checks
   372→    Test-Prerequisites
   373→}

   375→function Test-Prerequisites {
   376→    <#
   377→    .DESCRIPTION
   378→        Pre-flight checks: internet connectivity, disk space, pending reboot, battery.
   379→        Warns on issues but does not block execution.
   380→    #>
   381→    Write-Host "" -ForegroundColor Cyan
   382→    Write-Host "  [PRE-FLIGHT] System Prerequisites Check" -ForegroundColor Cyan
   383→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   384→    Write-Host ""
   385→
   386→    $checksPassed = 0; $checksWarned = 0
   387→
   388→    # 1. Internet connectivity
   389→    Write-Step "Checking internet connectivity..."
   390→    try {
   391→        $netTest = Test-Connection -ComputerName 8.8.8.8 -Count 1 -Quiet -ErrorAction Stop
   392→        if ($netTest) {
   393→            Write-Success "Internet connectivity: OK"
   394→            $checksPassed++
   395→        } else {
   396→            Write-Warn "Internet connectivity: OFFLINE (some features may fail)"
   397→            $checksWarned++
   398→        }
   399→    } catch {
   400→        Write-Warn "Internet connectivity: Could not check (some features may fail)"
   401→        $checksWarned++
   402→    }
   403→
   404→    # 2. Free disk space
   405→    Write-Step "Checking free disk space..."
   406→    try {
   407→        $sysDrive = Get-PSDrive -Name ($env:SYSTEMDRIVE -replace ':','') -ErrorAction Stop
   408→        $freeGB = [math]::Round($sysDrive.Free / 1GB, 2)
   409→        if ($freeGB -gt 10) {
   410→            Write-Success "Free disk space: $freeGB GB (OK)"
   411→            $checksPassed++
   412→        } else {
   413→            Write-Warn "Free disk space: $freeGB GB (LOW - recommend >10 GB)"
   414→            $checksWarned++
   415→        }
   416→    } catch {
   417→        Write-Warn "Could not check disk space: $_"
   418→        $checksWarned++
   419→    }
   420→
   421→    # 3. Pending reboot
   422→    Write-Step "Checking for pending reboot..."
   423→    $rebootPending = $false
   424→    $cbsPath = "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Component Based Servicing\RebootPending"
   425→    $wuPath = "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\WindowsUpdate\Auto Update\RebootRequired"
   426→    if (Test-Path $cbsPath -ErrorAction SilentlyContinue) { $rebootPending = $true }
   427→    if (Test-Path $wuPath -ErrorAction SilentlyContinue) { $rebootPending = $true }
   428→    if ($rebootPending) {
   429→        Write-Warn "Pending reboot detected! A restart is recommended before running optimizations."
   430→        $checksWarned++
   431→    } else {
   432→        Write-Success "No pending reboot."
   433→        $checksPassed++
   434→    }
   435→
   436→    # 4. Battery check (if laptop)
   437→    if ($script:IsLaptop) {
   438→        Write-Step "Checking battery level..."
   439→        try {
   440→            $battery = Get-CimInstance -ClassName Win32_Battery -ErrorAction Stop
   441→            $onAC = $battery.BatteryStatus -eq 2 -or $battery.BatteryStatus -eq 6 -or $battery.BatteryStatus -eq 7 -or $battery.BatteryStatus -eq 8
   442→            if ($onAC) {
   443→                Write-Success "On AC power."
   444→                $checksPassed++
   445→            } elseif ($battery.EstimatedChargeRemaining -gt 50) {
   446→                Write-Success "Battery level: $($battery.EstimatedChargeRemaining)% (OK)"
   447→                $checksPassed++
   448→            } else {
   449→                Write-Warn "Battery level: $($battery.EstimatedChargeRemaining)% - recommend AC power or >50% charge."
   450→                $checksWarned++
   451→            }
   452→        } catch {
   453→            Write-Warn "Could not check battery: $_"
   454→            $checksWarned++
   455→        }
   456→    }
   457→
   458→    Write-Host ""
   459→    Write-Host "  Pre-flight summary: $checksPassed passed, $checksWarned warnings" -ForegroundColor $(if ($checksWarned -gt 0) { 'Yellow' } else { 'Green' })
   460→    Write-Host ""
   461→}

   463→function Get-BootDriveType {
   464→    <#
   465→    .DESCRIPTION
   466→        Detects whether the boot drive is an SSD or HDD.
   467→        Returns "SSD" or "HDD".
   468→    #>
   469→    $bootType = "HDD"  # Default assumption
   470→    try {
   471→        # Method 1: Get-PhysicalDisk (most reliable on Win8+)
   472→        $physicalDisk = Get-PhysicalDisk -ErrorAction SilentlyContinue | Where-Object DeviceId -eq 0
   473→        if ($physicalDisk) {
   474→            if ($physicalDisk.MediaType -eq "SSD") { return "SSD" }
   475→            if ($physicalDisk.MediaType -eq "HDD") { return "HDD" }
   476→        }
   477→    } catch { }
   478→
   479→    try {
   480→        # Method 2: Get-Disk BusType fallback
   481→        $bootDisk = Get-Disk -ErrorAction SilentlyContinue | Where-Object IsBoot -eq $true
   482→        if ($bootDisk) {

   497→            # If scheduled defrag is disabled, likely SSD
   498→            # This is a weak heuristic, don't rely on it alone

   501→
   502→    return $bootType

   521→function Enable-SystemRestore {
   522→    <#
   523→    .DESCRIPTION
   524→        Ensures System Restore is enabled and configured on the system drive.
   525→        Enables the service if disabled, allocates disk space, and verifies
   526→        the feature is operational before returning.
   527→    #>
   528→    Write-Host "" -ForegroundColor Cyan
   529→    Write-Host "  [SAFETY] System Restore Verification" -ForegroundColor Cyan
   530→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   531→    Write-Host ""
   532→
   533→    $systemDrive = $env:SYSTEMDRIVE
   534→    $restoreEnabled = $false
   535→
   536→    # Step 1: Check if System Restore is currently enabled on the system drive
   537→    Write-Step "Checking System Restore status on $systemDrive..."
   538→    try {
   539→        $srStatus = Get-ComputerInfo -Property "SystemRestoreStatus" -ErrorAction SilentlyContinue
   540→        # Also check via WMI which is more reliable
   541→        $srInfo = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction SilentlyContinue
   542→        if ($srInfo) {
   543→            foreach ($sr in $srInfo) {
   544→                if ($sr.RPSequence -gt 0) {
   545→                    $restoreEnabled = $true
   546→                    break
   547→                }
   548→            }
   549→        }
   550→    } catch {
   551→        # WMI query may fail if System Restore is completely disabled
   552→        $restoreEnabled = $false
   553→    }
   554→
   555→    # Alternative check: try to list existing restore points
   556→    if (-not $restoreEnabled) {
   557→        try {
   558→            $existingPoints = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction Stop
   559→            if ($existingPoints -and $existingPoints.Count -ge 0) {
   560→                $restoreEnabled = $true
   561→            }
   562→        } catch {
   563→            $restoreEnabled = $false
   564→        }
   565→    }
   566→
   567→    # Step 2: Enable System Restore if not currently enabled
   568→    if (-not $restoreEnabled) {
   569→        Write-Warn "System Restore is NOT enabled on $systemDrive."
   570→        Write-Step "Enabling System Restore on $systemDrive..."
   571→
   572→        try {
   573→            # Enable System Restore via registry (most reliable method)
   574→            $srPolicyPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore"
   575→            if (-not (Test-Path $srPolicyPath)) {
   576→                New-Item -Path $srPolicyPath -Force -ErrorAction SilentlyContinue | Out-Null
   577→            }
   578→            # DisableSR = 0 means System Restore is ENABLED
   579→            Set-ItemProperty -Path $srPolicyPath -Name "DisableSR" -Value 0 -Type DWord -Force -ErrorAction Stop
   580→
   581→            # Also enable via PowerShell cmdlet (Windows 10/11)
   582→            try {
   583→                Enable-ComputerRestore -Drive $systemDrive -ErrorAction Stop
   584→            } catch {
   585→                # Fallback: use vssadmin to enable shadow copies
   586→                $vssResult = vssadmin list shadows 2>&1
   587→                if ($LASTEXITCODE -ne 0) {
   588→                    # Try creating a shadow copy to force-enable VSS
   589→                    vssadmin create shadow /For=$systemDrive 2>&1 | Out-Null
   590→                }
   591→            }
   592→
   593→            # Start the System Restore service if not running
   594→            $srService = Get-Service -Name "sr" -ErrorAction SilentlyContinue
   595→            if ($srService -and $srService.Status -ne 'Running') {
   596→                try {
   597→                    Set-Service -Name "sr" -StartupType Manual -ErrorAction Stop
   598→                    Start-Service -Name "sr" -ErrorAction Stop
   599→                } catch {
   600→                    Write-Warn "Could not start System Restore service: $_"
   601→                }
   602→            }
   603→
   604→            # Also ensure VSS (Volume Shadow Copy) service is running (required for System Restore)
   605→            $vssService = Get-Service -Name "VSS" -ErrorAction SilentlyContinue
   606→            if ($vssService -and $vssService.Status -ne 'Running') {
   607→                try {
   608→                    Set-Service -Name "VSS" -StartupType Manual -ErrorAction Stop
   609→                    Start-Service -Name "VSS" -ErrorAction Stop
   610→                    Write-Success "VSS service started."
   611→                } catch {
   612→                    Write-Warn "Could not start VSS service: $_"
   613→                }
   614→            }
   615→
   616→            Start-Sleep -Seconds 2
   617→            Write-Success "System Restore has been ENABLED on $systemDrive."
   618→        } catch {
   619→            Write-Err "Failed to enable System Restore: $_"
   620→            Write-Warn "Continuing without restore point. Manual enable: System Properties > System Protection > Configure > Turn on"
   621→        }
   622→    } else {
   623→        Write-Success "System Restore is already enabled on $systemDrive."
   624→    }
   625→
   626→    # Step 3: Configure disk space allocation for restore points
   627→    Write-Step "Configuring System Restore disk space..."
   628→    try {
   629→        # Set maximum disk space usage to 5% (reasonable for most systems)
   630→        # This is done via registry: RPSessionInterval and DiskPercent
   631→        $srCfgPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore\Cfg"
   632→        if (-not (Test-Path $srCfgPath)) {
   633→            New-Item -Path $srCfgPath -Force -ErrorAction SilentlyContinue | Out-Null
   634→        }
   635→        # DiskPercent = 5 means 5% of disk (0 = unlimited, we use 5% as safe default)
   636→        Set-ItemProperty -Path $srCfgPath -Name "DiskPercent" -Value 5 -Type DWord -Force -ErrorAction SilentlyContinue
   637→        # RPSessionInterval = 0 means create restore points at any interval
   638→        Set-ItemProperty -Path $srCfgPath -Name "RPSessionInterval" -Value 0 -Type DWord -Force -ErrorAction SilentlyContinue
   639→        Write-Success "Restore disk space set to 5% of drive."
   640→    } catch {
   641→        Write-Warn "Could not configure restore disk space: $_"
   642→    }
   643→
   644→    Write-Host ""
   645→}

   647→function New-SystemRestorePoint {
   648→    <#
   649→    .DESCRIPTION
   650→        Creates a System Restore point before any optimization changes are made.
   651→        This provides a safety net so the user can roll back all changes.
   652→    #>
   653→    Write-Host "" -ForegroundColor Cyan
   654→    Write-Host "  [SAFETY] Creating System Restore Point" -ForegroundColor Cyan
   655→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   656→    Write-Host ""
   657→
   658→    $restorePointCreated = $false
   659→    $description = "Pre-Optimization Restore Point - WinPostInstall v$ScriptVersion - $(Get-Date -Format 'yyyy-MM-dd HH:mm')"
   660→
   661→    # Method 1: Use Checkpoint-Computer (PowerShell cmdlet, Windows 8+)
   662→    Write-Step "Creating restore point: 'Pre-Optimization'..."
   663→    try {
   664→        Checkpoint-Computer -Description $description -RestorePointType "MODIFY_SETTINGS" -ErrorAction Stop
   665→        $restorePointCreated = $true
   666→        Write-Success "Restore point created successfully via Checkpoint-Computer."
   667→    } catch {
   668→        Write-Warn "Checkpoint-Computer failed: $_"
   669→    }
   670→
   671→    # Method 2: Fallback using WMI if Method 1 failed
   672→    if (-not $restorePointCreated) {
   673→        Write-Step "Trying WMI method to create restore point..."
   674→        try {
   675→            $wmiSR = [System.Windows.Forms.SystemRestore]
   676→            # WMI approach via Invoke-CimMethod
   677→            $result = Invoke-CimMethod -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" `
   678→                -MethodName "CreateRestorePoint" `
   679→                -Arguments @{
   680→                    Description         = $description
   681→                    RestorePointType    = 12  # MODIFY_SETTINGS
   682→                    EventType           = 100  # BEGIN_NESTED_SYSTEM_CHANGE
   683→                } -ErrorAction Stop
   684→            if ($result -and $result.ReturnValue -eq 0) {
   685→                $restorePointCreated = $true
   686→                Write-Success "Restore point created via WMI."
   687→            } else {
   688→                Write-Warn "WMI returned non-zero: $result"
   689→            }
   690→        } catch {
   691→            Write-Warn "WMI method failed: $_"
   692→        }
   693→    }
   694→
   695→    # Method 3: Final fallback via srclient.dll COM automation
   696→    if (-not $restorePointCreated) {
   697→        Write-Step "Trying COM automation (srclient.dll)..."
   698→        try {
   699→            $srClient = New-Object -ComObject "SRClient.RestorePoint" -ErrorAction Stop
   700→            # 7 = MODIFY_SETTINGS, 100 = BEGIN_NESTED_SYSTEM_CHANGE
   701→            $srClient.CreateRestorePoint($description, 7, 100)
   702→            $restorePointCreated = $true
   703→            Write-Success "Restore point created via COM automation."
   704→        } catch {
   705→            Write-Warn "COM method failed: $_"
   706→        }
   707→    }
   708→
   709→    if ($restorePointCreated) {
   710→        Write-Host ""
   711→        Write-Host "  ====================================================================" -ForegroundColor Green
   712→        Write-Host "  RESTORE POINT CREATED: 'Pre-Optimization'" -ForegroundColor Green
   713→        Write-Host "  You can roll back all changes via:" -ForegroundColor White
   714→        Write-Host "    System Properties > System Protection > System Restore" -ForegroundColor White
   715→        Write-Host "  ====================================================================" -ForegroundColor Green
   716→        Write-Host ""
   717→    } else {
   718→        Write-Host ""
   719→        Write-Warn "Could NOT create restore point automatically."
   720→        Write-Host "  Manual option: System Properties > System Protection > Create" -ForegroundColor Yellow
   721→        Write-Host "  Continuing without restore point backup." -ForegroundColor Yellow
   722→        Write-Host ""
   723→        if (-not $Silent) {
   724→            $continue = Read-Host "  Continue without restore point? (y/N)"
   725→            if ($continue -ne 'y' -and $continue -ne 'Y') {
   726→                Write-Host "  Aborted by user." -ForegroundColor Yellow
   727→                exit 0
   728→            }
   729→        }
   730→    }
   731→}

   733→function Set-Registry {
   734→    [CmdletBinding()]
   735→    param(
   736→        [string]$Path,
   737→        [string]$Name,
   738→        $Value,
   739→        [string]$Type = "DWord",
   740→        [ValidateSet("Safe","Moderate","Risky")]
   741→        [string]$SafetyLevel = "Safe",
   742→        [string]$Description = "",
   743→        [string]$SettingID = ""
   744→    )
   745→
   746→    # Generate a setting ID if not provided
   747→    if ([string]::IsNullOrEmpty($SettingID)) {
   748→        $SettingID = "$($Path.Replace('\','_').Replace(':','').Replace(' ',''))_$Name"
   749→    }
   750→
   751→    # Register in setting catalog for interactive menu
   752→    if (-not [string]::IsNullOrEmpty($Description)) {
   753→        $script:SettingCatalog[$SettingID] = @{
   754→            Section  = $script:CurrentSection
   755→            Safety   = $SafetyLevel
   756→            Desc     = $Description
   757→            Action   = "$Path\$Name = $Value"
   758→            Type     = "Registry"
   759→        }
   760→    }
   761→
   762→    # Collect risky descriptions for menu warnings
   763→    if ($SafetyLevel -eq "Risky" -and -not [string]::IsNullOrEmpty($Description)) {
   764→        $script:RiskyDescriptions += "$SettingID : $Description"
   765→    }
   766→
   767→    # Check if user has toggled this setting OFF via interactive menu
   768→    if ($script:SettingOverrides.ContainsKey($SettingID) -and $script:SettingOverrides[$SettingID]) {
   769→        Write-Host "  SKIPPED (user toggle): $Description" -ForegroundColor DarkGray
   770→        $script:TotalSkipped++
   771→        return
   772→    }
   773→
   774→    # Safety gate: Skip Risky settings unless explicitly allowed
   775→    if ($SafetyLevel -eq "Risky" -and -not $script:AllowRisky) {
   776→        Write-Host "  SKIPPED (risky): $Description  -  use -AllowRisky or enable in menu" -ForegroundColor Yellow
   777→        $script:TotalSkipped++
   778→        return
   779→    }
   780→
   781→    # Warn for Moderate settings
   782→    if ($SafetyLevel -eq "Moderate") {
   783→        Write-Host "  [MODERATE] $Description" -ForegroundColor DarkYellow
   784→    }
   785→
   786→    try {
   787→        if (-not (Test-Path $Path)) {
   788→            New-Item -Path $Path -Force -ErrorAction SilentlyContinue | Out-Null
   789→        }
   790→        # Idempotency: Check if the value already matches what we want to set
   791→        $currentValue = $null
   792→        try {
   793→            $currentValue = (Get-ItemProperty -Path $Path -Name $Name -ErrorAction Stop).$Name
   794→        } catch {
   795→            # Property doesn't exist yet  -  need to create it
   796→        }
   797→
   798→        # Compare current vs desired value
   799→        $valueMatches = $false
   800→        if ($null -ne $currentValue) {
   801→            if ($Type -eq "DWord" -or $Type -eq "QWord") {
   802→                # Numeric comparison - use [long] to handle values > Int32.MaxValue
   803→                # (e.g. NetworkThrottlingIndex = 0xFFFFFFFF = 4294967295)
   804→                $valueMatches = ([long]$currentValue -eq [long]$Value)
   805→            } elseif ($Type -eq "String" -or $Type -eq "ExpandString") {
   806→                $valueMatches = ([string]$currentValue -eq [string]$Value)
   807→            } else {
   808→                $valueMatches = ($currentValue -eq $Value)
   809→            }
   810→        }
   811→
   812→        if ($valueMatches) {
   813→            Write-Step "Already set: $Path\$Name = $Value"
   814→            $script:TotalSkipped++
   815→            return
   816→        }
   817→
   818→        # Capture old value for undo script before writing
   819→        if ($null -ne $currentValue) {
   820→            $oldVal = $currentValue.ToString()
   821→            $null = $script:UndoCommands.Add("Set-ItemProperty -Path '$Path' -Name '$Name' -Value '$oldVal' -Type '$Type' -Force")
   822→            $null = $script:UndoDescriptions.Add("Restore registry: $Path\$Name = $oldVal")
   823→        } else {
   824→            $null = $script:UndoCommands.Add("Remove-ItemProperty -Path '$Path' -Name '$Name' -Force -ErrorAction SilentlyContinue")
   825→            $null = $script:UndoDescriptions.Add("Remove registry key that was added: $Path\$Name")
   826→        }
   827→
   828→        # Set-ItemProperty cannot handle DWord values > Int32.MaxValue (2147483647).
   829→        # For such values (e.g. NetworkThrottlingIndex = 0xFFFFFFFF), use reg.exe instead.
   830→        if ($Type -eq "DWord" -and [long]$Value -gt [int]::MaxValue) {
   831→            # Convert registry path to HKLM/HKCU short form for reg.exe
   832→            $regKey = $Path -replace '^HKLM:\\', 'HKLM\' -replace '^HKCU:\\', 'HKCU\' `
   833→                                -replace '^HKEY_LOCAL_MACHINE\\', 'HKLM\' -replace '^HKEY_CURRENT_USER\\', 'HKCU\' `
   834→                                -replace '\\', '\'
   835→            $hexVal = '0x{0:X}' -f [long]$Value
   836→            $regArgs = @('add', $regKey, '/v', $Name, '/t', 'REG_DWORD', '/d', $hexVal, '/f')
   837→            $regResult = & reg.exe @regArgs 2>&1
   838→            if ($LASTEXITCODE -ne 0) {
   839→                throw "reg.exe failed: $regResult"
   840→            }
   841→        } elseif ($Type -eq "QWord" -and [long]$Value -gt [long]::MaxValue) {
   842→            # QWord values exceeding [long]::MaxValue are extremely rare; fallback to reg.exe
   843→            $regKey = $Path -replace '^HKLM:\\', 'HKLM\' -replace '^HKCU:\\', 'HKCU\' `
   844→                                -replace '^HKEY_LOCAL_MACHINE\\', 'HKLM\' -replace '^HKEY_CURRENT_USER\\', 'HKCU\' `
   845→                                -replace '\\', '\'
   846→            $hexVal = '0x{0:X}' -f [long]$Value
   847→            $regArgs = @('add', $regKey, '/v', $Name, '/t', 'REG_QWORD', '/d', $hexVal, '/f')
   848→            $regResult = & reg.exe @regArgs 2>&1
   849→            if ($LASTEXITCODE -ne 0) {
   850→                throw "reg.exe failed: $regResult"
   851→            }
   852→        } else {
   853→            Set-ItemProperty -Path $Path -Name $Name -Value $Value -Type $Type -Force -ErrorAction Stop
   854→        }
   855→        Write-Success "Registry: $Path\$Name = $Value"
   856→        $script:TotalChanged++
   857→    }
   858→    catch {
   859→        Write-Err "Failed to set $Path\$Name : $_"
   860→        $script:TotalFailed++
   861→    }
   862→}

   864→function Remove-Registry {
   865→    [CmdletBinding()]
   866→    param(
   867→        [string]$Path,
   868→        [string]$Name
   869→    )
   870→    try {
   871→        if (Test-Path $Path) {
   872→            # Idempotency: Check if property exists before removing
   873→            $existing = $null
   874→            try { $existing = (Get-ItemProperty -Path $Path -Name $Name -ErrorAction Stop).$Name } catch { }
   875→            if ($null -ne $existing) {
   876→                Remove-ItemProperty -Path $Path -Name $Name -Force -ErrorAction Stop 2>$null
   877→                Write-Success "Registry removed: $Path\$Name"
   878→            } else {
   879→                Write-Step "Already removed: $Path\$Name"
   880→            }
   881→        }
   882→    }
   883→    catch {
   884→        Write-Warn "Could not remove $Path\$Name : $_"
   885→    }

   913→    }
   914→
   915→    # Collect risky descriptions for menu warnings

   917→        $script:RiskyDescriptions += "$SettingID : $Description"
   918→    }

   923→        $script:TotalSkipped++
   924→        return
   925→    }
   926→
   927→    # Safety gate: Skip Risky settings unless explicitly allowed
   928→    if ($SafetyLevel -eq "Risky" -and -not $script:AllowRisky) {
   929→        Write-Host "  SKIPPED (risky): $Description  -  use -AllowRisky or enable in menu" -ForegroundColor Yellow
   930→        $script:TotalSkipped++
   931→        return
   932→    }
   933→
   934→    # Warn for Moderate settings
   935→    if ($SafetyLevel -eq "Moderate") {
   936→        Write-Host "  [MODERATE] $Description" -ForegroundColor DarkYellow
   937→    }
   938→
   939→    try {
   940→        $task = Get-ScheduledTask -TaskPath $TaskPath -TaskName $TaskName -ErrorAction SilentlyContinue
   941→        if ($task) {
   942→            if ($task.State -eq 'Disabled') {
   943→                Write-Step "Already disabled: $TaskPath$TaskName"
   944→                $script:TotalSkipped++
   945→            } else {
   946→                # Safety: Check for dependent services/tasks before disabling
   947→                $deps = $task.DependentOn
   948→                if ($deps) {
   949→                    Write-Host "  [DEPENDENCY] Task $TaskName has dependents: $($deps.TaskName -join ', ')" -ForegroundColor DarkYellow
   950→                }
   951→                $task | Disable-ScheduledTask -ErrorAction Stop
   952→                Write-Success "Task disabled: $TaskPath$TaskName"
   953→                $script:TotalChanged++
   954→                $null = $script:UndoCommands.Add("Enable-ScheduledTask -TaskPath '$TaskPath' -TaskName '$TaskName' -ErrorAction SilentlyContinue")
   955→                $null = $script:UndoDescriptions.Add("Re-enable scheduled task: $TaskPath$TaskName")
   956→            }
   957→        }
   958→    }
   959→    catch {
   960→        Write-Warn "Could not disable task $TaskPath$TaskName : $_"
   961→    }
   962→}

  1705→        @{ Name = "WbioSrvc";              Desc = "Windows Biometric Service";          Safety = "Risky";   Reason = "Disables Biometrics  -  WARNING: breaks Windows Hello fingerprint/face login" },
  1706→        @{ Name = "OneSyncSvc";            Desc = "OneSync Sync Host";                 Safety = "Moderate"; Reason = "Disables OneSync  -  may affect Mail/Calendar sync" },

  1750→        -Name "StartupDelayInMSec" -Value 0 -Type DWord -SafetyLevel "Safe" -Description "Disables startup delay  -  speeds up boot by removing artificial delay" -SettingID "ServiceStartup_NoStartupDelay"

  1762→function Invoke-PerformanceOptimization {

  1765→    Write-Host "--- PERFORMANCE and POWER SETTINGS ---" -ForegroundColor Magenta
  1766→    Write-Host ""
  1767→
  1768→    # === HARDWARE-AWARE: SSD vs HDD ===
  1769→    $script:BootDriveType = Get-BootDriveType
  1770→    Write-Step "Boot drive type detected: $($script:BootDriveType)"
  1771→
  1772→    # === HARDWARE-AWARE: Laptop vs Desktop ===
  1773→    if ($script:IsLaptop) {
  1774→        Write-Step "Chassis type: Laptop  -  keeping Balanced power plan."
  1775→    } else {
  1776→        Write-Step "Chassis type: Desktop  -  setting High Performance power plan."
  1777→    }
  1778→
  1779→    # === HARDWARE-AWARE: RAM detection ===
  1780→    $totalRAM = 0
  1781→    try {
  1782→        $totalRAM = [math]::Round((Get-CimInstance Win32_ComputerSystem).TotalPhysicalMemory / 1GB, 2)
  1783→        Write-Step "Total RAM detected: $totalRAM GB"
  1784→    } catch {
  1785→        Write-Warn "Could not detect RAM: $_"
  1786→    }
  1787→
  1788→    # 1. Disable Hibernation
  1789→    Write-Step "Disabling Hibernation..."
  1790→    try {
  1791→        powercfg.exe /hibernate off 2>$null
  1792→        Write-Success "Hibernation disabled. hiberfil.sys removed."
  1793→    } catch { Write-Warn "Hibernation disable failed: $_" }
  1794→
  1795→    # 2. Disable Fast Startup (causes partial shutdowns, driver issues)

  1798→        -Name "HiberbootEnabled" -Value 0 -Type DWord `
  1799→        -SafetyLevel "Safe" -Description "Disables Fast Startup  -  prevents partial shutdown/driver issues" -SettingID "Perf_NoFastStartup"
  1800→
  1801→    # 3. Disable Modern Standby Networking (battery drain + security)
  1802→    Write-Step "Disabling Modern Standby network connectivity..."
  1803→    Set-Registry -Path "HKLM:\SYSTEM\CurrentControlSet\Control\Power" `
  1804→        -Name "ModernSleep" -Value 0 -Type DWord `
  1805→        -SafetyLevel "Safe" -Description "Disables Modern Standby network  -  prevents battery drain during sleep" -SettingID "Perf_NoModernSleep"
  1806→
  1807→    # 4. Set power plan based on chassis type
  1808→    if ($script:IsLaptop) {
  1809→        # Laptop: keep Balanced, but warn about High Performance option
  1810→        Write-Host "  [MODERATE] High Performance on laptop  -  WARNING: drains battery. Keeping Balanced." -ForegroundColor DarkYellow
  1811→        Write-Step "Keeping Balanced power plan (laptop)."
  1812→    } else {
  1813→        # Desktop: set High Performance, disable lid close action
  1814→        Write-Step "Setting High Performance power plan (desktop)..."
  1815→        try {
  1816→            $highPerf = powercfg.exe /list | Select-String -Pattern "High performance|高性能"

  1818→                $guid = ($highPerf.ToString() -split '\s+')[3]
  1819→                powercfg.exe /setactive $guid 2>$null
  1820→                Write-Success "Power plan set to High Performance ($guid)"
  1821→            } else {
  1822→                powercfg.exe /duplicatecheme 8c5e7fda-e8bf-4a96-9a85-a6e23a8c635c 2>$null | Out-Null
  1823→                Start-Sleep -Seconds 1
  1824→                $highPerf = powercfg.exe /list | Select-String -Pattern "High performance|高性能"
  1825→                if ($highPerf) {
  1826→                    $guid = ($highPerf.ToString() -split '\s+')[3]
  1827→                    powercfg.exe /setactive $guid 2>$null
  1828→                    Write-Success "High Performance plan created and activated ($guid)"
  1829→                } else { Write-Warn "Could not find or create High Performance plan." }
  1830→            }
  1831→        } catch { Write-Warn "Power plan change failed: $_" }
  1832→
  1833→        # Disable lid close action (desktop doesn't need it)
  1834→        Write-Step "Disabling lid close action (desktop)..."
  1835→        try {
  1836→            powercfg.exe /setacvalueindex SCHEME_CURRENT SUB_BUTTONS LIDACTION 0 2>$null
  1837→            powercfg.exe /setdcvalueindex SCHEME_CURRENT SUB_BUTTONS LIDACTION 0 2>$null
  1838→            powercfg.exe /setactive SCHEME_CURRENT 2>$null
  1839→            Write-Success "Lid close action set to 'Do nothing' (desktop)."
  1840→        } catch { Write-Warn "Could not change lid close action: $_" }
  1841→    }
  1842→
  1843→    # 5. Set visual effects to Best Performance
  1844→    Write-Step "Setting visual effects to 'Best Performance'..."
  1845→    Set-Registry -Path "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Explorer\VisualEffects" `
  1846→        -Name "VisualFXSetting" -Value 2 -Type DWord `
  1847→        -SafetyLevel "Safe" -Description "Sets visual effects to Best Performance mode" -SettingID "Perf_VisualFX"

  1857→    }
  1858→    Set-Registry -Path "HKCU:\SOFTWARE\Microsoft\Windows\DWM" `
  1859→        -Name "EnableAeroPeek" -Value 0 -Type DWord -SafetyLevel "Safe" -Description "Disables Aero Peek  -  removes desktop preview on taskbar hover" -SettingID "Performance_NoAeroPeek"

  1861→    # 6. SysMain/Prefetch based on drive type

  1864→        try {
  1865→            $sysMain = Get-Service -Name "SysMain" -ErrorAction SilentlyContinue
  1866→            if ($sysMain -and $sysMain.StartType -ne 'Disabled') {

  1868→                Set-Service -Name "SysMain" -StartupType Disabled -ErrorAction Stop

  1873→        } catch { Write-Warn "Could not disable SysMain: $_" }

  1876→        Set-Registry -Path "HKLM:\SYSTEM\CurrentControlSet\Control\Session Manager\Memory Management\PrefetchParameters" `
  1877→            -Name "EnablePrefetcher" -Value 0 -Type DWord `
  1878→            -SafetyLevel "Safe" -Description "Disables Prefetch on SSD  -  safe, reduces unnecessary disk I/O" -SettingID "Perf_NoPrefetch_SSD"
  1879→        Set-Registry -Path "HKLM:\SYSTEM\CurrentControlSet\Control\Session Manager\Memory Management\PrefetchParameters" `
  1880→            -Name "EnableSuperfetch" -Value 0 -Type DWord `
  1881→            -SafetyLevel "Safe" -Description "Disables SuperFetch on SSD  -  safe, reduces unnecessary disk I/O" -SettingID "Perf_NoSuperfetch_SSD"
  1882→
  1883→        # Disable scheduled defrag on SSD
  1884→        Write-Step "Disabling scheduled defragmentation (SSD)..."
  1885→        try {
  1886→            $defragTask = Get-ScheduledTask -TaskName "ScheduledDefrag" -ErrorAction SilentlyContinue
  1887→            if ($defragTask -and $defragTask.State -ne 'Disabled') {
  1888→                Disable-AppTask -TaskName "ScheduledDefrag" -ErrorAction SilentlyContinue -SafetyLevel "Safe" -Description "Disables scheduled defragmentation  -  SSD does not need defrag" -SettingID "Performance_NoScheduledDefrag"
  1889→                Write-Success "Scheduled defrag disabled (SSD)."
  1890→            }
  1891→        } catch { Write-Warn "Could not disable scheduled defrag: $_" }
  1892→
  1893→        # Enable TRIM on SSD
  1894→        Write-Step "Enabling TRIM on SSD..."
  1895→        try {
  1896→            $trimResult = fsutil behavior query DisableDeleteNotify 2>$null
  1897→            if ($trimResult -match "1") {
  1898→                fsutil behavior set DisableDeleteNotify 0 2>$null | Out-Null
  1899→                Write-Success "TRIM enabled (SSD)."
  1900→            } else {
  1901→                Write-Step "TRIM already enabled."
  1902→            }
  1903→        } catch { Write-Warn "Could not check/enable TRIM: $_" }
  1904→    } else {
  1905→        # HDD: SysMain is IMPORTANT for performance  -  do NOT disable
  1906→        Write-Host "  HDD detected: Keeping SysMain (SuperFetch) enabled  -  disabling on HDD kills performance." -ForegroundColor Yellow
  1907→        Write-Step "HDD detected: Keeping SysMain (SuperFetch) and Prefetch enabled."
  1908→    }
  1909→
  1910→    # 7. Limit Windows Defender CPU usage to ~25%

  1912→    # Note: Windows Defender Tamper Protection blocks direct registry edits.
  1913→    # Use Set-MpPreference cmdlet where possible, fall back to registry.
  1914→    try {

  1917→    } catch {
