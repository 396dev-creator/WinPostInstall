   111→   111→function Set-DefenderExclusions {
   112→   112→    <#
   113→   113→    .DESCRIPTION
   114→   114→        Adds this script and its working directory to Windows Defender exclusions.
   115→   115→        Prevents the script from being flagged as malware during execution.
   116→   116→        This is safe: exclusions are path-specific and only protect this tool.
   117→   117→    #>
   118→   118→    Write-Host "" -ForegroundColor Cyan
   119→   119→    Write-Host "  [AV SHIELD] Windows Defender Self-Exclusion" -ForegroundColor Cyan
   120→   120→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   121→   121→    Write-Host ""
   122→   122→
   123→   123→    $scriptPath = $PSCommandPath
   124→   124→    if (-not $scriptPath) { $scriptPath = $MyInvocation.PSCommandPath }
   125→   125→
   126→   126→    $exclusionAdded = $false
   127→   127→
   128→   128→    # Check if the exclusion already exists
   129→   129→    try {
   130→   130→        $existingExclusions = (Get-MpPreference -ErrorAction SilentlyContinue).ExclusionPath
   131→   131→        if ($existingExclusions -and $scriptPath -in $existingExclusions) {
   132→   132→            Write-Step "Defender exclusion already exists for this script."
   133→   133→            return
   134→   134→        }
   135→   135→    } catch {
   136→   136→        Write-Warn "Could not read current Defender preferences: $_"
   137→   137→    }
   138→   138→
   139→   139→    # Exclude the script file itself
   140→   140→    Write-Step "Adding script to Defender exclusions..."
   141→   141→    try {
   142→   142→        Add-MpPreference -ExclusionPath $scriptPath -ErrorAction Stop
   143→   143→        Write-Success "Excluded: $scriptPath"
   144→   144→        $exclusionAdded = $true
   145→   145→    } catch {
   146→   146→        Write-Warn "Failed to exclude script path: $_"
   147→   147→    }
   148→   148→
   149→   149→    # Exclude the working directory (where the script lives)
   150→   150→    $scriptDir = Split-Path $scriptPath -Parent
   151→   151→    if ($scriptDir -and (Test-Path $scriptDir)) {
   152→   152→        try {
   153→   153→            Add-MpPreference -ExclusionPath $scriptDir -ErrorAction Stop
   154→   154→            Write-Success "Excluded directory: $scriptDir"
   155→   155→            $exclusionAdded = $true
   156→   156→        } catch {
   157→   157→            Write-Warn "Failed to exclude directory: $_"
   158→   158→        }
   159→   159→    }
   160→   160→
   161→   161→    # Exclude the log directory (C:\Temp)
   162→   162→    $logDir = "$env:SYSTEMDRIVE\Temp"
   163→   163→    try {
   164→   164→        Add-MpPreference -ExclusionPath $logDir -ErrorAction Stop
   165→   165→        Write-Success "Excluded log directory: $logDir"
   166→   166→        $exclusionAdded = $true
   167→   167→    } catch {
   168→   168→        Write-Warn "Failed to exclude log directory: $_"
   169→   169→    }
   170→   170→
   171→   171→    # Exclude common technician tool folders
   172→   172→    $toolPaths = @("C:\Tools", "C:\Scripts", "C:\TechnicianTools")
   173→   173→    foreach ($path in $toolPaths) {
   174→   174→        if (Test-Path $path) {
   175→   175→            try {
   176→   176→                Add-MpPreference -ExclusionPath $path -ErrorAction Stop
   177→   177→                Write-Success "Excluded: $path"
   178→   178→            } catch { }
   179→   179→        }
   180→   180→    }
   181→   181→
   182→   182→    if ($exclusionAdded) {
   183→   183→        Write-Success "Defender exclusions configured. This script is now trusted."
   184→   184→    }
   185→   185→
   186→   186→    Write-Host ""
   187→   187→    Write-Host "  NOTE: This only affects Windows Defender. For third-party AV," -ForegroundColor DarkGray
   188→   188→    Write-Host "  consider purchasing a Code Signing Certificate for full compatibility." -ForegroundColor DarkGray
   189→   189→    Write-Host ""
   190→   190→}

   250→   250→function Enable-SystemRestore {
   251→   251→    <#
   252→   252→    .DESCRIPTION
   253→   253→        Ensures System Restore is enabled and configured on the system drive.
   254→   254→        Enables the service if disabled, allocates disk space, and verifies
   255→   255→        the feature is operational before returning.
   256→   256→    #>
   257→   257→    Write-Host "" -ForegroundColor Cyan
   258→   258→    Write-Host "  [SAFETY] System Restore Verification" -ForegroundColor Cyan
   259→   259→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   260→   260→    Write-Host ""
   261→   261→
   262→   262→    $systemDrive = $env:SYSTEMDRIVE
   263→   263→    $restoreEnabled = $false
   264→   264→
   265→   265→    # Step 1: Check if System Restore is currently enabled on the system drive
   266→   266→    Write-Step "Checking System Restore status on $systemDrive..."
   267→   267→    try {
   268→   268→        $srStatus = Get-ComputerInfo -Property "SystemRestoreStatus" -ErrorAction SilentlyContinue
   269→   269→        # Also check via WMI which is more reliable
   270→   270→        $srInfo = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction SilentlyContinue
   271→   271→        if ($srInfo) {
   272→   272→            foreach ($sr in $srInfo) {
   273→   273→                if ($sr.RPSequence -gt 0) {
   274→   274→                    $restoreEnabled = $true
   275→   275→                    break
   276→   276→                }
   277→   277→            }
   278→   278→        }
   279→   279→    } catch {
   280→   280→        # WMI query may fail if System Restore is completely disabled
   281→   281→        $restoreEnabled = $false
   282→   282→    }
   283→   283→
   284→   284→    # Alternative check: try to list existing restore points
   285→   285→    if (-not $restoreEnabled) {
   286→   286→        try {
   287→   287→            $existingPoints = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction Stop
   288→   288→            if ($existingPoints -and $existingPoints.Count -ge 0) {
   289→   289→                $restoreEnabled = $true
   290→   290→            }
   291→   291→        } catch {
   292→   292→            $restoreEnabled = $false
   293→   293→        }
   294→   294→    }
   295→   295→
   296→   296→    # Step 2: Enable System Restore if not currently enabled
   297→   297→    if (-not $restoreEnabled) {
   298→   298→        Write-Warn "System Restore is NOT enabled on $systemDrive."
   299→   299→        Write-Step "Enabling System Restore on $systemDrive..."
   300→   300→
   301→   301→        try {
   302→   302→            # Enable System Restore via registry (most reliable method)
   303→   303→            $srPolicyPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore"
   304→   304→            if (-not (Test-Path $srPolicyPath)) {
   305→   305→                New-Item -Path $srPolicyPath -Force -ErrorAction SilentlyContinue | Out-Null
   306→   306→            }
   307→   307→            # DisableSR = 0 means System Restore is ENABLED
   308→   308→            Set-ItemProperty -Path $srPolicyPath -Name "DisableSR" -Value 0 -Type DWord -Force -ErrorAction Stop
   309→   309→
   310→   310→            # Also enable via PowerShell cmdlet (Windows 10/11)
   311→   311→            try {
   312→   312→                Enable-ComputerRestore -Drive $systemDrive -ErrorAction Stop
   313→   313→            } catch {
   314→   314→                # Fallback: use vssadmin to enable shadow copies
   315→   315→                $vssResult = vssadmin list shadows 2>&1
   316→   316→                if ($LASTEXITCODE -ne 0) {
   317→   317→                    # Try creating a shadow copy to force-enable VSS
   318→   318→                    vssadmin create shadow /For=$systemDrive 2>&1 | Out-Null
   319→   319→                }
   320→   320→            }
   321→   321→
   322→   322→            # Start the System Restore service if not running
   323→   323→            $srService = Get-Service -Name "sr" -ErrorAction SilentlyContinue
   324→   324→            if ($srService -and $srService.Status -ne 'Running') {
   325→   325→                try {
   326→   326→                    Set-Service -Name "sr" -StartupType Manual -ErrorAction Stop
   327→   327→                    Start-Service -Name "sr" -ErrorAction Stop
   328→   328→                } catch {
   329→   329→                    Write-Warn "Could not start System Restore service: $_"
   330→   330→                }
   331→   331→            }
   332→   332→
   333→   333→            # Also ensure VSS (Volume Shadow Copy) service is running (required for System Restore)
   334→   334→            $vssService = Get-Service -Name "VSS" -ErrorAction SilentlyContinue
   335→   335→            if ($vssService -and $vssService.Status -ne 'Running') {
   336→   336→                try {
   337→   337→                    Set-Service -Name "VSS" -StartupType Manual -ErrorAction Stop
   338→   338→                    Start-Service -Name "VSS" -ErrorAction Stop
   339→   339→                    Write-Success "VSS service started."
   340→   340→                } catch {
   341→   341→                    Write-Warn "Could not start VSS service: $_"
   342→   342→                }
   343→   343→            }
   344→   344→
   345→   345→            Start-Sleep -Seconds 2
   346→   346→            Write-Success "System Restore has been ENABLED on $systemDrive."
   347→   347→        } catch {
   348→   348→            Write-Err "Failed to enable System Restore: $_"
   349→   349→            Write-Warn "Continuing without restore point. Manual enable: System Properties > System Protection > Configure > Turn on"
   350→   350→        }
   351→   351→    } else {
   352→   352→        Write-Success "System Restore is already enabled on $systemDrive."
   353→   353→    }
   354→   354→
   355→   355→    # Step 3: Configure disk space allocation for restore points
   356→   356→    Write-Step "Configuring System Restore disk space..."
   357→   357→    try {
   358→   358→        # Set maximum disk space usage to 5% (reasonable for most systems)
   359→   359→        # This is done via registry: RPSessionInterval and DiskPercent
   360→   360→        $srCfgPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore\Cfg"
   361→   361→        if (-not (Test-Path $srCfgPath)) {
   362→   362→            New-Item -Path $srCfgPath -Force -ErrorAction SilentlyContinue | Out-Null
   363→   363→        }
   364→   364→        # DiskPercent = 5 means 5% of disk (0 = unlimited, we use 5% as safe default)
   365→   365→        Set-ItemProperty -Path $srCfgPath -Name "DiskPercent" -Value 5 -Type DWord -Force -ErrorAction SilentlyContinue
   366→   366→        # RPSessionInterval = 0 means create restore points at any interval
   367→   367→        Set-ItemProperty -Path $srCfgPath -Name "RPSessionInterval" -Value 0 -Type DWord -Force -ErrorAction SilentlyContinue
   368→   368→        Write-Success "Restore disk space set to 5% of drive."
   369→   369→    } catch {
   370→   370→        Write-Warn "Could not configure restore disk space: $_"
   371→   371→    }
   372→   372→
   373→   373→    Write-Host ""
   374→   374→}

   376→   376→function New-SystemRestorePoint {
   377→   377→    <#
   378→   378→    .DESCRIPTION
   379→   379→        Creates a System Restore point before any optimization changes are made.
   380→   380→        This provides a safety net so the user can roll back all changes.
   381→   381→    #>
   382→   382→    Write-Host "" -ForegroundColor Cyan
   383→   383→    Write-Host "  [SAFETY] Creating System Restore Point" -ForegroundColor Cyan
   384→   384→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   385→   385→    Write-Host ""
   386→   386→
   387→   387→    $restorePointCreated = $false
   388→   388→    $description = "Pre-Optimization Restore Point - WinPostInstall v$ScriptVersion - $(Get-Date -Format 'yyyy-MM-dd HH:mm')"
   389→   389→
   390→   390→    # Method 1: Use Checkpoint-Computer (PowerShell cmdlet, Windows 8+)
   391→   391→    Write-Step "Creating restore point: 'Pre-Optimization'..."
   392→   392→    try {
   393→   393→        Checkpoint-Computer -Description $description -RestorePointType "MODIFY_SETTINGS" -ErrorAction Stop
   394→   394→        $restorePointCreated = $true
   395→   395→        Write-Success "Restore point created successfully via Checkpoint-Computer."
   396→   396→    } catch {
   397→   397→        Write-Warn "Checkpoint-Computer failed: $_"
   398→   398→    }
   399→   399→
   400→   400→    # Method 2: Fallback using WMI if Method 1 failed
   401→   401→    if (-not $restorePointCreated) {
   402→   402→        Write-Step "Trying WMI method to create restore point..."
   403→   403→        try {
   404→   404→            $wmiSR = [System.Windows.Forms.SystemRestore]
   405→   405→            # WMI approach via Invoke-CimMethod
   406→   406→            $result = Invoke-CimMethod -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" `
   407→   407→                -MethodName "CreateRestorePoint" `
   408→   408→                -Arguments @{
   409→   409→                    Description         = $description
   410→   410→                    RestorePointType    = 12  # MODIFY_SETTINGS
   411→   411→                    EventType           = 100  # BEGIN_NESTED_SYSTEM_CHANGE
   412→   412→                } -ErrorAction Stop
   413→   413→            if ($result -and $result.ReturnValue -eq 0) {
   414→   414→                $restorePointCreated = $true
   415→   415→                Write-Success "Restore point created via WMI."
   416→   416→            } else {
   417→   417→                Write-Warn "WMI returned non-zero: $result"
   418→   418→            }
   419→   419→        } catch {
   420→   420→            Write-Warn "WMI method failed: $_"
   421→   421→        }
   422→   422→    }
   423→   423→
   424→   424→    # Method 3: Final fallback via srclient.dll COM automation
   425→   425→    if (-not $restorePointCreated) {
   426→   426→        Write-Step "Trying COM automation (srclient.dll)..."
   427→   427→        try {
   428→   428→            $srClient = New-Object -ComObject "SRClient.RestorePoint" -ErrorAction Stop
   429→   429→            # 7 = MODIFY_SETTINGS, 100 = BEGIN_NESTED_SYSTEM_CHANGE
   430→   430→            $srClient.CreateRestorePoint($description, 7, 100)
   431→   431→            $restorePointCreated = $true
   432→   432→            Write-Success "Restore point created via COM automation."
   433→   433→        } catch {
   434→   434→            Write-Warn "COM method failed: $_"
   435→   435→        }
   436→   436→    }
   437→   437→
   438→   438→    if ($restorePointCreated) {
   439→   439→        Write-Host ""
   440→   440→        Write-Host "  ====================================================================" -ForegroundColor Green
   441→   441→        Write-Host "  RESTORE POINT CREATED: 'Pre-Optimization'" -ForegroundColor Green
   442→   442→        Write-Host "  You can roll back all changes via:" -ForegroundColor White

   462→   462→function Set-Registry {
   463→   463→    [CmdletBinding()]
   464→   464→    param(
   465→   465→        [string]$Path,
   466→   466→        [string]$Name,
   467→   467→        $Value,
   468→   468→        [string]$Type = "DWord"
   469→   469→    )
   470→   470→    try {
   471→   471→        if (-not (Test-Path $Path)) {
   472→   472→            New-Item -Path $Path -Force -ErrorAction SilentlyContinue | Out-Null
   473→   473→        }
   474→   474→        # Idempotency: Check if the value already matches what we want to set
   475→   475→        $currentValue = $null
   476→   476→        try {
   477→   477→            $currentValue = (Get-ItemProperty -Path $Path -Name $Name -ErrorAction Stop).$Name
   478→   478→        } catch {
   479→   479→            # Property doesn't exist yet — need to create it
   480→   480→        }
   481→   481→
   482→   482→        # Compare current vs desired value
   483→   483→        $valueMatches = $false
   484→   484→        if ($null -ne $currentValue) {
   485→   485→            if ($Type -eq "DWord" -or $Type -eq "QWord") {
   486→   486→                # Numeric comparison (handle both int and string inputs)
   487→   487→                $valueMatches = ([int]$currentValue -eq [int]$Value)
   488→   488→            } elseif ($Type -eq "String" -or $Type -eq "ExpandString") {
   489→   489→                $valueMatches = ([string]$currentValue -eq [string]$Value)
   490→   490→            } else {
   491→   491→                $valueMatches = ($currentValue -eq $Value)
   492→   492→            }
   493→   493→        }
   494→   494→
   495→   495→        if ($valueMatches) {
   496→   496→            Write-Step "Already set: $Path\$Name = $Value"
   497→   497→            return
   498→   498→        }
   499→   499→
   500→   500→        Set-ItemProperty -Path $Path -Name $Name -Value $Value -Type $Type -Force -ErrorAction Stop
   501→   501→        Write-Success "Registry: $Path\$Name = $Value"
   502→   502→    }
   503→   503→    catch {
   504→   504→        Write-Err "Failed to set $Path\$Name : $_"
   505→   505→    }
   506→   506→}
