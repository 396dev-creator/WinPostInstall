    37→function Initialize-Script {
    38→    if (-not ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
    39→        Write-Host "[ERROR] This script must be run as Administrator." -ForegroundColor Red
    40→        Write-Host "Right-click PowerShell and select 'Run as Administrator', then re-run this script." -ForegroundColor Yellow
    41→        if (-not $Silent) { Read-Host "Press Enter to exit" }
    42→        exit 1
    43→    }
    44→
    45→    $logDir = "$env:SYSTEMDRIVE\Temp"
    46→    if (-not (Test-Path $logDir)) { New-Item -Path $logDir -ItemType Directory -Force | Out-Null }
    47→
    48→    Start-Transcript -Path $LogFile -Force | Out-Null
    49→
    50→    $osInfo = Get-CimInstance -ClassName Win32_OperatingSystem
    51→    $script:OSVersion = [version]$osInfo.Version
    52→    $script:OSBuild = $osInfo.BuildNumber
    53→    $script:IsWin11 = $script:OSBuild -ge 22000
    54→    $script:OSCaption = if ($script:IsWin11) { "Windows 11" } else { "Windows 10" }
    55→    $script:ComputerName = $env:COMPUTERNAME
    56→
    57→    Write-Host ""
    58→    Write-Host "================================================================" -ForegroundColor Cyan
    59→    Write-Host "  Windows Post-Installation Optimizer v$ScriptVersion" -ForegroundColor Cyan
    60→    Write-Host "  Target: $script:OSCaption (Build $script:OSBuild)" -ForegroundColor Cyan
    61→    Write-Host "  Computer: $script:ComputerName" -ForegroundColor Cyan
    62→    Write-Host "  Log file: $LogFile" -ForegroundColor DarkGray
    63→    Write-Host "================================================================" -ForegroundColor Cyan
    64→    Write-Host ""
    65→}

    83→function Set-Registry {
    84→    [CmdletBinding()]
    85→    param(
    86→        [string]$Path,
    87→        [string]$Name,
    88→        $Value,
    89→        [string]$Type = "DWord"
    90→    )
    91→    try {
    92→        if (-not (Test-Path $Path)) {
    93→            New-Item -Path $Path -Force -ErrorAction SilentlyContinue | Out-Null
    94→        }
    95→        Set-ItemProperty -Path $Path -Name $Name -Value $Value -Type $Type -Force -ErrorAction Stop
    96→        Write-Success "Registry: $Path\$Name = $Value"
    97→    }
    98→    catch {
    99→        Write-Err "Failed to set $Path\$Name : $_"
   100→    }
   101→}

   103→function Remove-Registry {
   104→    [CmdletBinding()]
   105→    param(
   106→        [string]$Path,
   107→        [string]$Name
   108→    )
   109→    try {
   110→        if (Test-Path $Path) {
   111→            Remove-ItemProperty -Path $Path -Name $Name -Force -ErrorAction Stop 2>$null
   112→            Write-Success "Registry removed: $Path\$Name"
   113→        }
   114→    }
   115→    catch {
   116→        Write-Warn "Could not remove $Path\$Name : $_"
   117→    }
   118→}

   120→function Disable-ScheduledTask {
   121→    [CmdletBinding()]
   122→    param(
   123→        [string]$TaskPath,
   124→        [string]$TaskName
   125→    )
   126→    try {
   127→        $task = Get-ScheduledTask -TaskPath $TaskPath -TaskName $TaskName -ErrorAction SilentlyContinue
   128→        if ($task) {
   129→            Disable-ScheduledTask -TaskPath $TaskPath -TaskName $TaskName -ErrorAction Stop | Out-Null
   130→            Write-Success "Task disabled: $TaskPath$TaskName"
   131→        }
   132→    }
   133→    catch {
   134→        Write-Warn "Could not disable task $TaskPath$TaskName : $_"
   135→    }
   136→}

   371→        "*ActiproSoftware*","*Disney*","*Duolingo*","*EclipseManager*",
   372→        "*Facebook*","*Fitbit*","*Flipboard*","*king.com.CandyCrush*",
   373→        "*king.com.CandyCrushSodaSaga*","*king.com.FarmHeroesSaga*",

   386→        if ($packages) {

   406→            } catch { Write-Warn "Could not remove provisioned $($prov.DisplayName): $_" }
   407→        }
   408→    }
   409→
   410→    Write-Host ""
   411→    Write-Success "Debloat complete. Removed $removed packages ($notFound not found)."
   412→}
