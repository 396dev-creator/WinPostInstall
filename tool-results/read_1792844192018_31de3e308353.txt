   153→# Self-elevate: If not running as Administrator, re-launch in an elevated window
   154→# This allows the user to simply double-click the script
   155→#
   156→# CRITICAL: When invoked via `powershell -Command "& 'script.ps1' ..."`, $PSCommandPath

   183→            Write-Host "  Elevated window opened (re-downloaded from GitHub). This window can be closed." -ForegroundColor Green

   201→function Set-DefenderExclusions {
   202→    <#
   203→    .DESCRIPTION
   204→        Adds this script and its working directory to Windows Defender exclusions.
   205→        Prevents the script from being flagged as malware during execution.
   206→        This is safe: exclusions are path-specific and only protect this tool.
   207→    #>
   208→    Write-Host "" -ForegroundColor Cyan
   209→    Write-Host "  [AV SHIELD] Windows Defender Self-Exclusion" -ForegroundColor Cyan
   210→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   211→    Write-Host ""
   212→
   213→    $scriptPath = $PSCommandPath
   214→    if (-not $scriptPath) { $scriptPath = $MyInvocation.PSCommandPath }
   215→
   216→    $exclusionAdded = $false
   217→
   218→    # Check if the exclusion already exists
   219→    try {
   220→        $existingExclusions = (Get-MpPreference -ErrorAction SilentlyContinue).ExclusionPath
   221→        if ($existingExclusions -and $scriptPath -in $existingExclusions) {
   222→            Write-Step "Defender exclusion already exists for this script."
   223→            return
   224→        }
   225→    } catch {
   226→        Write-Warn "Could not read current Defender preferences: $_"
   227→    }
   228→
   229→    # Exclude the script file itself
   230→    Write-Step "Adding script to Defender exclusions..."
   231→    try {
   232→        Add-MpPreference -ExclusionPath $scriptPath -ErrorAction Stop
   233→        Write-Success "Excluded: $scriptPath"
   234→        $exclusionAdded = $true
   235→    } catch {
   236→        Write-Warn "Failed to exclude script path: $_"
   237→    }
   238→
   239→    # Exclude the working directory (where the script lives)
   240→    $scriptDir = Split-Path $scriptPath -Parent
   241→    if ($scriptDir -and (Test-Path $scriptDir)) {
   242→        try {
   243→            Add-MpPreference -ExclusionPath $scriptDir -ErrorAction Stop
   244→            Write-Success "Excluded directory: $scriptDir"
   245→            $exclusionAdded = $true
   246→        } catch {
   247→            Write-Warn "Failed to exclude directory: $_"
   248→        }
   249→    }
   250→
   251→    # Exclude the log directory (C:\Temp)
   252→    $logDir = "$env:SYSTEMDRIVE\Temp"
   253→    try {
   254→        Add-MpPreference -ExclusionPath $logDir -ErrorAction Stop
   255→        Write-Success "Excluded log directory: $logDir"
   256→        $exclusionAdded = $true
   257→    } catch {
   258→        Write-Warn "Failed to exclude log directory: $_"
   259→    }
   260→
   261→    # Exclude common technician tool folders
   262→    $toolPaths = @("C:\Tools", "C:\Scripts", "C:\TechnicianTools")
   263→    foreach ($path in $toolPaths) {
   264→        if (Test-Path $path) {
   265→            try {
   266→                Add-MpPreference -ExclusionPath $path -ErrorAction Stop
   267→                Write-Success "Excluded: $path"
   268→            } catch { }
   269→        }
   270→    }
   271→
   272→    if ($exclusionAdded) {
   273→        Write-Success "Defender exclusions configured. This script is now trusted."
   274→    }
   275→
   276→    Write-Host ""
   277→    Write-Host "  NOTE: This only affects Windows Defender. For third-party AV," -ForegroundColor DarkGray
   278→    Write-Host "  consider purchasing a Code Signing Certificate for full compatibility." -ForegroundColor DarkGray
   279→    Write-Host ""
   280→}

   314→            New-Item -Path $script:BackupDir -ItemType Directory -Force | Out-Null

   326→                reg export $regExport.Hive $exportPath /y 2>$null | Out-Null

   329→                Write-Warn "Could not export $($regExport.Hive): $_"

   351→        }
   352→    } catch { }
   353→    if (-not $script:IsLaptop) {
   354→        try {
   355→            $script:IsLaptop = (Get-CimInstance -ClassName Win32_Battery -ErrorAction SilentlyContinue).Count -gt 0
   356→        } catch { }
   357→    }
   358→
   359→    Write-Host ""
   360→    Write-Host "================================================================" -ForegroundColor Cyan
   361→    Write-Host "  Windows Post-Installation Optimizer v$ScriptVersion" -ForegroundColor Cyan
   362→    Write-Host "  Target: $script:OSCaption (Build $script:OSBuild)" -ForegroundColor Cyan
   363→    Write-Host "  Computer: $script:LocalComputerName" -ForegroundColor Cyan
   364→    Write-Host "  Chassis: $(if ($script:IsLaptop) { 'Laptop' } else { 'Desktop' })" -ForegroundColor Cyan
   365→    Write-Host "  Log file: $LogFile" -ForegroundColor DarkGray
   366→    Write-Host "================================================================" -ForegroundColor Cyan
   367→    Write-Host ""

   373→function Test-Prerequisites {
   374→    <#
   375→    .DESCRIPTION
   376→        Pre-flight checks: internet connectivity, disk space, pending reboot, battery.
   377→        Warns on issues but does not block execution.
   378→    #>
   379→    Write-Host "" -ForegroundColor Cyan
   380→    Write-Host "  [PRE-FLIGHT] System Prerequisites Check" -ForegroundColor Cyan
   381→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   382→    Write-Host ""
   383→
   384→    $checksPassed = 0; $checksWarned = 0
   385→
   386→    # 1. Internet connectivity
   387→    Write-Step "Checking internet connectivity..."
   388→    try {
   389→        $netTest = Test-Connection -ComputerName 8.8.8.8 -Count 1 -Quiet -ErrorAction Stop
   390→        if ($netTest) {
   391→            Write-Success "Internet connectivity: OK"
   392→            $checksPassed++
   393→        } else {
   394→            Write-Warn "Internet connectivity: OFFLINE (some features may fail)"
   395→            $checksWarned++
   396→        }
   397→    } catch {
   398→        Write-Warn "Internet connectivity: Could not check (some features may fail)"
   399→        $checksWarned++
   400→    }
   401→
   402→    # 2. Free disk space
   403→    Write-Step "Checking free disk space..."
   404→    try {
   405→        $sysDrive = Get-PSDrive -Name ($env:SYSTEMDRIVE -replace ':','') -ErrorAction Stop
   406→        $freeGB = [math]::Round($sysDrive.Free / 1GB, 2)
   407→        if ($freeGB -gt 10) {
   408→            Write-Success "Free disk space: $freeGB GB (OK)"
   409→            $checksPassed++
   410→        } else {
   411→            Write-Warn "Free disk space: $freeGB GB (LOW - recommend >10 GB)"
   412→            $checksWarned++
   413→        }
   414→    } catch {
   415→        Write-Warn "Could not check disk space: $_"
   416→        $checksWarned++
   417→    }
   418→
   419→    # 3. Pending reboot
   420→    Write-Step "Checking for pending reboot..."
   421→    $rebootPending = $false
   422→    $cbsPath = "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Component Based Servicing\RebootPending"
   423→    $wuPath = "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\WindowsUpdate\Auto Update\RebootRequired"
   424→    if (Test-Path $cbsPath -ErrorAction SilentlyContinue) { $rebootPending = $true }
   425→    if (Test-Path $wuPath -ErrorAction SilentlyContinue) { $rebootPending = $true }
   426→    if ($rebootPending) {
   427→        Write-Warn "Pending reboot detected! A restart is recommended before running optimizations."
   428→        $checksWarned++
   429→    } else {
   430→        Write-Success "No pending reboot."
   431→        $checksPassed++
   432→    }
   433→
   434→    # 4. Battery check (if laptop)
   435→    if ($script:IsLaptop) {
   436→        Write-Step "Checking battery level..."
   437→        try {
   438→            $battery = Get-CimInstance -ClassName Win32_Battery -ErrorAction Stop
   439→            $onAC = $battery.BatteryStatus -eq 2 -or $battery.BatteryStatus -eq 6 -or $battery.BatteryStatus -eq 7 -or $battery.BatteryStatus -eq 8
   440→            if ($onAC) {
   441→                Write-Success "On AC power."
   442→                $checksPassed++
   443→            } elseif ($battery.EstimatedChargeRemaining -gt 50) {
   444→                Write-Success "Battery level: $($battery.EstimatedChargeRemaining)% (OK)"
   445→                $checksPassed++
   446→            } else {
   447→                Write-Warn "Battery level: $($battery.EstimatedChargeRemaining)% - recommend AC power or >50% charge."
   448→                $checksWarned++
   449→            }
   450→        } catch {
   451→            Write-Warn "Could not check battery: $_"
   452→            $checksWarned++
   453→        }
   454→    }
   455→
   456→    Write-Host ""
   457→    Write-Host "  Pre-flight summary: $checksPassed passed, $checksWarned warnings" -ForegroundColor $(if ($checksWarned -gt 0) { 'Yellow' } else { 'Green' })
   458→    Write-Host ""
   459→}

   461→function Get-BootDriveType {
   462→    <#
   463→    .DESCRIPTION
   464→        Detects whether the boot drive is an SSD or HDD.
   465→        Returns "SSD" or "HDD".
   466→    #>
   467→    $bootType = "HDD"  # Default assumption
   468→    try {
   469→        # Method 1: Get-PhysicalDisk (most reliable on Win8+)
   470→        $physicalDisk = Get-PhysicalDisk -ErrorAction SilentlyContinue | Where-Object DeviceId -eq 0
   471→        if ($physicalDisk) {
   472→            if ($physicalDisk.MediaType -eq "SSD") { return "SSD" }
   473→            if ($physicalDisk.MediaType -eq "HDD") { return "HDD" }
   474→        }
   475→    } catch { }
   476→
   477→    try {
   478→        # Method 2: Get-Disk BusType fallback
   479→        $bootDisk = Get-Disk -ErrorAction SilentlyContinue | Where-Object IsBoot -eq $true
   480→        if ($bootDisk) {
   481→            $ssdBusTypes = @("NVMe", "SATA", "RAID", "SCSI")

   519→function Enable-SystemRestore {
   520→    <#
   521→    .DESCRIPTION
   522→        Ensures System Restore is enabled and configured on the system drive.
   523→        Enables the service if disabled, allocates disk space, and verifies
   524→        the feature is operational before returning.
   525→    #>
   526→    Write-Host "" -ForegroundColor Cyan
   527→    Write-Host "  [SAFETY] System Restore Verification" -ForegroundColor Cyan
   528→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   529→    Write-Host ""
   530→
   531→    $systemDrive = $env:SYSTEMDRIVE
   532→    $restoreEnabled = $false
   533→
   534→    # Step 1: Check if System Restore is currently enabled on the system drive
   535→    Write-Step "Checking System Restore status on $systemDrive..."
   536→    try {
   537→        $srStatus = Get-ComputerInfo -Property "SystemRestoreStatus" -ErrorAction SilentlyContinue
   538→        # Also check via WMI which is more reliable
   539→        $srInfo = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction SilentlyContinue
   540→        if ($srInfo) {
   541→            foreach ($sr in $srInfo) {
   542→                if ($sr.RPSequence -gt 0) {
   543→                    $restoreEnabled = $true
   544→                    break
   545→                }
   546→            }
   547→        }
   548→    } catch {
   549→        # WMI query may fail if System Restore is completely disabled
   550→        $restoreEnabled = $false
   551→    }
   552→
   553→    # Alternative check: try to list existing restore points
   554→    if (-not $restoreEnabled) {
   555→        try {
   556→            $existingPoints = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction Stop
   557→            if ($existingPoints -and $existingPoints.Count -ge 0) {
   558→                $restoreEnabled = $true
   559→            }
   560→        } catch {
   561→            $restoreEnabled = $false
   562→        }
   563→    }
   564→
   565→    # Step 2: Enable System Restore if not currently enabled
   566→    if (-not $restoreEnabled) {
   567→        Write-Warn "System Restore is NOT enabled on $systemDrive."
   568→        Write-Step "Enabling System Restore on $systemDrive..."
   569→
   570→        try {
   571→            # Enable System Restore via registry (most reliable method)
   572→            $srPolicyPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore"
   573→            if (-not (Test-Path $srPolicyPath)) {
   574→                New-Item -Path $srPolicyPath -Force -ErrorAction SilentlyContinue | Out-Null
   575→            }
   576→            # DisableSR = 0 means System Restore is ENABLED
   577→            Set-ItemProperty -Path $srPolicyPath -Name "DisableSR" -Value 0 -Type DWord -Force -ErrorAction Stop
   578→
   579→            # Also enable via PowerShell cmdlet (Windows 10/11)
   580→            try {
   581→                Enable-ComputerRestore -Drive $systemDrive -ErrorAction Stop
   582→            } catch {
   583→                # Fallback: use vssadmin to enable shadow copies
   584→                $vssResult = vssadmin list shadows 2>&1
   585→                if ($LASTEXITCODE -ne 0) {
   586→                    # Try creating a shadow copy to force-enable VSS
   587→                    vssadmin create shadow /For=$systemDrive 2>&1 | Out-Null
   588→                }
   589→            }
   590→
   591→            # Start the System Restore service if not running
   592→            $srService = Get-Service -Name "sr" -ErrorAction SilentlyContinue
   593→            if ($srService -and $srService.Status -ne 'Running') {
   594→                try {
   595→                    Set-Service -Name "sr" -StartupType Manual -ErrorAction Stop
   596→                    Start-Service -Name "sr" -ErrorAction Stop
   597→                } catch {
   598→                    Write-Warn "Could not start System Restore service: $_"
   599→                }
   600→            }
   601→
   602→            # Also ensure VSS (Volume Shadow Copy) service is running (required for System Restore)
   603→            $vssService = Get-Service -Name "VSS" -ErrorAction SilentlyContinue
   604→            if ($vssService -and $vssService.Status -ne 'Running') {
   605→                try {
   606→                    Set-Service -Name "VSS" -StartupType Manual -ErrorAction Stop
   607→                    Start-Service -Name "VSS" -ErrorAction Stop
   608→                    Write-Success "VSS service started."
   609→                } catch {
   610→                    Write-Warn "Could not start VSS service: $_"
   611→                }
   612→            }
   613→
   614→            Start-Sleep -Seconds 2
   615→            Write-Success "System Restore has been ENABLED on $systemDrive."
   616→        } catch {
   617→            Write-Err "Failed to enable System Restore: $_"
   618→            Write-Warn "Continuing without restore point. Manual enable: System Properties > System Protection > Configure > Turn on"
   619→        }
   620→    } else {
   621→        Write-Success "System Restore is already enabled on $systemDrive."
   622→    }
   623→
   624→    # Step 3: Configure disk space allocation for restore points
   625→    Write-Step "Configuring System Restore disk space..."
   626→    try {
   627→        # Set maximum disk space usage to 5% (reasonable for most systems)
   628→        # This is done via registry: RPSessionInterval and DiskPercent
   629→        $srCfgPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore\Cfg"
   630→        if (-not (Test-Path $srCfgPath)) {
   631→            New-Item -Path $srCfgPath -Force -ErrorAction SilentlyContinue | Out-Null
   632→        }
   633→        # DiskPercent = 5 means 5% of disk (0 = unlimited, we use 5% as safe default)
   634→        Set-ItemProperty -Path $srCfgPath -Name "DiskPercent" -Value 5 -Type DWord -Force -ErrorAction SilentlyContinue
   635→        # RPSessionInterval = 0 means create restore points at any interval
   636→        Set-ItemProperty -Path $srCfgPath -Name "RPSessionInterval" -Value 0 -Type DWord -Force -ErrorAction SilentlyContinue
   637→        Write-Success "Restore disk space set to 5% of drive."
   638→    } catch {
   639→        Write-Warn "Could not configure restore disk space: $_"
   640→    }
   641→
   642→    Write-Host ""
   643→}

   645→function New-SystemRestorePoint {
   646→    <#
   647→    .DESCRIPTION
   648→        Creates a System Restore point before any optimization changes are made.
   649→        This provides a safety net so the user can roll back all changes.
   650→    #>
   651→    Write-Host "" -ForegroundColor Cyan
   652→    Write-Host "  [SAFETY] Creating System Restore Point" -ForegroundColor Cyan
   653→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   654→    Write-Host ""
   655→
   656→    $restorePointCreated = $false
   657→    $description = "Pre-Optimization Restore Point - WinPostInstall v$ScriptVersion - $(Get-Date -Format 'yyyy-MM-dd HH:mm')"
   658→
   659→    # Method 1: Use Checkpoint-Computer (PowerShell cmdlet, Windows 8+)
   660→    Write-Step "Creating restore point: 'Pre-Optimization'..."
   661→    try {
   662→        Checkpoint-Computer -Description $description -RestorePointType "MODIFY_SETTINGS" -ErrorAction Stop
   663→        $restorePointCreated = $true
   664→        Write-Success "Restore point created successfully via Checkpoint-Computer."
   665→    } catch {
   666→        Write-Warn "Checkpoint-Computer failed: $_"
   667→    }
   668→
   669→    # Method 2: Fallback using WMI if Method 1 failed
   670→    if (-not $restorePointCreated) {
   671→        Write-Step "Trying WMI method to create restore point..."
   672→        try {
   673→            $wmiSR = [System.Windows.Forms.SystemRestore]
   674→            # WMI approach via Invoke-CimMethod
   675→            $result = Invoke-CimMethod -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" `
   676→                -MethodName "CreateRestorePoint" `
   677→                -Arguments @{
   678→                    Description         = $description
   679→                    RestorePointType    = 12  # MODIFY_SETTINGS
   680→                    EventType           = 100  # BEGIN_NESTED_SYSTEM_CHANGE
   681→                } -ErrorAction Stop
   682→            if ($result -and $result.ReturnValue -eq 0) {
   683→                $restorePointCreated = $true
   684→                Write-Success "Restore point created via WMI."
   685→            } else {
   686→                Write-Warn "WMI returned non-zero: $result"
   687→            }
   688→        } catch {
   689→            Write-Warn "WMI method failed: $_"
   690→        }
   691→    }
   692→
   693→    # Method 3: Final fallback via srclient.dll COM automation
   694→    if (-not $restorePointCreated) {
   695→        Write-Step "Trying COM automation (srclient.dll)..."
   696→        try {
   697→            $srClient = New-Object -ComObject "SRClient.RestorePoint" -ErrorAction Stop
   698→            # 7 = MODIFY_SETTINGS, 100 = BEGIN_NESTED_SYSTEM_CHANGE
   699→            $srClient.CreateRestorePoint($description, 7, 100)
   700→            $restorePointCreated = $true
   701→            Write-Success "Restore point created via COM automation."
   702→        } catch {
   703→            Write-Warn "COM method failed: $_"
   704→        }
   705→    }
   706→
   707→    if ($restorePointCreated) {
   708→        Write-Host ""
   709→        Write-Host "  ====================================================================" -ForegroundColor Green
   710→        Write-Host "  RESTORE POINT CREATED: 'Pre-Optimization'" -ForegroundColor Green
   711→        Write-Host "  You can roll back all changes via:" -ForegroundColor White
   712→        Write-Host "    System Properties > System Protection > System Restore" -ForegroundColor White
   713→        Write-Host "  ====================================================================" -ForegroundColor Green
   714→        Write-Host ""
   715→    } else {
   716→        Write-Host ""
   717→        Write-Warn "Could NOT create restore point automatically."
   718→        Write-Host "  Manual option: System Properties > System Protection > Create" -ForegroundColor Yellow
   719→        Write-Host "  Continuing without restore point backup." -ForegroundColor Yellow
   720→        Write-Host ""
   721→        if (-not $Silent) {
   722→            $continue = Read-Host "  Continue without restore point? (y/N)"
   723→            if ($continue -ne 'y' -and $continue -ne 'Y') {
   724→                Write-Host "  Aborted by user." -ForegroundColor Yellow
   725→                exit 0
   726→            }
   727→        }
   728→    }
   729→}

   731→function Set-Registry {
   732→    [CmdletBinding()]
   733→    param(
   734→        [string]$Path,
   735→        [string]$Name,
   736→        $Value,
   737→        [string]$Type = "DWord",
   738→        [ValidateSet("Safe","Moderate","Risky")]
   739→        [string]$SafetyLevel = "Safe",
   740→        [string]$Description = "",
   741→        [string]$SettingID = ""
   742→    )
   743→
   744→    # Generate a setting ID if not provided
   745→    if ([string]::IsNullOrEmpty($SettingID)) {
   746→        $SettingID = "$($Path.Replace('\','_').Replace(':','').Replace(' ',''))_$Name"
   747→    }
   748→
   749→    # Register in setting catalog for interactive menu
   750→    if (-not [string]::IsNullOrEmpty($Description)) {
   751→        $script:SettingCatalog[$SettingID] = @{
   752→            Section  = $script:CurrentSection
   753→            Safety   = $SafetyLevel
   754→            Desc     = $Description
   755→            Action   = "$Path\$Name = $Value"
   756→            Type     = "Registry"
   757→        }
   758→    }
   759→
   760→    # Collect risky descriptions for menu warnings
   761→    if ($SafetyLevel -eq "Risky" -and -not [string]::IsNullOrEmpty($Description)) {
   762→        $script:RiskyDescriptions += "$SettingID : $Description"
   763→    }
   764→
   765→    # Check if user has toggled this setting OFF via interactive menu
   766→    if ($script:SettingOverrides.ContainsKey($SettingID) -and $script:SettingOverrides[$SettingID]) {
   767→        Write-Host "  SKIPPED (user toggle): $Description" -ForegroundColor DarkGray
   768→        $script:TotalSkipped++
   769→        return
   770→    }
   771→
   772→    # Safety gate: Skip Risky settings unless explicitly allowed
   773→    if ($SafetyLevel -eq "Risky" -and -not $script:AllowRisky) {
   774→        Write-Host "  SKIPPED (risky): $Description — use -AllowRisky or enable in menu" -ForegroundColor Yellow
   775→        $script:TotalSkipped++
   776→        return
   777→    }
   778→
   779→    # Warn for Moderate settings
   780→    if ($SafetyLevel -eq "Moderate") {
   781→        Write-Host "  [MODERATE] $Description" -ForegroundColor DarkYellow
   782→    }
   783→
   784→    try {
   785→        if (-not (Test-Path $Path)) {
   786→            New-Item -Path $Path -Force -ErrorAction SilentlyContinue | Out-Null
   787→        }
   788→        # Idempotency: Check if the value already matches what we want to set
   789→        $currentValue = $null
   790→        try {
   791→            $currentValue = (Get-ItemProperty -Path $Path -Name $Name -ErrorAction Stop).$Name
   792→        } catch {
   793→            # Property doesn't exist yet — need to create it
   794→        }
   795→
   796→        # Compare current vs desired value
   797→        $valueMatches = $false
   798→        if ($null -ne $currentValue) {
   799→            if ($Type -eq "DWord" -or $Type -eq "QWord") {
   800→                # Numeric comparison (handle both int and string inputs)
   801→                $valueMatches = ([int]$currentValue -eq [int]$Value)
   802→            } elseif ($Type -eq "String" -or $Type -eq "ExpandString") {
   803→                $valueMatches = ([string]$currentValue -eq [string]$Value)
   804→            } else {
   805→                $valueMatches = ($currentValue -eq $Value)
   806→            }
   807→        }
   808→
   809→        if ($valueMatches) {
   810→            Write-Step "Already set: $Path\$Name = $Value"
   811→            $script:TotalSkipped++
   812→            return
   813→        }
   814→
   815→        # Capture old value for undo script before writing
   816→        if ($null -ne $currentValue) {
   817→            $oldVal = $currentValue.ToString()
   818→            $null = $script:UndoCommands.Add("Set-ItemProperty -Path '$Path' -Name '$Name' -Value '$oldVal' -Type '$Type' -Force")
   819→            $null = $script:UndoDescriptions.Add("Restore registry: $Path\$Name = $oldVal")
   820→        } else {
   821→            $null = $script:UndoCommands.Add("Remove-ItemProperty -Path '$Path' -Name '$Name' -Force -ErrorAction SilentlyContinue")
   822→            $null = $script:UndoDescriptions.Add("Remove registry key that was added: $Path\$Name")
   823→        }
   824→
   825→        Set-ItemProperty -Path $Path -Name $Name -Value $Value -Type $Type -Force -ErrorAction Stop
   826→        Write-Success "Registry: $Path\$Name = $Value"
   827→        $script:TotalChanged++
   828→    }
   829→    catch {
   830→        Write-Err "Failed to set $Path\$Name : $_"
   831→        $script:TotalFailed++
   832→    }
   833→}

   928→        }

   932→    }
   933→}

  1712→            } else {
  1713→                Write-Warn "Service not found: $($svc.Name)"
  1714→            }
  1715→        } catch { Write-Warn "Could not disable $($svc.Name): $_" }
  1716→    }
  1717→

  1719→    Write-Step "Disabling Startup delay..."
  1720→    Set-Registry -Path "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Explorer\Serialize" `
  1721→        -Name "StartupDelayInMSec" -Value 0 -Type DWord -SafetyLevel "Safe" -Description "Disables startup delay — speeds up boot by removing artificial delay" -SettingID "ServiceStartup_NoStartupDelay"

  1724→    Write-Success "Service and Startup optimization complete."

  1733→function Invoke-PerformanceOptimization {
  1734→    $script:CurrentSection = "Performance"
  1735→    Write-Host ""
  1736→    Write-Host "--- PERFORMANCE and POWER SETTINGS ---" -ForegroundColor Magenta
  1737→    Write-Host ""
  1738→
  1739→    # === HARDWARE-AWARE: SSD vs HDD ===
  1740→    $script:BootDriveType = Get-BootDriveType
  1741→    Write-Step "Boot drive type detected: $($script:BootDriveType)"
  1742→
  1743→    # === HARDWARE-AWARE: Laptop vs Desktop ===
  1744→    if ($script:IsLaptop) {
  1745→        Write-Step "Chassis type: Laptop — keeping Balanced power plan."
  1746→    } else {
  1747→        Write-Step "Chassis type: Desktop — setting High Performance power plan."
  1748→    }
  1749→
  1750→    # === HARDWARE-AWARE: RAM detection ===
  1751→    $totalRAM = 0
  1752→    try {
  1753→        $totalRAM = [math]::Round((Get-CimInstance Win32_ComputerSystem).TotalPhysicalMemory / 1GB, 2)
  1754→        Write-Step "Total RAM detected: $totalRAM GB"
  1755→    } catch {
  1756→        Write-Warn "Could not detect RAM: $_"
  1757→    }
  1758→
  1759→    # 1. Disable Hibernation
  1760→    Write-Step "Disabling Hibernation..."
  1761→    try {
  1762→        powercfg.exe /hibernate off 2>$null
  1763→        Write-Success "Hibernation disabled. hiberfil.sys removed."
  1764→    } catch { Write-Warn "Hibernation disable failed: $_" }
  1765→
  1766→    # 2. Disable Fast Startup (causes partial shutdowns, driver issues)
  1767→    Write-Step "Disabling Fast Startup..."
  1768→    Set-Registry -Path "HKLM:\SYSTEM\CurrentControlSet\Control\Session Manager\Power" `
  1769→        -Name "HiberbootEnabled" -Value 0 -Type DWord `
  1770→        -SafetyLevel "Safe" -Description "Disables Fast Startup — prevents partial shutdown/driver issues" -SettingID "Perf_NoFastStartup"
  1771→
  1772→    # 3. Disable Modern Standby Networking (battery drain + security)
  1773→    Write-Step "Disabling Modern Standby network connectivity..."
  1774→    Set-Registry -Path "HKLM:\SYSTEM\CurrentControlSet\Control\Power" `
  1775→        -Name "ModernSleep" -Value 0 -Type DWord `
  1776→        -SafetyLevel "Safe" -Description "Disables Modern Standby network — prevents battery drain during sleep" -SettingID "Perf_NoModernSleep"
  1777→
  1778→    # 4. Set power plan based on chassis type
  1779→    if ($script:IsLaptop) {
  1780→        # Laptop: keep Balanced, but warn about High Performance option
  1781→        Write-Host "  [MODERATE] High Performance on laptop — WARNING: drains battery. Keeping Balanced." -ForegroundColor DarkYellow
  1782→        Write-Step "Keeping Balanced power plan (laptop)."
  1783→    } else {
  1784→        # Desktop: set High Performance, disable lid close action
  1785→        Write-Step "Setting High Performance power plan (desktop)..."
  1786→        try {
  1787→            $highPerf = powercfg.exe /list | Select-String -Pattern "High performance|高性能"
  1788→            if ($highPerf) {
  1789→                $guid = ($highPerf.ToString() -split '\s+')[3]
  1790→                powercfg.exe /setactive $guid 2>$null
  1791→                Write-Success "Power plan set to High Performance ($guid)"
  1792→            } else {
  1793→                powercfg.exe /duplicatecheme 8c5e7fda-e8bf-4a96-9a85-a6e23a8c635c 2>$null | Out-Null
  1794→                Start-Sleep -Seconds 1
  1795→                $highPerf = powercfg.exe /list | Select-String -Pattern "High performance|高性能"
  1796→                if ($highPerf) {
  1797→                    $guid = ($highPerf.ToString() -split '\s+')[3]
  1798→                    powercfg.exe /setactive $guid 2>$null
  1799→                    Write-Success "High Performance plan created and activated ($guid)"
  1800→                } else { Write-Warn "Could not find or create High Performance plan." }
  1801→            }
  1802→        } catch { Write-Warn "Power plan change failed: $_" }
  1803→
  1804→        # Disable lid close action (desktop doesn't need it)
  1805→        Write-Step "Disabling lid close action (desktop)..."
  1806→        try {
  1807→            powercfg.exe /setacvalueindex SCHEME_CURRENT SUB_BUTTONS LIDACTION 0 2>$null
  1808→            powercfg.exe /setdcvalueindex SCHEME_CURRENT SUB_BUTTONS LIDACTION 0 2>$null
  1809→            powercfg.exe /setactive SCHEME_CURRENT 2>$null
  1810→            Write-Success "Lid close action set to 'Do nothing' (desktop)."
  1811→        } catch { Write-Warn "Could not change lid close action: $_" }
  1812→    }
  1813→
  1814→    # 5. Set visual effects to Best Performance
  1815→    Write-Step "Setting visual effects to 'Best Performance'..."
  1816→    Set-Registry -Path "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Explorer\VisualEffects" `
  1817→        -Name "VisualFXSetting" -Value 2 -Type DWord `
  1818→        -SafetyLevel "Safe" -Description "Sets visual effects to Best Performance mode" -SettingID "Perf_VisualFX"
  1819→
  1820→    $visualFx = @{
  1821→        "AnimateMinMaxMenus"=0; "ComboBoxAnimation"=0; "CursorShadow"=0;
  1822→        "DragFullWindows"=0; "ListBoxSmoothScrolling"=0; "ListviewAlphaSelect"=0;
  1823→        "ListviewShadow"=0; "MenuAnimation"=0; "SelectionFade"=0;
  1824→        "TaskbarAnimations"=0; "TooltipAnimation"=0
  1825→    }
  1826→    foreach ($key in $visualFx.Keys) {
  1827→        Set-Registry -Path "HKCU:\Control Panel\Desktop" -Name $key -Value $visualFx[$key] -Type DWord -SafetyLevel "Safe" -Description "Disables visual animation — improves performance" -SettingID "Performance_VisFX_Performance_VisFX"
  1828→    }
  1829→    Set-Registry -Path "HKCU:\SOFTWARE\Microsoft\Windows\DWM" `
  1830→        -Name "EnableAeroPeek" -Value 0 -Type DWord -SafetyLevel "Safe" -Description "Disables Aero Peek — removes desktop preview on taskbar hover" -SettingID "Performance_NoAeroPeek"
  1831→
  1832→    # 6. SysMain/Prefetch based on drive type
  1833→    if ($script:BootDriveType -eq "SSD") {
  1834→        Write-Step "SSD detected: Disabling SysMain (SuperFetch) and Prefetch..."
  1835→        try {
  1836→            $sysMain = Get-Service -Name "SysMain" -ErrorAction SilentlyContinue
  1837→            if ($sysMain -and $sysMain.StartType -ne 'Disabled') {
  1838→                Stop-Service -Name "SysMain" -Force -ErrorAction SilentlyContinue
  1839→                Set-Service -Name "SysMain" -StartupType Disabled -ErrorAction Stop

  1848→            -Name "EnablePrefetcher" -Value 0 -Type DWord `

  1851→            -Name "EnableSuperfetch" -Value 0 -Type DWord `

  1861→            }
  1862→        } catch { Write-Warn "Could not disable scheduled defrag: $_" }
  1863→
  1864→        # Enable TRIM on SSD
  1865→        Write-Step "Enabling TRIM on SSD..."
  1866→        try {
  1867→            $trimResult = fsutil behavior query DisableDeleteNotify 2>$null
  1868→            if ($trimResult -match "1") {
  1869→                fsutil behavior set DisableDeleteNotify 0 2>$null | Out-Null

  1871→            } else {
  1872→                Write-Step "TRIM already enabled."

  1877→        Write-Host "  HDD detected: Keeping SysMain (SuperFetch) enabled — disabling on HDD kills performance." -ForegroundColor Yellow
  1878→        Write-Step "HDD detected: Keeping SysMain (SuperFetch) and Prefetch enabled."
  1879→    }
  1880→
  1881→    # 7. Limit Windows Defender CPU usage to ~25%

  1883→    # Note: Windows Defender Tamper Protection blocks direct registry edits.
  1884→    # Use Set-MpPreference cmdlet where possible, fall back to registry.
  1885→    try {
  1886→        Set-MpPreference -ScanAvgCPULoadFactor 25 -ErrorAction Stop

  1893→    }
  1894→
  1895→    # 8. Disable OEM telemetry services
  1896→    Write-Step "Disabling OEM telemetry services..."
  1897→    $oemServices = @("HPAppHelperShell","HPDiags","HPHealth","HPSysInfo",
  1898→        "HpTouchpointAnalyticsService","HpTouchpointMgr","hpsvcsss",

  1903→            $s = Get-Service -Name $svc -ErrorAction SilentlyContinue
  1904→            if ($s) {
  1905→                Stop-Service -Name $svc -Force -ErrorAction SilentlyContinue
  1906→                Set-Service -Name $svc -StartupType Disabled -ErrorAction SilentlyContinue

  1912→    # 9. Increase SvcHost splitting threshold (reduce svchost.exe proliferation)

  1917→    # === RAM-AWARE TWEAKS ===
  1918→
  1919→    # 10. Page file optimization based on RAM
  1920→    if ($totalRAM -ge 32) {
  1921→        # >=32GB: Risky to disable page file entirely — only recommend, don't apply
  1922→        Write-Host "  [RISKY] Disabling page file with 32GB+ RAM — WARNING: can crash memory-heavy workloads" -ForegroundColor Yellow
  1923→        Write-Host "  (Not auto-applied. To disable: System Properties > Advanced > Performance > Virtual Memory > No paging file)" -ForegroundColor DarkGray

  1929→            -Name "PagingFiles" -Value "C:\pagefile.sys 1024 1024" -Type MultiString `
  1930→            -SafetyLevel "Moderate" -Description "Sets page file to fixed 1GB — OK with 16GB+ RAM" -SettingID "Perf_FixedPageFile"
  1931→    } elseif ($totalRAM -le 8 -and $totalRAM -gt 0) {
  1932→        # <=8GB: keep system-managed page file, but warn

  1934→    }
  1935→
  1936→    # 11. Disable NTFS last access time updates (improves SSD performance)
  1937→    Write-Step "Disabling NTFS last access time updates..."

  1939→        fsutil behavior set DisableLastAccess 1 2>$null | Out-Null
  1940→        Write-Success "NTFS last access time updates disabled."
  1941→    } catch {
  1942→        Write-Warn "Could not set DisableLastAccess: $_"
  1943→    }
  1944→
  1945→    Write-Host ""
  1946→    Write-Success "Performance and Power optimization complete."
  1947→}
