    76→function Set-DefenderExclusions {
    77→    <#
    78→    .DESCRIPTION
    79→        Adds this script and its working directory to Windows Defender exclusions.
    80→        Prevents the script from being flagged as malware during execution.
    81→        This is safe: exclusions are path-specific and only protect this tool.
    82→    #>
    83→    Write-Host "" -ForegroundColor Cyan
    84→    Write-Host "  [AV SHIELD] Windows Defender Self-Exclusion" -ForegroundColor Cyan
    85→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
    86→    Write-Host ""
    87→
    88→    $scriptPath = $PSCommandPath
    89→    if (-not $scriptPath) { $scriptPath = $MyInvocation.PSCommandPath }
    90→
    91→    $exclusionAdded = $false
    92→
    93→    # Check if the exclusion already exists
    94→    try {
    95→        $existingExclusions = (Get-MpPreference -ErrorAction SilentlyContinue).ExclusionPath
    96→        if ($existingExclusions -and $scriptPath -in $existingExclusions) {
    97→            Write-Step "Defender exclusion already exists for this script."
    98→            return
    99→        }
   100→    } catch {
   101→        Write-Warn "Could not read current Defender preferences: $_"
   102→    }
   103→
   104→    # Exclude the script file itself
   105→    Write-Step "Adding script to Defender exclusions..."
   106→    try {
   107→        Add-MpPreference -ExclusionPath $scriptPath -ErrorAction Stop
   108→        Write-Success "Excluded: $scriptPath"
   109→        $exclusionAdded = $true
   110→    } catch {
   111→        Write-Warn "Failed to exclude script path: $_"
   112→    }
   113→
   114→    # Exclude the working directory (where the script lives)
   115→    $scriptDir = Split-Path $scriptPath -Parent
   116→    if ($scriptDir -and (Test-Path $scriptDir)) {
   117→        try {
   118→            Add-MpPreference -ExclusionPath $scriptDir -ErrorAction Stop
   119→            Write-Success "Excluded directory: $scriptDir"
   120→            $exclusionAdded = $true
   121→        } catch {
   122→            Write-Warn "Failed to exclude directory: $_"
   123→        }
   124→    }
   125→
   126→    # Exclude the log directory (C:\Temp)
   127→    $logDir = "$env:SYSTEMDRIVE\Temp"
   128→    try {
   129→        Add-MpPreference -ExclusionPath $logDir -ErrorAction Stop
   130→        Write-Success "Excluded log directory: $logDir"
   131→        $exclusionAdded = $true
   132→    } catch {
   133→        Write-Warn "Failed to exclude log directory: $_"
   134→    }
   135→
   136→    # Exclude common technician tool folders
   137→    $toolPaths = @("C:\Tools", "C:\Scripts", "C:\TechnicianTools")
   138→    foreach ($path in $toolPaths) {
   139→        if (Test-Path $path) {
   140→            try {
   141→                Add-MpPreference -ExclusionPath $path -ErrorAction Stop
   142→                Write-Success "Excluded: $path"
   143→            } catch { }
   144→        }
   145→    }
   146→
   147→    if ($exclusionAdded) {
   148→        Write-Success "Defender exclusions configured. This script is now trusted."
   149→    }
   150→
   151→    Write-Host ""
   152→    Write-Host "  NOTE: This only affects Windows Defender. For third-party AV," -ForegroundColor DarkGray
   153→    Write-Host "  consider purchasing a Code Signing Certificate for full compatibility." -ForegroundColor DarkGray
   154→    Write-Host ""
   155→}

   157→function Initialize-Script {
   158→    if (-not ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
   159→        Write-Host "[ERROR] This script must be run as Administrator." -ForegroundColor Red
   160→        Write-Host "Right-click PowerShell and select 'Run as Administrator', then re-run this script." -ForegroundColor Yellow
   161→        throw "Not running as Administrator. Cannot continue."
   162→    }
   163→
   164→    $logDir = "$env:SYSTEMDRIVE\Temp"
   165→    if (-not (Test-Path $logDir)) { New-Item -Path $logDir -ItemType Directory -Force | Out-Null }
   166→
   167→    try { Start-Transcript -Path $LogFile -Force -ErrorAction Stop | Out-Null } catch {
   168→        Write-Warn "Could not start transcript to $LogFile : $_"
   169→    }
   170→
   171→    # Set up Defender self-exclusion FIRST (before any optimization work)
   172→    Set-DefenderExclusions
   173→
   174→    $osInfo = Get-CimInstance -ClassName Win32_OperatingSystem
   175→    $script:OSVersion = [version]$osInfo.Version
   176→    $script:OSBuild = $osInfo.BuildNumber
   177→    $script:IsWin11 = $script:OSBuild -ge 22000
   178→    $script:OSCaption = if ($script:IsWin11) { "Windows 11" } else { "Windows 10" }
   179→    $script:ComputerName = $env:COMPUTERNAME
   180→
   181→    Write-Host ""
   182→    Write-Host "================================================================" -ForegroundColor Cyan
   183→    Write-Host "  Windows Post-Installation Optimizer v$ScriptVersion" -ForegroundColor Cyan
   184→    Write-Host "  Target: $script:OSCaption (Build $script:OSBuild)" -ForegroundColor Cyan
   185→    Write-Host "  Computer: $script:ComputerName" -ForegroundColor Cyan
   186→    Write-Host "  Log file: $LogFile" -ForegroundColor DarkGray
   187→    Write-Host "================================================================" -ForegroundColor Cyan
   188→    Write-Host ""
   189→}

   207→function Set-Registry {
   208→    [CmdletBinding()]
   209→    param(
   210→        [string]$Path,
   211→        [string]$Name,
   212→        $Value,
   213→        [string]$Type = "DWord"
   214→    )
   215→    try {
   216→        if (-not (Test-Path $Path)) {
   217→            New-Item -Path $Path -Force -ErrorAction SilentlyContinue | Out-Null
   218→        }
   219→        Set-ItemProperty -Path $Path -Name $Name -Value $Value -Type $Type -Force -ErrorAction Stop
   220→        Write-Success "Registry: $Path\$Name = $Value"
   221→    }
   222→    catch {
   223→        Write-Err "Failed to set $Path\$Name : $_"
   224→    }
   225→}

   227→function Remove-Registry {
   228→    [CmdletBinding()]
   229→    param(
   230→        [string]$Path,
   231→        [string]$Name
   232→    )
   233→    try {
   234→        if (Test-Path $Path) {
   235→            Remove-ItemProperty -Path $Path -Name $Name -Force -ErrorAction Stop 2>$null
   236→            Write-Success "Registry removed: $Path\$Name"
   237→        }
   238→    }
   239→    catch {
   240→        Write-Warn "Could not remove $Path\$Name : $_"
   241→    }
   242→}

   244→function Disable-ScheduledTask {
   245→    [CmdletBinding()]
   246→    param(
   247→        [string]$TaskPath,
   248→        [string]$TaskName
   249→    )
   250→    try {
   251→        $task = Get-ScheduledTask -TaskPath $TaskPath -TaskName $TaskName -ErrorAction SilentlyContinue
   252→        if ($task) {
   253→            Disable-ScheduledTask -TaskPath $TaskPath -TaskName $TaskName -ErrorAction Stop | Out-Null
   254→            Write-Success "Task disabled: $TaskPath$TaskName"
   255→        }
   256→    }
   257→    catch {
   258→        Write-Warn "Could not disable task $TaskPath$TaskName : $_"
   259→    }
   260→}
