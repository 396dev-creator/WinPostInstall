    63→function Set-DefenderExclusions {
    64→    <#
    65→    .DESCRIPTION
    66→        Adds this script and its working directory to Windows Defender exclusions.
    67→        Prevents the script from being flagged as malware during execution.
    68→        This is safe: exclusions are path-specific and only protect this tool.
    69→    #>
    70→    Write-Host "" -ForegroundColor Cyan
    71→    Write-Host "  [AV SHIELD] Windows Defender Self-Exclusion" -ForegroundColor Cyan
    72→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
    73→    Write-Host ""
    74→
    75→    $scriptPath = $PSCommandPath
    76→    if (-not $scriptPath) { $scriptPath = $MyInvocation.PSCommandPath }
    77→
    78→    $exclusionAdded = $false
    79→
    80→    # Check if the exclusion already exists
    81→    try {
    82→        $existingExclusions = (Get-MpPreference -ErrorAction SilentlyContinue).ExclusionPath
    83→        if ($existingExclusions -and $scriptPath -in $existingExclusions) {
    84→            Write-Step "Defender exclusion already exists for this script."
    85→            return
    86→        }
    87→    } catch {
    88→        Write-Warn "Could not read current Defender preferences: $_"
    89→    }
    90→
    91→    # Exclude the script file itself
    92→    Write-Step "Adding script to Defender exclusions..."
    93→    try {
    94→        Add-MpPreference -ExclusionPath $scriptPath -ErrorAction Stop
    95→        Write-Success "Excluded: $scriptPath"
    96→        $exclusionAdded = $true
    97→    } catch {
    98→        Write-Warn "Failed to exclude script path: $_"
    99→    }
   100→
   101→    # Exclude the working directory (where the script lives)
   102→    $scriptDir = Split-Path $scriptPath -Parent
   103→    if ($scriptDir -and (Test-Path $scriptDir)) {
   104→        try {
   105→            Add-MpPreference -ExclusionPath $scriptDir -ErrorAction Stop
   106→            Write-Success "Excluded directory: $scriptDir"
   107→            $exclusionAdded = $true
   108→        } catch {
   109→            Write-Warn "Failed to exclude directory: $_"
   110→        }
   111→    }
   112→
   113→    # Exclude the log directory (C:\Temp)
   114→    $logDir = "$env:SYSTEMDRIVE\Temp"
   115→    try {
   116→        Add-MpPreference -ExclusionPath $logDir -ErrorAction Stop
   117→        Write-Success "Excluded log directory: $logDir"
   118→        $exclusionAdded = $true
   119→    } catch {
   120→        Write-Warn "Failed to exclude log directory: $_"
   121→    }
   122→
   123→    # Exclude common technician tool folders
   124→    $toolPaths = @("C:\Tools", "C:\Scripts", "C:\TechnicianTools")
   125→    foreach ($path in $toolPaths) {
   126→        if (Test-Path $path) {
   127→            try {
   128→                Add-MpPreference -ExclusionPath $path -ErrorAction Stop
   129→                Write-Success "Excluded: $path"
   130→            } catch { }
   131→        }
   132→    }
   133→
   134→    if ($exclusionAdded) {
   135→        Write-Success "Defender exclusions configured. This script is now trusted."
   136→    }
   137→
   138→    Write-Host ""
   139→    Write-Host "  NOTE: This only affects Windows Defender. For third-party AV," -ForegroundColor DarkGray
   140→    Write-Host "  consider purchasing a Code Signing Certificate for full compatibility." -ForegroundColor DarkGray
   141→    Write-Host ""
   142→}

   144→function Initialize-Script {
   145→    if (-not ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
   146→        Write-Host "[ERROR] This script must be run as Administrator." -ForegroundColor Red
   147→        Write-Host "Right-click PowerShell and select 'Run as Administrator', then re-run this script." -ForegroundColor Yellow
   148→        if (-not $Silent) { Read-Host "Press Enter to exit" }
   149→        exit 1
   150→    }
   151→
   152→    $logDir = "$env:SYSTEMDRIVE\Temp"
   153→    if (-not (Test-Path $logDir)) { New-Item -Path $logDir -ItemType Directory -Force | Out-Null }
   154→
   155→    Start-Transcript -Path $LogFile -Force | Out-Null
   156→
   157→    # Set up Defender self-exclusion FIRST (before any optimization work)
   158→    Set-DefenderExclusions
   159→
   160→    $osInfo = Get-CimInstance -ClassName Win32_OperatingSystem
   161→    $script:OSVersion = [version]$osInfo.Version
   162→    $script:OSBuild = $osInfo.BuildNumber
   163→    $script:IsWin11 = $script:OSBuild -ge 22000
   164→    $script:OSCaption = if ($script:IsWin11) { "Windows 11" } else { "Windows 10" }
   165→    $script:ComputerName = $env:COMPUTERNAME
   166→
   167→    Write-Host ""
   168→    Write-Host "================================================================" -ForegroundColor Cyan
   169→    Write-Host "  Windows Post-Installation Optimizer v$ScriptVersion" -ForegroundColor Cyan
   170→    Write-Host "  Target: $script:OSCaption (Build $script:OSBuild)" -ForegroundColor Cyan
   171→    Write-Host "  Computer: $script:ComputerName" -ForegroundColor Cyan
   172→    Write-Host "  Log file: $LogFile" -ForegroundColor DarkGray
   173→    Write-Host "================================================================" -ForegroundColor Cyan
   174→    Write-Host ""
   175→}

   193→function Set-Registry {
   194→    [CmdletBinding()]
   195→    param(
   196→        [string]$Path,
   197→        [string]$Name,
   198→        $Value,
   199→        [string]$Type = "DWord"
   200→    )
   201→    try {
   202→        if (-not (Test-Path $Path)) {
   203→            New-Item -Path $Path -Force -ErrorAction SilentlyContinue | Out-Null
   204→        }
   205→        Set-ItemProperty -Path $Path -Name $Name -Value $Value -Type $Type -Force -ErrorAction Stop
   206→        Write-Success "Registry: $Path\$Name = $Value"
   207→    }
   208→    catch {
   209→        Write-Err "Failed to set $Path\$Name : $_"
   210→    }
   211→}

   224→    }
   225→    catch {
   226→        Write-Warn "Could not remove $Path\$Name : $_"

  1922→function Get-UserSelection {
  1923→    Show-Menu
  1924→    $input = Read-Host "  Enter section numbers (comma-separated, e.g. '1,3,5') or 'A' for all"
  1925→
  1926→    if ($input -eq '0') {

  1928→        Stop-Transcript | Out-Null
  1929→        exit 0
  1930→    }
  1931→
  1932→    $selected = @()

  1941→            '5'='Performance';'6'='Security';'7'='Network';'8'='WindowsUpdate'

  1947→            else { Write-Warn "Invalid selection: $num (skipped)" }
  1948→        }
  1949→    }
  1950→    return $selected
