   111→function Set-DefenderExclusions {
   112→    <#
   113→    .DESCRIPTION
   114→        Adds this script and its working directory to Windows Defender exclusions.
   115→        Prevents the script from being flagged as malware during execution.
   116→        This is safe: exclusions are path-specific and only protect this tool.
   117→    #>
   118→    Write-Host "" -ForegroundColor Cyan
   119→    Write-Host "  [AV SHIELD] Windows Defender Self-Exclusion" -ForegroundColor Cyan
   120→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   121→    Write-Host ""
   122→
   123→    $scriptPath = $PSCommandPath
   124→    if (-not $scriptPath) { $scriptPath = $MyInvocation.PSCommandPath }
   125→
   126→    $exclusionAdded = $false
   127→
   128→    # Check if the exclusion already exists
   129→    try {
   130→        $existingExclusions = (Get-MpPreference -ErrorAction SilentlyContinue).ExclusionPath
   131→        if ($existingExclusions -and $scriptPath -in $existingExclusions) {
   132→            Write-Step "Defender exclusion already exists for this script."
   133→            return
   134→        }
   135→    } catch {
   136→        Write-Warn "Could not read current Defender preferences: $_"
   137→    }
   138→
   139→    # Exclude the script file itself
   140→    Write-Step "Adding script to Defender exclusions..."
   141→    try {
   142→        Add-MpPreference -ExclusionPath $scriptPath -ErrorAction Stop
   143→        Write-Success "Excluded: $scriptPath"
   144→        $exclusionAdded = $true
   145→    } catch {
   146→        Write-Warn "Failed to exclude script path: $_"
   147→    }
   148→
   149→    # Exclude the working directory (where the script lives)
   150→    $scriptDir = Split-Path $scriptPath -Parent
   151→    if ($scriptDir -and (Test-Path $scriptDir)) {
   152→        try {
   153→            Add-MpPreference -ExclusionPath $scriptDir -ErrorAction Stop
   154→            Write-Success "Excluded directory: $scriptDir"
   155→            $exclusionAdded = $true
   156→        } catch {
   157→            Write-Warn "Failed to exclude directory: $_"
   158→        }
   159→    }
   160→
   161→    # Exclude the log directory (C:\Temp)
   162→    $logDir = "$env:SYSTEMDRIVE\Temp"
   163→    try {
   164→        Add-MpPreference -ExclusionPath $logDir -ErrorAction Stop
   165→        Write-Success "Excluded log directory: $logDir"
   166→        $exclusionAdded = $true
   167→    } catch {
   168→        Write-Warn "Failed to exclude log directory: $_"
   169→    }
   170→
   171→    # Exclude common technician tool folders
   172→    $toolPaths = @("C:\Tools", "C:\Scripts", "C:\TechnicianTools")
   173→    foreach ($path in $toolPaths) {
   174→        if (Test-Path $path) {
   175→            try {
   176→                Add-MpPreference -ExclusionPath $path -ErrorAction Stop
   177→                Write-Success "Excluded: $path"
   178→            } catch { }
   179→        }
   180→    }
   181→
   182→    if ($exclusionAdded) {
   183→        Write-Success "Defender exclusions configured. This script is now trusted."
   184→    }
   185→
   186→    Write-Host ""
   187→    Write-Host "  NOTE: This only affects Windows Defender. For third-party AV," -ForegroundColor DarkGray
   188→    Write-Host "  consider purchasing a Code Signing Certificate for full compatibility." -ForegroundColor DarkGray
   189→    Write-Host ""
   190→}

   250→function Enable-SystemRestore {
   251→    <#
   252→    .DESCRIPTION
   253→        Ensures System Restore is enabled and configured on the system drive.
   254→        Enables the service if disabled, allocates disk space, and verifies
   255→        the feature is operational before returning.
   256→    #>
   257→    Write-Host "" -ForegroundColor Cyan
   258→    Write-Host "  [SAFETY] System Restore Verification" -ForegroundColor Cyan
   259→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   260→    Write-Host ""
   261→
   262→    $systemDrive = $env:SYSTEMDRIVE
   263→    $restoreEnabled = $false
   264→
   265→    # Step 1: Check if System Restore is currently enabled on the system drive
   266→    Write-Step "Checking System Restore status on $systemDrive..."
   267→    try {
   268→        $srStatus = Get-ComputerInfo -Property "SystemRestoreStatus" -ErrorAction SilentlyContinue
   269→        # Also check via WMI which is more reliable
   270→        $srInfo = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction SilentlyContinue
   271→        if ($srInfo) {
   272→            foreach ($sr in $srInfo) {
   273→                if ($sr.RPSequence -gt 0) {
   274→                    $restoreEnabled = $true
   275→                    break
   276→                }
   277→            }
   278→        }
   279→    } catch {
   280→        # WMI query may fail if System Restore is completely disabled
   281→        $restoreEnabled = $false
   282→    }
   283→
   284→    # Alternative check: try to list existing restore points
   285→    if (-not $restoreEnabled) {
   286→        try {
   287→            $existingPoints = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction Stop
   288→            if ($existingPoints -and $existingPoints.Count -ge 0) {
   289→                $restoreEnabled = $true
   290→            }
   291→        } catch {
   292→            $restoreEnabled = $false
   293→        }
   294→    }
   295→
   296→    # Step 2: Enable System Restore if not currently enabled
   297→    if (-not $restoreEnabled) {
   298→        Write-Warn "System Restore is NOT enabled on $systemDrive."
   299→        Write-Step "Enabling System Restore on $systemDrive..."
   300→
   301→        try {
   302→            # Enable System Restore via registry (most reliable method)
   303→            $srPolicyPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore"
   304→            if (-not (Test-Path $srPolicyPath)) {
   305→                New-Item -Path $srPolicyPath -Force -ErrorAction SilentlyContinue | Out-Null
   306→            }
   307→            # DisableSR = 0 means System Restore is ENABLED
   308→            Set-ItemProperty -Path $srPolicyPath -Name "DisableSR" -Value 0 -Type DWord -Force -ErrorAction Stop
   309→
   310→            # Also enable via PowerShell cmdlet (Windows 10/11)
   311→            try {
   312→                Enable-ComputerRestore -Drive $systemDrive -ErrorAction Stop
   313→            } catch {
   314→                # Fallback: use vssadmin to enable shadow copies
   315→                $vssResult = vssadmin list shadows 2>&1
   316→                if ($LASTEXITCODE -ne 0) {
   317→                    # Try creating a shadow copy to force-enable VSS
   318→                    vssadmin create shadow /For=$systemDrive 2>&1 | Out-Null
   319→                }
   320→            }
   321→
   322→            # Start the System Restore service if not running
   323→            $srService = Get-Service -Name "sr" -ErrorAction SilentlyContinue
   324→            if ($srService -and $srService.Status -ne 'Running') {
   325→                try {
   326→                    Set-Service -Name "sr" -StartupType Manual -ErrorAction Stop
   327→                    Start-Service -Name "sr" -ErrorAction Stop
   328→                } catch {
   329→                    Write-Warn "Could not start System Restore service: $_"
   330→                }
   331→            }
   332→
   333→            # Also ensure VSS (Volume Shadow Copy) service is running (required for System Restore)
   334→            $vssService = Get-Service -Name "VSS" -ErrorAction SilentlyContinue
   335→            if ($vssService -and $vssService.Status -ne 'Running') {
   336→                try {
   337→                    Set-Service -Name "VSS" -StartupType Manual -ErrorAction Stop
   338→                    Start-Service -Name "VSS" -ErrorAction Stop
   339→                    Write-Success "VSS service started."
   340→                } catch {
   341→                    Write-Warn "Could not start VSS service: $_"
   342→                }
   343→            }
   344→
   345→            Start-Sleep -Seconds 2
   346→            Write-Success "System Restore has been ENABLED on $systemDrive."
   347→        } catch {
   348→            Write-Err "Failed to enable System Restore: $_"
   349→            Write-Warn "Continuing without restore point. Manual enable: System Properties > System Protection > Configure > Turn on"
   350→        }
   351→    } else {
   352→        Write-Success "System Restore is already enabled on $systemDrive."
   353→    }
   354→
   355→    # Step 3: Configure disk space allocation for restore points
   356→    Write-Step "Configuring System Restore disk space..."
   357→    try {
   358→        # Set maximum disk space usage to 5% (reasonable for most systems)
   359→        # This is done via registry: RPSessionInterval and DiskPercent
   360→        $srCfgPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore\Cfg"
   361→        if (-not (Test-Path $srCfgPath)) {
   362→            New-Item -Path $srCfgPath -Force -ErrorAction SilentlyContinue | Out-Null
   363→        }
   364→        # DiskPercent = 5 means 5% of disk (0 = unlimited, we use 5% as safe default)
   365→        Set-ItemProperty -Path $srCfgPath -Name "DiskPercent" -Value 5 -Type DWord -Force -ErrorAction SilentlyContinue
   366→        # RPSessionInterval = 0 means create restore points at any interval
   367→        Set-ItemProperty -Path $srCfgPath -Name "RPSessionInterval" -Value 0 -Type DWord -Force -ErrorAction SilentlyContinue
   368→        Write-Success "Restore disk space set to 5% of drive."
   369→    } catch {
   370→        Write-Warn "Could not configure restore disk space: $_"
   371→    }
   372→
   373→    Write-Host ""
   374→}

   376→function New-SystemRestorePoint {
   377→    <#
   378→    .DESCRIPTION
   379→        Creates a System Restore point before any optimization changes are made.
   380→        This provides a safety net so the user can roll back all changes.
   381→    #>
   382→    Write-Host "" -ForegroundColor Cyan
   383→    Write-Host "  [SAFETY] Creating System Restore Point" -ForegroundColor Cyan
   384→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   385→    Write-Host ""
   386→
   387→    $restorePointCreated = $false
   388→    $description = "Pre-Optimization Restore Point - WinPostInstall v$ScriptVersion - $(Get-Date -Format 'yyyy-MM-dd HH:mm')"
   389→
   390→    # Method 1: Use Checkpoint-Computer (PowerShell cmdlet, Windows 8+)
   391→    Write-Step "Creating restore point: 'Pre-Optimization'..."
   392→    try {
   393→        Checkpoint-Computer -Description $description -RestorePointType "MODIFY_SETTINGS" -ErrorAction Stop
   394→        $restorePointCreated = $true
   395→        Write-Success "Restore point created successfully via Checkpoint-Computer."
   396→    } catch {
   397→        Write-Warn "Checkpoint-Computer failed: $_"
   398→    }
   399→
   400→    # Method 2: Fallback using WMI if Method 1 failed
   401→    if (-not $restorePointCreated) {
   402→        Write-Step "Trying WMI method to create restore point..."
   403→        try {
   404→            $wmiSR = [System.Windows.Forms.SystemRestore]
   405→            # WMI approach via Invoke-CimMethod
   406→            $result = Invoke-CimMethod -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" `
   407→                -MethodName "CreateRestorePoint" `
   408→                -Arguments @{
   409→                    Description         = $description
   410→                    RestorePointType    = 12  # MODIFY_SETTINGS
   411→                    EventType           = 100  # BEGIN_NESTED_SYSTEM_CHANGE
   412→                } -ErrorAction Stop
   413→            if ($result -and $result.ReturnValue -eq 0) {
   414→                $restorePointCreated = $true
   415→                Write-Success "Restore point created via WMI."
   416→            } else {
   417→                Write-Warn "WMI returned non-zero: $result"
   418→            }
   419→        } catch {
   420→            Write-Warn "WMI method failed: $_"
   421→        }
   422→    }
   423→
   424→    # Method 3: Final fallback via srclient.dll COM automation
   425→    if (-not $restorePointCreated) {
   426→        Write-Step "Trying COM automation (srclient.dll)..."
   427→        try {
   428→            $srClient = New-Object -ComObject "SRClient.RestorePoint" -ErrorAction Stop
   429→            # 7 = MODIFY_SETTINGS, 100 = BEGIN_NESTED_SYSTEM_CHANGE
   430→            $srClient.CreateRestorePoint($description, 7, 100)
   431→            $restorePointCreated = $true
   432→            Write-Success "Restore point created via COM automation."
   433→        } catch {
   434→            Write-Warn "COM method failed: $_"
   435→        }
   436→    }
   437→
   438→    if ($restorePointCreated) {
   439→        Write-Host ""
   440→        Write-Host "  ====================================================================" -ForegroundColor Green
   441→        Write-Host "  RESTORE POINT CREATED: 'Pre-Optimization'" -ForegroundColor Green
   442→        Write-Host "  You can roll back all changes via:" -ForegroundColor White
   443→        Write-Host "    System Properties > System Protection > System Restore" -ForegroundColor White
   444→        Write-Host "  ====================================================================" -ForegroundColor Green
   445→        Write-Host ""
   446→    } else {
   447→        Write-Host ""
   448→        Write-Warn "Could NOT create restore point automatically."
   449→        Write-Host "  Manual option: System Properties > System Protection > Create" -ForegroundColor Yellow
   450→        Write-Host "  Continuing without restore point backup." -ForegroundColor Yellow
   451→        Write-Host ""
   452→        if (-not $Silent) {
   453→            $continue = Read-Host "  Continue without restore point? (y/N)"
   454→            if ($continue -ne 'y' -and $continue -ne 'Y') {
   455→                Write-Host "  Aborted by user." -ForegroundColor Yellow
   456→                exit 0
   457→            }
   458→        }
   459→    }
   460→}

   462→function Set-Registry {
   463→    [CmdletBinding()]
   464→    param(
   465→        [string]$Path,
   466→        [string]$Name,
   467→        $Value,
   468→        [string]$Type = "DWord"
   469→    )
   470→    try {
   471→        if (-not (Test-Path $Path)) {
   472→            New-Item -Path $Path -Force -ErrorAction SilentlyContinue | Out-Null
   473→        }
   474→        # Idempotency: Check if the value already matches what we want to set
   475→        $currentValue = $null
   476→        try {
   477→            $currentValue = (Get-ItemProperty -Path $Path -Name $Name -ErrorAction Stop).$Name
   478→        } catch {
   479→            # Property doesn't exist yet — need to create it
   480→        }
   481→
   482→        # Compare current vs desired value
   483→        $valueMatches = $false
   484→        if ($null -ne $currentValue) {
   485→            if ($Type -eq "DWord" -or $Type -eq "QWord") {
   486→                # Numeric comparison (handle both int and string inputs)
   487→                $valueMatches = ([int]$currentValue -eq [int]$Value)
   488→            } elseif ($Type -eq "String" -or $Type -eq "ExpandString") {
   489→                $valueMatches = ([string]$currentValue -eq [string]$Value)
   490→            } else {
   491→                $valueMatches = ($currentValue -eq $Value)
   492→            }
   493→        }
   494→
   495→        if ($valueMatches) {
   496→            Write-Step "Already set: $Path\$Name = $Value"
   497→            return
   498→        }
   499→
   500→        Set-ItemProperty -Path $Path -Name $Name -Value $Value -Type $Type -Force -ErrorAction Stop
   501→        Write-Success "Registry: $Path\$Name = $Value"
   502→    }
   503→    catch {
   504→        Write-Err "Failed to set $Path\$Name : $_"
   505→    }
   506→}

   508→function Remove-Registry {
   509→    [CmdletBinding()]
   510→    param(
   511→        [string]$Path,
   512→        [string]$Name
   513→    )
   514→    try {
   515→        if (Test-Path $Path) {
   516→            # Idempotency: Check if property exists before removing
   517→            $existing = $null
   518→            try { $existing = (Get-ItemProperty -Path $Path -Name $Name -ErrorAction Stop).$Name } catch { }
   519→            if ($null -ne $existing) {
   520→                Remove-ItemProperty -Path $Path -Name $Name -Force -ErrorAction Stop 2>$null
   521→                Write-Success "Registry removed: $Path\$Name"
   522→            } else {
   523→                Write-Step "Already removed: $Path\$Name"
   524→            }
   525→        }
   526→    }
   527→    catch {
   528→        Write-Warn "Could not remove $Path\$Name : $_"
   529→    }
   530→}

   532→function Disable-ScheduledTask {
   533→    [CmdletBinding()]
   534→    param(
   535→        [string]$TaskPath,
   536→        [string]$TaskName
   537→    )
   538→    try {
   539→        $task = Get-ScheduledTask -TaskPath $TaskPath -TaskName $TaskName -ErrorAction SilentlyContinue
   540→        if ($task) {
