   169→function Set-DefenderExclusions {
   170→    <#
   171→    .DESCRIPTION
   172→        Adds this script and its working directory to Windows Defender exclusions.
   173→        Prevents the script from being flagged as malware during execution.
   174→        This is safe: exclusions are path-specific and only protect this tool.
   175→    #>
   176→    Write-Host "" -ForegroundColor Cyan
   177→    Write-Host "  [AV SHIELD] Windows Defender Self-Exclusion" -ForegroundColor Cyan
   178→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   179→    Write-Host ""
   180→
   181→    $scriptPath = $PSCommandPath
   182→    if (-not $scriptPath) { $scriptPath = $MyInvocation.PSCommandPath }
   183→
   184→    $exclusionAdded = $false
   185→
   186→    # Check if the exclusion already exists
   187→    try {
   188→        $existingExclusions = (Get-MpPreference -ErrorAction SilentlyContinue).ExclusionPath
   189→        if ($existingExclusions -and $scriptPath -in $existingExclusions) {
   190→            Write-Step "Defender exclusion already exists for this script."
   191→            return
   192→        }
   193→    } catch {
   194→        Write-Warn "Could not read current Defender preferences: $_"
   195→    }
   196→
   197→    # Exclude the script file itself
   198→    Write-Step "Adding script to Defender exclusions..."
   199→    try {
   200→        Add-MpPreference -ExclusionPath $scriptPath -ErrorAction Stop
   201→        Write-Success "Excluded: $scriptPath"
   202→        $exclusionAdded = $true
   203→    } catch {
   204→        Write-Warn "Failed to exclude script path: $_"
   205→    }
   206→
   207→    # Exclude the working directory (where the script lives)
   208→    $scriptDir = Split-Path $scriptPath -Parent
   209→    if ($scriptDir -and (Test-Path $scriptDir)) {
   210→        try {
   211→            Add-MpPreference -ExclusionPath $scriptDir -ErrorAction Stop
   212→            Write-Success "Excluded directory: $scriptDir"
   213→            $exclusionAdded = $true
   214→        } catch {
   215→            Write-Warn "Failed to exclude directory: $_"
   216→        }
   217→    }
   218→
   219→    # Exclude the log directory (C:\Temp)
   220→    $logDir = "$env:SYSTEMDRIVE\Temp"
   221→    try {
   222→        Add-MpPreference -ExclusionPath $logDir -ErrorAction Stop
   223→        Write-Success "Excluded log directory: $logDir"
   224→        $exclusionAdded = $true
   225→    } catch {
   226→        Write-Warn "Failed to exclude log directory: $_"
   227→    }
   228→
   229→    # Exclude common technician tool folders
   230→    $toolPaths = @("C:\Tools", "C:\Scripts", "C:\TechnicianTools")
   231→    foreach ($path in $toolPaths) {
   232→        if (Test-Path $path) {
   233→            try {
   234→                Add-MpPreference -ExclusionPath $path -ErrorAction Stop
   235→                Write-Success "Excluded: $path"
   236→            } catch { }
   237→        }
   238→    }
   239→
   240→    if ($exclusionAdded) {
   241→        Write-Success "Defender exclusions configured. This script is now trusted."
   242→    }
   243→
   244→    Write-Host ""
   245→    Write-Host "  NOTE: This only affects Windows Defender. For third-party AV," -ForegroundColor DarkGray
   246→    Write-Host "  consider purchasing a Code Signing Certificate for full compatibility." -ForegroundColor DarkGray
   247→    Write-Host ""
   248→}

   319→        }
   320→    } catch { }
   321→    if (-not $script:IsLaptop) {
   322→        try {
   323→            $script:IsLaptop = (Get-CimInstance -ClassName Win32_Battery -ErrorAction SilentlyContinue).Count -gt 0
   324→        } catch { }
   325→    }
   326→
   327→    Write-Host ""
   328→    Write-Host "================================================================" -ForegroundColor Cyan
   329→    Write-Host "  Windows Post-Installation Optimizer v$ScriptVersion" -ForegroundColor Cyan
   330→    Write-Host "  Target: $script:OSCaption (Build $script:OSBuild)" -ForegroundColor Cyan
   331→    Write-Host "  Computer: $script:LocalComputerName" -ForegroundColor Cyan
   332→    Write-Host "  Chassis: $(if ($script:IsLaptop) { 'Laptop' } else { 'Desktop' })" -ForegroundColor Cyan
   333→    Write-Host "  Log file: $LogFile" -ForegroundColor DarkGray
   334→    Write-Host "================================================================" -ForegroundColor Cyan
   335→    Write-Host ""

   338→    Test-Prerequisites
   339→}

   341→function Test-Prerequisites {
   342→    <#
   343→    .DESCRIPTION
   344→        Pre-flight checks: internet connectivity, disk space, pending reboot, battery.
   345→        Warns on issues but does not block execution.
   346→    #>
   347→    Write-Host "" -ForegroundColor Cyan
   348→    Write-Host "  [PRE-FLIGHT] System Prerequisites Check" -ForegroundColor Cyan
   349→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   350→    Write-Host ""
   351→
   352→    $checksPassed = 0; $checksWarned = 0
   353→
   354→    # 1. Internet connectivity
   355→    Write-Step "Checking internet connectivity..."
   356→    try {
   357→        $netTest = Test-Connection -ComputerName 8.8.8.8 -Count 1 -Quiet -ErrorAction Stop
   358→        if ($netTest) {
   359→            Write-Success "Internet connectivity: OK"
   360→            $checksPassed++
   361→        } else {
   362→            Write-Warn "Internet connectivity: OFFLINE (some features may fail)"
   363→            $checksWarned++
   364→        }
   365→    } catch {
   366→        Write-Warn "Internet connectivity: Could not check (some features may fail)"
   367→        $checksWarned++
   368→    }
   369→
   370→    # 2. Free disk space
   371→    Write-Step "Checking free disk space..."
   372→    try {
   373→        $sysDrive = Get-PSDrive -Name ($env:SYSTEMDRIVE -replace ':','') -ErrorAction Stop
   374→        $freeGB = [math]::Round($sysDrive.Free / 1GB, 2)
   375→        if ($freeGB -gt 10) {
   376→            Write-Success "Free disk space: $freeGB GB (OK)"
   377→            $checksPassed++
   378→        } else {
   379→            Write-Warn "Free disk space: $freeGB GB (LOW - recommend >10 GB)"
   380→            $checksWarned++
   381→        }
   382→    } catch {
   383→        Write-Warn "Could not check disk space: $_"
   384→        $checksWarned++
   385→    }
   386→
   387→    # 3. Pending reboot
   388→    Write-Step "Checking for pending reboot..."
   389→    $rebootPending = $false
   390→    $cbsPath = "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Component Based Servicing\RebootPending"
   391→    $wuPath = "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\WindowsUpdate\Auto Update\RebootRequired"
   392→    if (Test-Path $cbsPath -ErrorAction SilentlyContinue) { $rebootPending = $true }
   393→    if (Test-Path $wuPath -ErrorAction SilentlyContinue) { $rebootPending = $true }
   394→    if ($rebootPending) {
   395→        Write-Warn "Pending reboot detected! A restart is recommended before running optimizations."
   396→        $checksWarned++
   397→    } else {
   398→        Write-Success "No pending reboot."
   399→        $checksPassed++
   400→    }
   401→
   402→    # 4. Battery check (if laptop)
   403→    if ($script:IsLaptop) {
   404→        Write-Step "Checking battery level..."
   405→        try {
   406→            $battery = Get-CimInstance -ClassName Win32_Battery -ErrorAction Stop
   407→            $onAC = $battery.BatteryStatus -eq 2 -or $battery.BatteryStatus -eq 6 -or $battery.BatteryStatus -eq 7 -or $battery.BatteryStatus -eq 8
   408→            if ($onAC) {
   409→                Write-Success "On AC power."
   410→                $checksPassed++
   411→            } elseif ($battery.EstimatedChargeRemaining -gt 50) {
   412→                Write-Success "Battery level: $($battery.EstimatedChargeRemaining)% (OK)"
   413→                $checksPassed++
   414→            } else {
   415→                Write-Warn "Battery level: $($battery.EstimatedChargeRemaining)% - recommend AC power or >50% charge."
   416→                $checksWarned++
   417→            }
   418→        } catch {
   419→            Write-Warn "Could not check battery: $_"
   420→            $checksWarned++
   421→        }
   422→    }
   423→
   424→    Write-Host ""
   425→    Write-Host "  Pre-flight summary: $checksPassed passed, $checksWarned warnings" -ForegroundColor $(if ($checksWarned -gt 0) { 'Yellow' } else { 'Green' })
   426→    Write-Host ""
   427→}

   429→function Get-BootDriveType {
   430→    <#
   431→    .DESCRIPTION
   432→        Detects whether the boot drive is an SSD or HDD.
   433→        Returns "SSD" or "HDD".
   434→    #>
   435→    $bootType = "HDD"  # Default assumption
   436→    try {
   437→        # Method 1: Get-PhysicalDisk (most reliable on Win8+)
   438→        $physicalDisk = Get-PhysicalDisk -ErrorAction SilentlyContinue | Where-Object DeviceId -eq 0
   439→        if ($physicalDisk) {
   440→            if ($physicalDisk.MediaType -eq "SSD") { return "SSD" }
   441→            if ($physicalDisk.MediaType -eq "HDD") { return "HDD" }
   442→        }
   443→    } catch { }
   444→
   445→    try {
   446→        # Method 2: Get-Disk BusType fallback
   447→        $bootDisk = Get-Disk -ErrorAction SilentlyContinue | Where-Object IsBoot -eq $true
   448→        if ($bootDisk) {
   449→            $ssdBusTypes = @("NVMe", "SATA", "RAID", "SCSI")
   450→            # Check via rotation rate (0 = SSD on some controllers)
   451→            if ($bootDisk.BusType -in $ssdBusTypes) {
   452→                # Not conclusive, but check rotation rate
   453→                $detail = Get-PhysicalDisk -ErrorAction SilentlyContinue | Where-Object DeviceId -eq $bootDisk.Number
   454→                if ($detail -and $detail.MediaType -eq "SSD") { return "SSD" }
   455→            }
   456→        }
   457→    } catch { }
   458→
   459→    try {
   460→        # Method 3: Check if defrag is scheduled (SSD = not scheduled)
   461→        $defragTask = Get-ScheduledTask -TaskName "ScheduledDefrag" -ErrorAction SilentlyContinue
   462→        if ($defragTask -and $defragTask.State -eq 'Disabled') {
   463→            # If scheduled defrag is disabled, likely SSD
   464→            # This is a weak heuristic, don't rely on it alone
   465→        }
   466→    } catch { }
   467→
   468→    return $bootType
   469→}

   503→    Write-Step "Checking System Restore status on $systemDrive..."
   504→    try {
   505→        $srStatus = Get-ComputerInfo -Property "SystemRestoreStatus" -ErrorAction SilentlyContinue
   506→        # Also check via WMI which is more reliable
   507→        $srInfo = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction SilentlyContinue
   508→        if ($srInfo) {
   509→            foreach ($sr in $srInfo) {
   510→                if ($sr.RPSequence -gt 0) {
   511→                    $restoreEnabled = $true
   512→                    break
   513→                }
   514→            }
   515→        }
   516→    } catch {
   517→        # WMI query may fail if System Restore is completely disabled
   518→        $restoreEnabled = $false
   519→    }
   520→
   521→    # Alternative check: try to list existing restore points
   522→    if (-not $restoreEnabled) {
   523→        try {
   524→            $existingPoints = Get-CimInstance -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" -ErrorAction Stop
   525→            if ($existingPoints -and $existingPoints.Count -ge 0) {
   526→                $restoreEnabled = $true
   527→            }
   528→        } catch {
   529→            $restoreEnabled = $false
   530→        }
   531→    }
   532→
   533→    # Step 2: Enable System Restore if not currently enabled
   534→    if (-not $restoreEnabled) {
   535→        Write-Warn "System Restore is NOT enabled on $systemDrive."
   536→        Write-Step "Enabling System Restore on $systemDrive..."
   537→
   538→        try {
   539→            # Enable System Restore via registry (most reliable method)
   540→            $srPolicyPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore"
   541→            if (-not (Test-Path $srPolicyPath)) {
   542→                New-Item -Path $srPolicyPath -Force -ErrorAction SilentlyContinue | Out-Null
   543→            }
   544→            # DisableSR = 0 means System Restore is ENABLED
   545→            Set-ItemProperty -Path $srPolicyPath -Name "DisableSR" -Value 0 -Type DWord -Force -ErrorAction Stop
   546→
   547→            # Also enable via PowerShell cmdlet (Windows 10/11)
   548→            try {
   549→                Enable-ComputerRestore -Drive $systemDrive -ErrorAction Stop
   550→            } catch {
   551→                # Fallback: use vssadmin to enable shadow copies
   552→                $vssResult = vssadmin list shadows 2>&1
   553→                if ($LASTEXITCODE -ne 0) {
   554→                    # Try creating a shadow copy to force-enable VSS
   555→                    vssadmin create shadow /For=$systemDrive 2>&1 | Out-Null
   556→                }
   557→            }
   558→
   559→            # Start the System Restore service if not running
   560→            $srService = Get-Service -Name "sr" -ErrorAction SilentlyContinue
   561→            if ($srService -and $srService.Status -ne 'Running') {
   562→                try {
   563→                    Set-Service -Name "sr" -StartupType Manual -ErrorAction Stop
   564→                    Start-Service -Name "sr" -ErrorAction Stop
   565→                } catch {
   566→                    Write-Warn "Could not start System Restore service: $_"
   567→                }
   568→            }
   569→
   570→            # Also ensure VSS (Volume Shadow Copy) service is running (required for System Restore)
   571→            $vssService = Get-Service -Name "VSS" -ErrorAction SilentlyContinue
   572→            if ($vssService -and $vssService.Status -ne 'Running') {
   573→                try {
   574→                    Set-Service -Name "VSS" -StartupType Manual -ErrorAction Stop
   575→                    Start-Service -Name "VSS" -ErrorAction Stop
   576→                    Write-Success "VSS service started."
   577→                } catch {
   578→                    Write-Warn "Could not start VSS service: $_"
   579→                }
   580→            }
   581→
   582→            Start-Sleep -Seconds 2
   583→            Write-Success "System Restore has been ENABLED on $systemDrive."
   584→        } catch {
   585→            Write-Err "Failed to enable System Restore: $_"
   586→            Write-Warn "Continuing without restore point. Manual enable: System Properties > System Protection > Configure > Turn on"
   587→        }
   588→    } else {
   589→        Write-Success "System Restore is already enabled on $systemDrive."
   590→    }
   591→
   592→    # Step 3: Configure disk space allocation for restore points
   593→    Write-Step "Configuring System Restore disk space..."
   594→    try {
   595→        # Set maximum disk space usage to 5% (reasonable for most systems)
   596→        # This is done via registry: RPSessionInterval and DiskPercent
   597→        $srCfgPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore\Cfg"
   598→        if (-not (Test-Path $srCfgPath)) {
   599→            New-Item -Path $srCfgPath -Force -ErrorAction SilentlyContinue | Out-Null
   600→        }
   601→        # DiskPercent = 5 means 5% of disk (0 = unlimited, we use 5% as safe default)
   602→        Set-ItemProperty -Path $srCfgPath -Name "DiskPercent" -Value 5 -Type DWord -Force -ErrorAction SilentlyContinue
   603→        # RPSessionInterval = 0 means create restore points at any interval
   604→        Set-ItemProperty -Path $srCfgPath -Name "RPSessionInterval" -Value 0 -Type DWord -Force -ErrorAction SilentlyContinue
   605→        Write-Success "Restore disk space set to 5% of drive."
   606→    } catch {
   607→        Write-Warn "Could not configure restore disk space: $_"
   608→    }
   609→
   610→    Write-Host ""
   611→}

   613→function New-SystemRestorePoint {
   614→    <#
   615→    .DESCRIPTION
   616→        Creates a System Restore point before any optimization changes are made.
   617→        This provides a safety net so the user can roll back all changes.
   618→    #>
   619→    Write-Host "" -ForegroundColor Cyan
   620→    Write-Host "  [SAFETY] Creating System Restore Point" -ForegroundColor Cyan
   621→    Write-Host "  " + ("-" * 50) -ForegroundColor DarkGray
   622→    Write-Host ""
   623→
   624→    $restorePointCreated = $false
   625→    $description = "Pre-Optimization Restore Point - WinPostInstall v$ScriptVersion - $(Get-Date -Format 'yyyy-MM-dd HH:mm')"
   626→
   627→    # Method 1: Use Checkpoint-Computer (PowerShell cmdlet, Windows 8+)
   628→    Write-Step "Creating restore point: 'Pre-Optimization'..."
   629→    try {
   630→        Checkpoint-Computer -Description $description -RestorePointType "MODIFY_SETTINGS" -ErrorAction Stop
   631→        $restorePointCreated = $true
   632→        Write-Success "Restore point created successfully via Checkpoint-Computer."
   633→    } catch {
   634→        Write-Warn "Checkpoint-Computer failed: $_"
   635→    }
   636→
   637→    # Method 2: Fallback using WMI if Method 1 failed
   638→    if (-not $restorePointCreated) {
   639→        Write-Step "Trying WMI method to create restore point..."
   640→        try {
   641→            $wmiSR = [System.Windows.Forms.SystemRestore]
   642→            # WMI approach via Invoke-CimMethod
   643→            $result = Invoke-CimMethod -Namespace "ROOT/DEFAULT" -ClassName "SystemRestore" `
   644→                -MethodName "CreateRestorePoint" `
   645→                -Arguments @{
   646→                    Description         = $description
   647→                    RestorePointType    = 12  # MODIFY_SETTINGS
   648→                    EventType           = 100  # BEGIN_NESTED_SYSTEM_CHANGE
   649→                } -ErrorAction Stop
   650→            if ($result -and $result.ReturnValue -eq 0) {
   651→                $restorePointCreated = $true
   652→                Write-Success "Restore point created via WMI."
   653→            } else {
   654→                Write-Warn "WMI returned non-zero: $result"
   655→            }
   656→        } catch {
   657→            Write-Warn "WMI method failed: $_"
   658→        }
   659→    }
   660→
   661→    # Method 3: Final fallback via srclient.dll COM automation
   662→    if (-not $restorePointCreated) {
   663→        Write-Step "Trying COM automation (srclient.dll)..."
   664→        try {
   665→            $srClient = New-Object -ComObject "SRClient.RestorePoint" -ErrorAction Stop
   666→            # 7 = MODIFY_SETTINGS, 100 = BEGIN_NESTED_SYSTEM_CHANGE
   667→            $srClient.CreateRestorePoint($description, 7, 100)
   668→            $restorePointCreated = $true
   669→            Write-Success "Restore point created via COM automation."
   670→        } catch {
   671→            Write-Warn "COM method failed: $_"
   672→        }
   673→    }
   674→
   675→    if ($restorePointCreated) {
   676→        Write-Host ""
   677→        Write-Host "  ====================================================================" -ForegroundColor Green
   678→        Write-Host "  RESTORE POINT CREATED: 'Pre-Optimization'" -ForegroundColor Green
   679→        Write-Host "  You can roll back all changes via:" -ForegroundColor White
   680→        Write-Host "    System Properties > System Protection > System Restore" -ForegroundColor White
   681→        Write-Host "  ====================================================================" -ForegroundColor Green
   682→        Write-Host ""
   683→    } else {
   684→        Write-Host ""
   685→        Write-Warn "Could NOT create restore point automatically."
   686→        Write-Host "  Manual option: System Properties > System Protection > Create" -ForegroundColor Yellow
   687→        Write-Host "  Continuing without restore point backup." -ForegroundColor Yellow
   688→        Write-Host ""
   689→        if (-not $Silent) {
   690→            $continue = Read-Host "  Continue without restore point? (y/N)"
   691→            if ($continue -ne 'y' -and $continue -ne 'Y') {
   692→                Write-Host "  Aborted by user." -ForegroundColor Yellow
   693→                exit 0
   694→            }
   695→        }
   696→    }
   697→}

   699→function Set-Registry {
   700→    [CmdletBinding()]
   701→    param(
   702→        [string]$Path,
   703→        [string]$Name,
   704→        $Value,
   705→        [string]$Type = "DWord",
   706→        [ValidateSet("Safe","Moderate","Risky")]
   707→        [string]$SafetyLevel = "Safe",
   708→        [string]$Description = "",
   709→        [string]$SettingID = ""
   710→    )
   711→
   712→    # Generate a setting ID if not provided
   713→    if ([string]::IsNullOrEmpty($SettingID)) {
   714→        $SettingID = "$($Path.Replace('\','_').Replace(':','').Replace(' ',''))_$Name"
   715→    }
   716→
   717→    # Register in setting catalog for interactive menu
   718→    if (-not [string]::IsNullOrEmpty($Description)) {
   719→        $script:SettingCatalog[$SettingID] = @{
   720→            Section  = $script:CurrentSection
   721→            Safety   = $SafetyLevel
   722→            Desc     = $Description
   723→            Action   = "$Path\$Name = $Value"
   724→            Type     = "Registry"
   725→        }
   726→    }
   727→
   728→    # Collect risky descriptions for menu warnings
   729→    if ($SafetyLevel -eq "Risky" -and -not [string]::IsNullOrEmpty($Description)) {
   730→        $script:RiskyDescriptions += "$SettingID : $Description"
   731→    }
   732→
   733→    # Check if user has toggled this setting OFF via interactive menu
   734→    if ($script:SettingOverrides.ContainsKey($SettingID) -and $script:SettingOverrides[$SettingID]) {
   735→        Write-Host "  SKIPPED (user toggle): $Description" -ForegroundColor DarkGray
   736→        $script:TotalSkipped++
   737→        return
   738→    }
   739→
   740→    # Safety gate: Skip Risky settings unless explicitly allowed
   741→    if ($SafetyLevel -eq "Risky" -and -not $script:AllowRisky) {
   742→        Write-Host "  SKIPPED (risky): $Description — use -AllowRisky or enable in menu" -ForegroundColor Yellow
   743→        $script:TotalSkipped++
   744→        return
   745→    }
   746→
   747→    # Warn for Moderate settings
   748→    if ($SafetyLevel -eq "Moderate") {
   749→        Write-Host "  [MODERATE] $Description" -ForegroundColor DarkYellow
   750→    }
   751→
   752→    try {
   753→        if (-not (Test-Path $Path)) {
   754→            New-Item -Path $Path -Force -ErrorAction SilentlyContinue | Out-Null
   755→        }
   756→        # Idempotency: Check if the value already matches what we want to set
   757→        $currentValue = $null
   758→        try {
   759→            $currentValue = (Get-ItemProperty -Path $Path -Name $Name -ErrorAction Stop).$Name
   760→        } catch {
   761→            # Property doesn't exist yet — need to create it
   762→        }
   763→
   764→        # Compare current vs desired value
   765→        $valueMatches = $false
   766→        if ($null -ne $currentValue) {
   767→            if ($Type -eq "DWord" -or $Type -eq "QWord") {
   768→                # Numeric comparison (handle both int and string inputs)
   769→                $valueMatches = ([int]$currentValue -eq [int]$Value)
   770→            } elseif ($Type -eq "String" -or $Type -eq "ExpandString") {
   771→                $valueMatches = ([string]$currentValue -eq [string]$Value)
   772→            } else {
   773→                $valueMatches = ($currentValue -eq $Value)
   774→            }
   775→        }
   776→
   777→        if ($valueMatches) {
   778→            Write-Step "Already set: $Path\$Name = $Value"
   779→            $script:TotalSkipped++
   780→            return
   781→        }
   782→
   783→        # Capture old value for undo script before writing
   784→        if ($null -ne $currentValue) {
   785→            $oldVal = $currentValue.ToString()
   786→            $null = $script:UndoCommands.Add("Set-ItemProperty -Path '$Path' -Name '$Name' -Value '$oldVal' -Type '$Type' -Force")
   787→            $null = $script:UndoDescriptions.Add("Restore registry: $Path\$Name = $oldVal")
   788→        } else {
   789→            $null = $script:UndoCommands.Add("Remove-ItemProperty -Path '$Path' -Name '$Name' -Force -ErrorAction SilentlyContinue")
   790→            $null = $script:UndoDescriptions.Add("Remove registry key that was added: $Path\$Name")
   791→        }
   792→
   793→        Set-ItemProperty -Path $Path -Name $Name -Value $Value -Type $Type -Force -ErrorAction Stop
   794→        Write-Success "Registry: $Path\$Name = $Value"
   795→        $script:TotalChanged++
   796→    }
   797→    catch {
   798→        Write-Err "Failed to set $Path\$Name : $_"
   799→        $script:TotalFailed++
   800→    }
   801→}

   803→function Remove-Registry {
   804→    [CmdletBinding()]
   805→    param(
   806→        [string]$Path,
   807→        [string]$Name
   808→    )
   809→    try {
   810→        if (Test-Path $Path) {
   811→            # Idempotency: Check if property exists before removing
   812→            $existing = $null
   813→            try { $existing = (Get-ItemProperty -Path $Path -Name $Name -ErrorAction Stop).$Name } catch { }
   814→            if ($null -ne $existing) {
   815→                Remove-ItemProperty -Path $Path -Name $Name -Force -ErrorAction Stop 2>$null
   816→                Write-Success "Registry removed: $Path\$Name"
   817→            } else {
   818→                Write-Step "Already removed: $Path\$Name"
   819→            }
   820→        }
   821→    }
   822→    catch {
   823→        Write-Warn "Could not remove $Path\$Name : $_"
   824→    }

   832→        [ValidateSet("Safe","Moderate","Risky")]

   836→    )
   837→
   838→    # Generate a setting ID if not provided
   839→    if ([string]::IsNullOrEmpty($SettingID)) {
   840→        $SettingID = "Task_$($TaskPath.Replace('\','_').Replace(' ',''))_$TaskName"
   841→    }
   842→
   843→    # Register in setting catalog for interactive menu
   844→    if (-not [string]::IsNullOrEmpty($Description)) {
   845→        $script:SettingCatalog[$SettingID] = @{
   846→            Section  = $script:CurrentSection
   847→            Safety   = $SafetyLevel
   848→            Desc     = $Description
   849→            Action   = "Disable task: $TaskPath$TaskName"
   850→            Type     = "ScheduledTask"
   851→        }
   852→    }
   853→
   854→    # Collect risky descriptions for menu warnings
   855→    if ($SafetyLevel -eq "Risky" -and -not [string]::IsNullOrEmpty($Description)) {
   856→        $script:RiskyDescriptions += "$SettingID : $Description"
   857→    }
   858→
   859→    # Check if user has toggled this setting OFF via interactive menu
   860→    if ($script:SettingOverrides.ContainsKey($SettingID) -and $script:SettingOverrides[$SettingID]) {
   861→        Write-Host "  SKIPPED (user toggle): $Description" -ForegroundColor DarkGray
   862→        $script:TotalSkipped++
   863→        return
   864→    }
   865→
   866→    # Safety gate: Skip Risky settings unless explicitly allowed
   867→    if ($SafetyLevel -eq "Risky" -and -not $script:AllowRisky) {
   868→        Write-Host "  SKIPPED (risky): $Description — use -AllowRisky or enable in menu" -ForegroundColor Yellow
   869→        $script:TotalSkipped++
   870→        return
   871→    }
   872→
   873→    # Warn for Moderate settings
   874→    if ($SafetyLevel -eq "Moderate") {
   875→        Write-Host "  [MODERATE] $Description" -ForegroundColor DarkYellow
   876→    }
   877→
   878→    try {
   879→        $task = Get-ScheduledTask -TaskPath $TaskPath -TaskName $TaskName -ErrorAction SilentlyContinue
   880→        if ($task) {
   881→            if ($task.State -eq 'Disabled') {
   882→                Write-Step "Already disabled: $TaskPath$TaskName"
   883→                $script:TotalSkipped++
   884→            } else {
   885→                # Safety: Check for dependent services/tasks before disabling
   886→                $deps = $task.DependentOn
   887→                if ($deps) {
   888→                    Write-Host "  [DEPENDENCY] Task $TaskName has dependents: $($deps.TaskName -join ', ')" -ForegroundColor DarkYellow
   889→                }
   890→                $task | Disable-ScheduledTask -ErrorAction Stop
   891→                Write-Success "Task disabled: $TaskPath$TaskName"
   892→                $script:TotalChanged++
   893→                $null = $script:UndoCommands.Add("Enable-ScheduledTask -TaskPath '$TaskPath' -TaskName '$TaskName' -ErrorAction SilentlyContinue")
   894→                $null = $script:UndoDescriptions.Add("Re-enable scheduled task: $TaskPath$TaskName")
   895→            }
   896→        }
   897→    }
   898→    catch {
   899→        Write-Warn "Could not disable task $TaskPath$TaskName : $_"
   900→    }
   901→}
