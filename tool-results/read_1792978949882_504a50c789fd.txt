     4→  /** Setting IDs that this setting requires to be enabled */
     5→  requires?: string[];
     6→  /** Setting IDs that conflict with this setting (cannot both be enabled) */

     8→  /** Human-readable conflict message */
     9→  conflictMessage?: string;
    10→}

    14→  name: string;
    15→  safety: SafetyLevel;
    16→  desc: string;
    17→  /** Dependencies and conflicts with other settings */
    18→  deps?: SettingDependency;
    19→}

   613→/**
   614→ * Check for conflicts and unsatisfied dependencies given current settings state
   615→ */
   616→export function checkSettingConflicts(
   617→  settingId: string,
   618→  settingsState: Record<string, boolean>
   619→): { type: "conflict" | "unmet_require"; settingId: string; settingName: string; message: string }[] {
   620→  const warnings: { type: "conflict" | "unmet_require"; settingId: string; settingName: string; message: string }[] = [];
   621→  const deps = settingDependencies[settingId];
   622→  if (!deps) return warnings;
   623→
   624→  // Check conflicts
   625→  if (deps.conflicts) {
   626→    for (const conflictId of deps.conflicts) {
   627→      if (settingsState[conflictId]) {
   628→        // Find the name of the conflicting setting
   629→        let conflictName = conflictId;
   630→        for (const section of sections) {
   631→          const found = section.settings.find((s) => s.id === conflictId);
   632→          if (found) { conflictName = found.name; break; }
   633→        }
   634→        warnings.push({
   635→          type: "conflict",
   636→          settingId: conflictId,
   637→          settingName: conflictName,
   638→          message: deps.conflictMessage || `${settingId} conflicts with ${conflictName}`,
   639→        });
   640→      }
   641→    }
   642→  }
   643→
   644→  // Check unmet requirements (only warn when this setting is being enabled)
   645→  if (deps.requires && settingsState[settingId]) {
   646→    for (const reqId of deps.requires) {
   647→      if (!settingsState[reqId]) {
   648→        let reqName = reqId;
   649→        for (const section of sections) {
   650→          const found = section.settings.find((s) => s.id === reqId);
   651→          if (found) { reqName = found.name; break; }
   652→        }
   653→        warnings.push({
   654→          type: "unmet_require",
   655→          settingId: reqId,
   656→          settingName: reqName,
   657→          message: deps.conflictMessage || `${settingId} requires ${reqName} to be enabled`,
   658→        });
   659→      }
   660→    }
   661→  }
   662→
   663→  return warnings;
   664→}

   666→/**
   667→ * Get all conflicts for all currently enabled settings
   668→ */
   669→export function getAllConflicts(
   670→  settingsState: Record<string, boolean>
   671→): { settingId: string; settingName: string; type: "conflict" | "unmet_require"; targetId: string; targetName: string; message: string }[] {
   672→  const allWarnings: { settingId: string; settingName: string; type: "conflict" | "unmet_require"; targetId: string; targetName: string; message: string }[] = [];
   673→  const seen = new Set<string>();
   674→
   675→  for (const section of sections) {
   676→    for (const setting of section.settings) {
   677→      if (!settingsState[setting.id]) continue;
   678→      const deps = settingDependencies[setting.id];
   679→      if (!deps) continue;
   680→
   681→      if (deps.conflicts) {
   682→        for (const conflictId of deps.conflicts) {
   683→          if (settingsState[conflictId]) {
   684→            const key = [setting.id, conflictId].sort().join("|");
   685→            if (seen.has(key)) continue;
   686→            seen.add(key);
   687→            let conflictName = conflictId;
   688→            for (const s2 of sections) {
   689→              const found = s2.settings.find((s) => s.id === conflictId);
   690→              if (found) { conflictName = found.name; break; }
   691→            }
   692→            allWarnings.push({
   693→              settingId: setting.id,
   694→              settingName: setting.name,
   695→              type: "conflict",
   696→              targetId: conflictId,
   697→              targetName: conflictName,
   698→              message: deps.conflictMessage || `${setting.name} conflicts with ${conflictName}`,
   699→            });
   700→          }
   701→        }
   702→      }
   703→
   704→      if (deps.requires) {
   705→        for (const reqId of deps.requires) {
   706→          if (!settingsState[reqId]) {
   707→            const key = `${setting.id}|req|${reqId}`;
   708→            if (seen.has(key)) continue;
   709→            seen.add(key);
   710→            let reqName = reqId;
   711→            for (const s2 of sections) {
   712→              const found = s2.settings.find((s) => s.id === reqId);
   713→              if (found) { reqName = found.name; break; }
   714→            }
   715→            allWarnings.push({
   716→              settingId: setting.id,
   717→              settingName: setting.name,
   718→              type: "unmet_require",
   719→              targetId: reqId,
   720→              targetName: reqName,
   721→              message: deps.conflictMessage || `${setting.name} requires ${reqName}`,
   722→            });
   723→          }
   724→        }
   725→      }
   726→    }
   727→  }
   728→
   729→  return allWarnings;
   730→}
